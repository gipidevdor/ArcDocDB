;;;; Derivazione riproducibile di riepiloghi; dati letti, mai valutati.
(require :asdf)
(load "tools/evidence-storage.lisp")
(defun read-data (path) (arcdocdb.evidence:read-evidence path))
(defun save-data (data path)
  (with-open-file (stream path :direction :output :if-exists :error)
    (let ((*print-readably* t)) (write data :stream stream :pretty t) (terpri stream))))
(defun statistics (values)
  (unless (and values (every #'numberp values)) (error "Campioni mancanti."))
  (let* ((ordered (sort (copy-list values) #'<)) (n (length ordered)))
    (list :samples n :minimum (first ordered) :maximum (car (last ordered))
          :median (if (oddp n) (nth (floor n 2) ordered)
                      (/ (+ (nth (1- (floor n 2)) ordered) (nth (floor n 2) ordered)) 2)))))
(let* ((args (uiop:command-line-arguments))
       (bench (read-data (first args))) (commands (read-data (second args)))
       (windows (append (getf bench :serial) (getf bench :parallel)))
       (groups nil) (producers nil))
  (unless (and (= 3 (length args)) (eq :ok (getf bench :status))
               (= 80 (length (getf bench :serial))) (= 240 (length (getf bench :parallel)))
    (error "Benchmark incompleto."))
  (dolist (path (getf commands :commands))
    (let ((record (read-data path)))
      (unless (and (member (getf record :status) '(:ok :failed))
                   (eq :stable (getf record :source-consistency)))
        (error "Produttore senza esito stabile: ~A" path))
      (push (list :record path :status (getf record :status) :exit-code (getf record :exit-code)
                  :commit (getf (getf record :environment) :commit)
                  :command (getf record :command) :wall-seconds (getf record :wall-seconds)) producers)))
  (dolist (n '(1 8 32 64))
    (dolist (v '(16 256))
      (dolist (scenario '(:resolve :cache-hit))
        (dolist (execution '(:serial :parallel))
          (dolist (workers (if (eq execution :serial) '(1) '(1 2 4)))
            (let ((selected (remove-if-not
                             (lambda (w) (and (eql n (getf w :segment-count))
                                              (eql v (getf w :value-bytes))
                                              (eq scenario (getf w :scenario))
                                              (eq execution (getf w :execution))
                                              (eql workers (getf w :worker-count)))) windows)))
              (unless (and (= 5 (length selected))
                           (every (lambda (w) (and (eq :ok (getf w :status))
                                                  (getf w :quiescent)
                                                  (= (* 20000 workers) (getf w :completed)))) selected))
                (error "Cella incompleta."))
              (push (list :segment-count n :value-bytes v :scenario scenario :execution execution
                          :workers workers :ops-per-second (statistics (mapcar (lambda (w) (getf w :ops-per-second)) selected))
                          :seconds (statistics (mapcar (lambda (w) (getf w :seconds)) selected))
                          :heap-bytes (statistics (mapcar (lambda (w) (getf w :heap-bytes)) selected))) groups)))))))
  (let ((result (list :schema-version 1 :kind :get-directory-derived-results :status :ok
                      :source-benchmark (first args) :source-command-manifest (second args)
                      :measured-cycles (reduce #'+ windows :key (lambda (w) (getf w :completed)))
                      :warmup-cycles (reduce #'+ windows :key (lambda (w) (* (getf w :worker-count) (getf w :warmup-per-worker))))
                      :serial-windows 80 :parallel-windows 240
                      :zero-heap-serial-windows (count 0 (getf bench :serial) :key (lambda (w) (getf w :heap-bytes)))
                      :commands (nreverse producers) :groups (nreverse groups)
                      :limits '(:five-fixed-replicates :no-db-throughput-claim
                                :checks-inside-windows :resolve-two-selections
                                :cache-hit-one-selection :parallel-startup-and-join-included
                                :heap-global-allocated-not-rss :external-load-uncontrolled))))
    (unless (= 12800000 (getf result :measured-cycles)) (error "Conteggio campagna diverso."))
    (save-data result (third args))
    (with-open-file (md (make-pathname :type "md" :defaults (third args))
                        :direction :output :if-exists :error)
      (format md "# Benchmark directory GET CLOSED~%~%12.800.000 cicli misurati; 80 finestre seriali, 240 parallele; cinque repliche fisse.~%~%")
      (format md "Heap seriale osservato zero in ~D/80 finestre. Heap è memoria allocata nel processo, non RSS. Il parallelo include startup, gate e join; i controlli di correttezza sono dentro i tempi.~%~%" (getf result :zero-heap-serial-windows))
      (format md "| N | V | Ciclo | Seriale, mediana/s | 4 lettori, mediana/s | Rapporto | Heap mediano, 4 lettori B |~%|---:|---:|---|---:|---:|---:|---:|~%")
      (dolist (n '(1 8 32 64))
        (dolist (v '(16 256))
          (dolist (scenario '(:resolve :cache-hit))
            (flet ((find-group (execution workers)
                     (find-if (lambda (g) (and (= n (getf g :segment-count)) (= v (getf g :value-bytes))
                                              (eq scenario (getf g :scenario)) (eq execution (getf g :execution))
                                              (= workers (getf g :workers)))) (getf result :groups))))
              (let* ((serial (find-group :serial 1)) (parallel (find-group :parallel 4))
                     (a (getf (getf serial :ops-per-second) :median))
                     (b (getf (getf parallel :ops-per-second) :median)))
                (format md "| ~D | ~D | ~A | ~,0F | ~,0F | ~,2Fx | ~D |~%"
                        n v scenario a b (/ b a) (getf (getf parallel :heap-bytes) :median)))))))
      (format md "~%Resolve comprende due selezioni directory, avvio/abbandono e oracoli. Cache-hit comprende una selezione, lettura cache, copia completa e conclusione. CLOSED v2 corrente ordinario, CSN massimo u64, cache calda: questo rate non è throughput del database né una misura I/O sostenuta. Il dominio geometrico 65.536 è distinto dai profili nativi N≤64. Carico esterno non controllato; nessuna soglia prestazionale o P99. Minimi/massimi, uno e due lettori e raw sono nel riepilogo Lisp e nel report della campagna.~%"))
    (write result :pretty t) (terpri)))
