(require :asdf)
(load "tools/evidence-storage.lisp")
(defparameter *base* (truename "spikes/results/2026-10-11-get-directory/"))
(defun read-data (path) (arcdocdb.evidence:read-evidence path))
(defun save-data (data path)
  (with-open-file (out path :direction :output :if-exists :supersede)
    (let ((*print-readably* t)) (write data :stream out :pretty t) (terpri out))))
(defun exact-copy (source destination)
  (if (probe-file destination)
      (unless (equal (arcdocdb.evidence:file-sha256 source) (arcdocdb.evidence:file-sha256 destination))
        (error "Copia diversa: ~A" destination))
      (uiop:copy-file source destination)))
(defun commands ()
  (loop for dir in (sort (uiop:subdirectories "spikes/out/") #'string< :key #'namestring)
        for name = (car (last (pathname-directory dir)))
        for path = (merge-pathnames "report.lisp" dir)
        when (and (digit-char-p (char name 0)) (search "-command-" name) (probe-file path))
        append (let ((r (read-data path)))
                 (cond ((eq :running (getf r :status)) nil)
                       ((and (eq :command-verification (getf r :kind))
                             (member (getf r :status) '(:ok :failed))
                             (integerp (getf r :finished-at-universal-time))
                             (eq :stable (getf r :source-consistency))) (list (list name path r)))
                       (t (error "Comando senza chiusura stabile: ~A" path))))))
(defun raw-entry (name)
  (let ((path (merge-pathnames name *base*)))
    (list :file name :bytes (arcdocdb.evidence:file-bytes path)
          :sha256 (arcdocdb.evidence:file-sha256 path) :source :preserved-local-artifact)))
(defun prepare ()
  (let* ((path (merge-pathnames "publication-roots.lisp" *base*)) (r (read-data path)))
    (dolist (c (commands)) (pushnew (first c) (getf r :directories) :test #'string=))
    (save-data r path)))
(defun supplement ()
  (let* ((path (merge-pathnames "catalogo.lisp" *base*)) (catalog (read-data path)))
    (dolist (name '("benchmark-summary-f.lisp" "benchmark-summary-g.lisp" "history.lisp"))
      (let ((r (read-data (merge-pathnames name *base*))))
        (push (list :artifact name :kind (getf r :kind) :status (getf r :status)
                    :scope :derived-results-or-development-history
                    :sha256 (arcdocdb.evidence:file-sha256 (merge-pathnames name *base*)))
              (getf catalog :entries))))
    (dolist (name '("README.md" "report.md" "benchmark-summary-f.md" "benchmark-summary-g.md"
                    "derive-results.lisp" "derive-source.txt" "derive-source-accepted.txt"
                    "assemble-manifest.lisp" "assemble-source-attempt1.txt" "assemble-source-attempt2.txt"
                    "assemble-source-accepted.txt" "command-manifest.lisp"
                    "finish-publication.lisp" "finish-publication-source.txt"))
      (push (raw-entry name) (getf catalog :associated-raw-files)))
    (save-data catalog path)))
(defun copy-record (source destination)
  (with-open-file (in source)
    (let* ((*read-eval* nil) (descriptor (read in)))
      (if (eq :compressed-evidence (getf descriptor :kind))
          (let* ((new-name (concatenate 'string (file-namestring destination) ".gz"))
                 (payload (merge-pathnames (getf descriptor :payload) source)))
            (exact-copy payload (merge-pathnames new-name destination))
            (setf (getf descriptor :payload) new-name)
            (if (probe-file destination)
                (unless (equalp (read-data source) (read-data destination)) (error "Record diverso."))
                (save-data descriptor destination)))
          (exact-copy source destination))))
  (unless (equalp (read-data source) (read-data destination)) (error "Record risolto diverso.")))
(defun retain ()
  (let ((entries nil) (count 0) (bytes 0) (maximum 0))
    (dolist (c (commands))
      (let ((name (format nil "~A-report.lisp" (first c))))
        (copy-record (second c) (merge-pathnames name *base*))
        (push (raw-entry name) entries)
        (dolist (leaf (uiop:directory-files (uiop:pathname-directory-pathname (second c))))
          (unless (member (file-namestring leaf) '("report.lisp" "report.lisp.gz") :test #'string=)
            (let ((target (format nil "~A-~A" (first c) (file-namestring leaf))))
              (exact-copy leaf (merge-pathnames target *base*))
              (push (raw-entry target) entries))))))
    (labels ((walk (dir)
               (dolist (path (uiop:directory-files dir))
                 (let ((n (arcdocdb.evidence:file-bytes path)))
                   (when (> n 1048576) (error "File oltre 1 MiB: ~A" path))
                   (incf count) (incf bytes n) (setf maximum (max n maximum))))
               (dolist (sub (uiop:subdirectories dir)) (walk sub))))
      (walk *base*))
    (save-data (list :schema-version 1 :kind :get-directory-closure-inventory :status :ok
                     :records (nreverse entries) :files count :bytes bytes :largest-file-bytes maximum
                     :limits '(:point-in-time :own-running-record-excluded :no-self-certification))
               (merge-pathnames "closure-inventory.lisp" *base*))))
(let ((args (uiop:command-line-arguments)))
  (cond ((equal args '("--prepare")) (prepare))
        ((equal args '("--supplement")) (supplement))
        ((equal args '("--retain")) (retain))
        (t (error "Modalità pubblicazione sconosciuta.")))
  (format t "~S~%" (list :schema-version 1 :kind :get-directory-publication-step :status :ok :mode (first args))))
