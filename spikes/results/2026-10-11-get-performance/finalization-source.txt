(require :asdf)
(load "tools/evidence-storage.lisp")
(defun read-data (path) (arcdocdb.evidence:read-evidence path))
(defun save-data (path data)
  (ensure-directories-exist path)
  (with-open-file (s path :direction :output :if-exists :supersede)
    (let ((*print-readably* t)) (write data :stream s :pretty t) (terpri s))))
(defun git-output (&rest arguments)
  (string-trim '(#\Newline #\Return)
    (uiop:run-program (cons "git" arguments) :output :string :error-output :string)))
(defparameter *base* #P"spikes/results/2026-10-11-get-performance/")
(let ((args (uiop:command-line-arguments)))
 (cond
  ((equal args '("--source-equivalence"))
   (let* ((freeze "df8b98ec17baef9a4846ef2ca16559e424336db1")
          (head (git-output "rev-parse" "HEAD"))
          (delta (git-output "diff" "--name-status" freeze "HEAD" "--" "." ":!spikes/results/**"))
          (dirty (git-output "status" "--porcelain" "--" "." ":!spikes/results/**")))
    (unless (and (zerop (length delta)) (zerop (length dirty)))
      (error "Sorgenti diversi o non congelati: ~S ~S" delta dirty))
    (let ((data (list :schema-version 1 :kind :get-performance-final-source-equivalence
                     :status :ok :tested-freeze freeze :integrated-head head
                     :diff-paths nil :working-tree-outside-proofs dirty
                     :scope :repository-except-spikes-results
                     :limits '(:static-byte-equivalence :no-new-product-execution
                               :proof-artifacts-excluded))))
      (save-data #P"spikes/out/get-performance-final-source/report.lisp" data)
      (write data :pretty t) (terpri))))
  ((equal args '("--manifest"))
   (let ((names '("get-performance-summary" "get-performance-final-source" "get-performance-mutations-frozen")))
    (dolist (directory (uiop:subdirectories "spikes/out/"))
      (let* ((name (car (last (pathname-directory directory))))
             (report (merge-pathnames "report.lisp" directory)))
       (when (and (or (search "-command-" name) (search "-check-" name)) (probe-file report))
        (let ((data (read-data report)))
          (unless (member (getf data :status) '(:running :pending)) (push name names))))))
    (save-data (merge-pathnames "publication-roots.lisp" *base*)
      (list :schema-version 1 :kind :get-performance-publication-roots
            :directories (sort (remove-duplicates names :test #'equal) #'string<)
            :ownership :exclusive-get-verificato-worktree
            :limits '(:closed-records-only :explicit-common-roots :not-other-worktrees)))
    (format t "~D radici esplicite conservate.~%" (length names))))
  ((equal args '("--supplement"))
   (let* ((path (merge-pathnames "catalogo.lisp" *base*)) (data (read-data path)))
    (dolist (name '("sviluppo.lisp" "regressione-cache.lisp" "benchmark-summary.lisp"))
      (let* ((file (merge-pathnames name *base*)) (record (read-data file)))
        (unless (eql 1 (getf record :schema-version)) (error "Schema manuale invalido: ~A" file))
        (unless (find name (getf data :entries) :key (lambda (e) (getf e :artifact)) :test #'equal)
          (setf (getf data :entries)
            (append (getf data :entries)
              (list (list :artifact name :kind (getf record :kind) :status (getf record :status)
                          :scope :get-performance-derived-or-manual-record
                          :sha256 (arcdocdb.evidence:file-sha256 file))))))))
    (dolist (name '("summary-source.txt" "finalization-source.txt" "README.md" "risultati.md"))
      (let ((file (merge-pathnames name *base*)))
        (unless (find name (getf data :associated-raw-files) :key (lambda (e) (getf e :file)) :test #'equal)
          (setf (getf data :associated-raw-files)
            (append (getf data :associated-raw-files)
              (list (list :file name :bytes (arcdocdb.evidence:file-bytes file)
                          :sha256 (arcdocdb.evidence:file-sha256 file)
                          :source :owned-derived-or-documentary-artifact)))))))
    (save-data path data)
    (format t "Catalogo completato: ~D record, ~D file grezzi.~%"
            (length (getf data :entries)) (length (getf data :associated-raw-files)))))
  (t (error "Modo sconosciuto: ~S" args))))
