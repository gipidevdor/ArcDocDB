(require :asdf)
(load "tools/evidence-storage.lisp")
(defun save-data (path data)
 (with-open-file (s path :direction :output :if-exists :error)
  (let ((*print-readably* t)) (write data :stream s :pretty t) (terpri s))))
(defun read-data (path) (arcdocdb.evidence:read-evidence path))
(defun copy-exact (source target)
 (when (probe-file target) (error "Target gia presente: ~A" target))
 (uiop:copy-file source target)
 (unless (and (= (arcdocdb.evidence:file-bytes source) (arcdocdb.evidence:file-bytes target))
              (equal (arcdocdb.evidence:file-sha256 source) (arcdocdb.evidence:file-sha256 target)))
  (error "Copia diversa: ~A" target))
 (list :file (file-namestring target) :bytes (arcdocdb.evidence:file-bytes target)
       :sha256 (arcdocdb.evidence:file-sha256 target) :source (namestring source)))
(let* ((base #P"spikes/results/2026-10-11-get-performance/")
       (catalog (read-data (merge-pathnames "catalogo.lisp" base))) (copies nil))
 (dolist (spec '(("4000713127-command-97801-0" "manifest-final")
                ("4000713148-command-97931-0" "publication-final")
                ("4000713221-command-99817-0" "catalog-supplement")
                ("4000713238-command-127-0" "integrity-final")
                ("4000713238-command-128-0" "size-final")
                ("4000713238-command-129-0" "repository-evidence-final")))
  (let* ((source (merge-pathnames (format nil "spikes/out/~A/" (first spec)) (truename "./")))
         (command (read-data (merge-pathnames "report.lisp" source))))
   (unless (and (eq :ok (getf command :status)) (eql 0 (getf command :exit-code))
                (eq :stable (getf command :source-consistency)))
    (error "Chiusura non positiva o sorgenti cambiati: ~A" source))
   (dolist (leaf '("report.lisp" "conservazione.lisp"))
    (let* ((original (merge-pathnames leaf source))
           (retained (merge-pathnames (format nil "~A-~A" (second spec) leaf) base)))
     (with-open-file (s original)
      (let ((*read-eval* nil))
       (when (eq :compressed-evidence (getf (read s) :kind))
        (error "Registro conclusivo compresso inatteso: ~A" original))))
     (push (append (list :role (second spec)) (copy-exact original retained)) copies)))))
 (let* ((size-record (read-data (merge-pathnames "size-final-report.lisp" base)))
        (size (let ((*read-eval* nil)) (read-from-string (getf size-record :stdout))))
        (global (read-data (merge-pathnames "repository-evidence-final-report.lisp" base)))
        (global-output (getf global :stdout))
        (position (search "(:STATUS :OK" global-output))
        (global-data (when position (let ((*read-eval* nil))
                       (read-from-string global-output nil nil :start position)))))
  (unless (and (eq :ok (getf size :status)) (eq :ok (getf global-data :status)))
   (error "Sintesi degli audit mancante."))
  (push (copy-exact *load-truename* (merge-pathnames "seal-source.txt" base)) copies)
  (save-data (merge-pathnames "chiusura.lisp" base)
   (list :schema-version 1 :kind :get-performance-closure :status :ok
         :tested-freeze "df8b98ec17baef9a4846ef2ca16559e424336db1"
         :catalog-sha256 (arcdocdb.evidence:file-sha256 (merge-pathnames "catalogo.lisp" base))
         :catalog-records (length (getf catalog :entries))
         :catalog-raw-files (length (getf catalog :associated-raw-files))
         :pre-closure-size (getf size :roots) :repository-evidence-observation global-data
         :copies (nreverse copies)
         :limits '(:final-audits-observe-closed-catalog :closure-files-outside-catalog
                   :static-source-equivalence-no-new-product-execution
                   :size-before-addition-of-closure-records :point-in-time-observations)))
  (format t "Chiusura conservata: ~D record, ~D file grezzi, audit positivi e sorgenti stabili.~%"
          (length (getf catalog :entries)) (length (getf catalog :associated-raw-files)))))
