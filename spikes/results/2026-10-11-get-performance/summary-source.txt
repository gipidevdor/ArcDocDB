(require :asdf)
(load "tools/evidence-storage.lisp")
(defun median5 (xs)
  (unless (= (length xs) 5) (error "Expected five replicas."))
  (third (sort (copy-list xs) #'<)))
(unless (= 3 (median5 '(5 1 3 2 4))) (error "Median self-test."))
(let* ((source "spikes/out/get-performance-bench-frozen/report.lisp")
       (bench (arcdocdb.evidence:read-evidence source))
       (serial (getf bench :serial)) (parallel (getf bench :parallel))
       (base #p"spikes/results/2026-10-11-get-performance/") (cells nil))
  (unless (and (eq :ok (getf bench :status)) (= 90 (length serial)) (= 270 (length parallel)))
    (error "Incomplete benchmark."))
  (dolist (execution '(:serial :parallel))
    (dolist (scenario '(:cache-hit :segment :absent))
      (dolist (size '(16 256 4096))
        (dolist (csn '(7 #xffffffffffffffff))
          (dolist (workers (if (eq execution :serial) '(1) '(1 2 4)))
            (let* ((rows (remove-if-not
                         (lambda (r) (and (eq scenario (getf r :scenario)) (= size (getf r :value-bytes))
                                          (= csn (getf r :csn)) (= workers (getf r :worker-count))))
                         (getf bench execution)))
                   (rates (mapcar (lambda (r) (getf r :ops-per-second)) rows))
                   (heaps (mapcar (lambda (r) (getf r :heap-bytes)) rows)))
              (push (list :execution execution :scenario scenario :value-bytes size :csn csn
                          :workers workers :replicas 5 :ops-per-second-median (median5 rates)
                          :ops-per-second-min (apply #'min rates) :ops-per-second-max (apply #'max rates)
                          :heap-bytes-median (median5 heaps) :heap-bytes-min (apply #'min heaps)
                          :heap-bytes-max (apply #'max heaps)) cells)))))))
  (setf cells (nreverse cells))
  (let ((summary (list :schema-version 1 :kind :derived-get-performance-summary :status :ok
                       :source-freeze "df8b98ec17baef9a4846ef2ca16559e424336db1"
                       :source source :source-sha256 (arcdocdb.evidence:file-sha256 source)
                       :generator "summary-source.txt" :generator-sha256 (arcdocdb.evidence:file-sha256 *load-truename*)
                       :median-self-test :ok :measured-operations (reduce #'+ (append serial parallel) :key (lambda (r) (getf r :completed)))
                       :serial-windows 90 :parallel-windows 270
                       :serial-zero-heap-windows (count 0 serial :key (lambda (r) (getf r :heap-bytes)))
                       :serial-heap-bytes (reduce #'+ serial :key (lambda (r) (getf r :heap-bytes)))
                       :parallel-zero-heap-windows (count 0 parallel :key (lambda (r) (getf r :heap-bytes)))
                       :parallel-heap-bytes (reduce #'+ parallel :key (lambda (r) (getf r :heap-bytes)))
                       :cells cells :limits '(:whole-process-heap :validation-in-measured-loop :parallel-thread-start-and-join-in-window
                                             :external-load-uncontrolled :no-server-latency :no-mcdc :no-full-engine-qualification))))
    (dolist (path (list (merge-pathnames "benchmark-summary.lisp" base) #p"spikes/out/get-performance-summary/report.lisp"))
      (with-open-file (s path :direction :output :if-exists :error)
        (let ((*print-readably* t)) (write summary :stream s :pretty t) (terpri s))))
    (uiop:copy-file *load-truename* (merge-pathnames "summary-source.txt" base))
    (with-open-file (s (merge-pathnames "risultati.md" base) :direction :output :if-exists :error)
      (format s "# GET: correzione della cache e prestazioni~%~%Baseline `a1b00e9`: la regressione della copia reinserita fallisce, con sorgenti stabili. Dopo la correzione, il posto rifiutato/disabilitato viene saltato e la copia valida resta raggiungibile entro otto way. Verifica seqlock, CRC, identità e CSN mantenuta.~%~%Freeze integrato `df8b98e`: **836 test nominati + 2 smoke**, 19 test GET dedicati, 7/7 mutanti rilevati, 10 spike e controlli del repository passati. Gli agenti non hanno eseguito prove preliminari non registrate.~%~%## Misure osservate~%~%**~D GET** misurati, 90 finestre seriali e 270 gruppi paralleli. Cinque repliche per cella, COUNT 20.000 per worker e warmup 1.000, senza adattamento. Chiave 16 B, otto chiavi presenti/assenti, record v2 non prepared.~%~%Heap seriale: ~D/90 finestre con delta zero, totale ~D B. Heap parallelo: ~D/270 con delta zero, totale ~D B. Contatore del processo; avvio/join dei thread inclusi. Nessuna attribuzione automatica dei byte al kernel GET.~%~%Mediane in milioni di operazioni/s. La misura comprende oracolo completo della risposta, sentinelle, clock e controlli EBR. Il segmento usa page cache riscaldata e cache del DB disattivata.~%~%| Caso | Valore B | CSN | Seriale | Thread 1 | Thread 2 | Thread 4 | Heap seriale mediano B | Heap thread 4 mediano B |~%|---|---:|---|---:|---:|---:|---:|---:|---:|~%"
              (getf summary :measured-operations) (getf summary :serial-zero-heap-windows)
              (getf summary :serial-heap-bytes) (getf summary :parallel-zero-heap-windows) (getf summary :parallel-heap-bytes))
      (dolist (scenario '(:cache-hit :segment :absent))
        (dolist (size '(16 256 4096))
          (dolist (csn '(7 #xffffffffffffffff))
            (let ((selected (remove-if-not (lambda (c) (and (eq scenario (getf c :scenario)) (= size (getf c :value-bytes)) (= csn (getf c :csn)))) cells)))
              (format s "| ~A | ~D | ~A |~{ ~,3F |~} ~D | ~D |~%"
                      scenario size (if (= csn 7) "7" "MAX-u64")
                      (mapcar (lambda (c) (/ (getf c :ops-per-second-median) 1000000d0)) selected)
                      (getf (first selected) :heap-bytes-median) (getf (fourth selected) :heap-bytes-median))))))
      (format s "~%## Strumenti, integrità e limiti~%~%Autoverifica: 72 finestre funzionali e 23 diagnostiche; sensore positivo con quattro MiB trattenuti. I due comandi negativi escono con 1; la prova del self-test parziale conserva una finestra e la diagnostica del campo mancante. I fallimenti attesi rimangono nei registri.~%~%Copertura SBCL selezionata, compreso `cache/read`: **955/1324 espressioni, 71/158 rami**. Tutte le guardie restano nel denominatore. Non è MC/DC: prepared, matrice completa di errori e integrazione del server appartengono a campagne distinte. I rapporti HTML e lo stato grezzo sono conservati.~%~%Carico host, scheduler e cache CPU non controllati; il test usa fixture piccole, un singolo insieme cache condiviso e thread nuovi. Non misura pool persistenti, P99, rete, snapshot nel benchmark, migrazione, storico, NVMe o durability. Zero heap osservato nelle finestre non costituisce una garanzia universale.~%~%[Metodo preregistrato](../../../docs/implementazione/get-performance-metodo.md), [revisione indipendente](../../../docs/implementazione/get-performance-revisione.md), [dati derivati](benchmark-summary.lisp). Ambiente completo, argv, sorgenti prima/dopo e output grezzi sono nei registri del [catalogo](catalogo.lisp).~%"))
    (write (list :operations (getf summary :measured-operations) :serial-zero (getf summary :serial-zero-heap-windows)
                 :serial-heap (getf summary :serial-heap-bytes) :parallel-heap (getf summary :parallel-heap-bytes)
                 :example (first cells))) (terpri)))
