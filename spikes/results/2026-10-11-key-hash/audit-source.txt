;;;; Audit operativo della campagna: dati, byte e hash; nessuna esecuzione dei record.
(require :asdf)
(load "tools/evidence-storage.lisp")
(let* ((base (truename "spikes/results/2026-10-11-key-hash/"))
       (catalog (merge-pathnames "catalogo.lisp" base))
       (data (arcdocdb.evidence:read-evidence catalog))
       (records 0) (raw 0) (decoded 0) (files 0) (bytes 0)
       (maximum 0) (maximum-file nil) (raw-names (make-hash-table :test #'equal)))
  (labels ((target (name)
             (let ((path (merge-pathnames name base)))
               (unless (and (uiop:relative-pathname-p name) (uiop:subpathp path base)
                            (not (member :up (pathname-directory path))))
                 (error "Percorso fuori campagna: ~S" name))
               (unless (probe-file path) (error "File assente: ~A" name))
               path))
           (check-hash (path expected)
             (unless (equal expected (arcdocdb.evidence:file-sha256 path))
               (error "SHA256 diverso: ~A" path)))
           (walk (directory)
             (dolist (path (uiop:directory-files directory))
               (let* ((size (arcdocdb.evidence:file-bytes path))
                      (limit (if (equal "gz" (pathname-type path)) (* 8 1024 1024) (* 1024 1024))))
                 (when (or (> size limit) (equal "fasl" (pathname-type path))
                           (search ".candidate-" (file-namestring path))
                           (search ".decoded-" (file-namestring path)))
                   (error "File fuori budget o temporaneo: ~A" path))
                 (incf files) (incf bytes size)
                 (when (> size maximum)
                   (setf maximum size maximum-file (enough-namestring path base)))))
             (dolist (subdirectory (uiop:subdirectories directory)) (walk subdirectory))))
    (unless (and (eql 1 (getf data :schema-version))
                 (eq :evidence-catalog (getf data :kind))
                 (eq :key-hash (getf data :scope)))
      (error "Catalogo inatteso."))
    (dolist (entry (getf data :entries))
      (let* ((path (target (getf entry :artifact)))
             (record (arcdocdb.evidence:read-evidence path)))
        (check-hash path (getf entry :sha256))
        (unless (and (eql 1 (getf record :schema-version))
                     (eq (getf record :kind) (getf entry :kind))
                     (eq (getf record :status) (getf entry :status))
                     (not (eq :running (getf record :status))))
          (error "Record concluso o metadata diversi: ~A" path))
        (incf records)))
    (dolist (entry (getf data :associated-raw-files))
      (let* ((name (getf entry :file)) (path (target name)))
        (when (gethash name raw-names) (error "Raw duplicato: ~A" name))
        (setf (gethash name raw-names) t)
        (check-hash path (getf entry :sha256))
        (unless (= (getf entry :bytes) (arcdocdb.evidence:file-bytes path))
          (error "Dimensione raw diversa: ~A" path))
        (when (getf entry :codec)
          (unless (eq :gzip (getf entry :codec)) (error "Codec inatteso."))
          (uiop:with-temporary-file (:pathname temporary :keep nil)
            (uiop:run-program (list "gzip" "-d" "-c" (namestring path)) :output temporary)
            (unless (= (getf entry :uncompressed-bytes) (arcdocdb.evidence:file-bytes temporary))
              (error "Dimensione decodificata diversa: ~A" path))
            (check-hash temporary (getf entry :uncompressed-sha256)))
          (incf decoded))
        (incf raw)))
    (walk base)
    (write (list :schema-version 1 :kind :key-hash-integrity-audit :status :ok
                 :catalog-sha256 (arcdocdb.evidence:file-sha256 catalog)
                 :audit-source-sha256 (arcdocdb.evidence:file-sha256 *load-truename*)
                 :records records :raw-files raw :decoded-raw-files decoded
                 :stored-files files :stored-bytes bytes :maximum-file maximum-file
                 :maximum-file-bytes maximum
                 :limits '(:integrity-size-metadata :no-result-reinterpretation))
           :readably t :pretty t)
    (terpri)))
