(require :asdf)
(let ((sb-ext:*posix-argv* '("sbcl" "--library")))
  (load "tools/get-active-init-mutation.lisp"))
(in-package #:arcdocdb.get.active.init.mutation)
(let* ((base "spikes/out/get-series-mutations-active-empty-C/raw/evidence/baseline/")
       (config (read-record (merge-pathnames "config.lisp" base)))
       (child (read-record (merge-pathnames "child-report.lisp" base)))
       (copy (getf config :copy-directory))
       (expected (getf config :compiled-source-files))
       (entries (getf child :compiled-files))
       (tests (getf config :expected-tests)))
  (let ((*print-readably* nil) (*print-escape* t))
    (write
     (list :schema-version 1 :kind :active-empty-verification-debug :status :ok
           :config-keys (loop for (key val) on config by #'cddr collect key)
           :copy copy :expected-files (length expected) :entries (length entries)
           :loaded-set-equal (equal expected (getf child :loaded-source-files))
           :entry-set-equal (equal (sort (mapcar (lambda (e) (getf e :file)) entries) #'string<) expected)
           :registered-equal (equal tests (getf child :registered-tests))
           :manifest-before-equal (equal (getf config :manifest) (getf child :source-manifest-before))
           :manifest-after-equal (equal (getf config :manifest) (getf child :source-manifest))
           :fasl-first (getf (first entries) :fasl)
           :config-hash-equal (string= (sha256 (merge-pathnames "config.lisp" base)) (getf child :config-sha256))
           :source-first (first entries)) :pretty t)
    (terpri)))
