(let* ((base "spikes/results/2026-10-13-get-active-vuoto/")
        (files nil) (records nil))
  (dolist (spec '(("publication" "4000915581-command-31977-0" :ok)
                  ("completion" "4000915729-command-35651-0" :ok)
                  ("audit" "4000915742-command-35906-0" :ok)
                  ("retention-prerequisite-failure" "4000915825-command-37101-0" :failed)))
    (let* ((phase (first spec))
           (source (format nil "spikes/out/~A/" (second spec)))
           (target (format nil "~Achiusura/~A/" base phase))
           (record (arcdocdb.evidence:read-evidence (merge-pathnames "report.lisp" source))))
      (unless (and (eq :command-verification (getf record :kind))
                   (eq (third spec) (getf record :status))
                   (eq :stable (getf record :source-consistency))
                   (eql (if (eq :ok (third spec)) 0 1) (getf record :exit-code))
                   (integerp (getf record :finished-at-universal-time)))
        (error "Produttore di chiusura non riuscito o non concluso: ~A" source))
      (push (list :phase phase :directory (second spec) :status (getf record :status)
                  :finished-at-universal-time (getf record :finished-at-universal-time)) records)
      (ensure-directories-exist (merge-pathnames "report.lisp" target))
      (dolist (leaf '("report.lisp" "report.lisp.gz" "conservazione.lisp"))
        (let ((src (merge-pathnames leaf source)) (dst (merge-pathnames leaf target)))
          (when (probe-file src)
            (unless (probe-file dst) (uiop:copy-file src dst))
            (unless (equal (arcdocdb.evidence:file-sha256 src)
                           (arcdocdb.evidence:file-sha256 dst))
              (error "Copia di chiusura discordante: ~A" dst))
            (push (list :file (enough-namestring dst base)
                        :bytes (arcdocdb.evidence:file-bytes dst)
                        :sha256 (arcdocdb.evidence:file-sha256 dst)
                        :source (namestring (truename src))) files))))
      (unless (equalp record (arcdocdb.evidence:read-evidence
                              (merge-pathnames "report.lisp" target)))
        (error "Record di chiusura risolto diverso."))))
  (push (list :file "retention-attempt-A-source.txt"
              :bytes (arcdocdb.evidence:file-bytes (merge-pathnames "retention-attempt-A-source.txt" base))
              :sha256 (arcdocdb.evidence:file-sha256 (merge-pathnames "retention-attempt-A-source.txt" base))) files)
  (push (list :file "retention-source.txt"
              :bytes (arcdocdb.evidence:file-bytes (merge-pathnames "retention-source.txt" base))
              :sha256 (arcdocdb.evidence:file-sha256 (merge-pathnames "retention-source.txt" base)))
        files)
  (let* ((manifest (list :schema-version 1 :kind :active-empty-closing-retention :status :ok
                         :integration-head "fee62ba8453d7bc992c6f7e7563bdeb47b3eeb4a"
                         :catalog-sha256
                         (arcdocdb.evidence:file-sha256 (merge-pathnames "catalogo.lisp" base))
                         :records (nreverse records) :files (nreverse files)
                         :limits '(:local-copy-integrity :no-host-authentication
                                   :closing-producer-stored-separately :failed-record-not-accepted-proof)))
         (*print-readably* nil) (*print-escape* t) (*print-length* nil) (*print-level* nil))
    (with-open-file (out (merge-pathnames "chiusura/manifesto.lisp" base)
                          :direction :output :if-exists :error)
      (write manifest :stream out :pretty t) (terpri out))
    (write manifest :pretty t) (terpri)))
