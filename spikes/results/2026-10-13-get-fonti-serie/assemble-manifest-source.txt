;;;; Elenco esplicito delle sole prove possedute da questo checkout isolato.
(require :asdf)
(load "tools/evidence-storage.lisp")
(defun save-data (data path)
  (with-open-file (s path :direction :output :if-exists :error)
    (let ((*print-readably* nil) (*print-escape* t)) (write data :stream s :pretty t) (terpri s))))
(let ((roots nil) (commands nil) (raw nil)
      (base "spikes/results/2026-10-13-get-fonti-serie/"))
  (dolist (directory (sort (uiop:subdirectories "spikes/out/") #'string< :key #'namestring))
    (let* ((name (car (last (pathname-directory directory)))) (path (merge-pathnames "report.lisp" directory)))
      (when (and (digit-char-p (char name 0)) (or (search "-command-" name) (search "-check-" name))
                 (probe-file path))
        (let ((data (arcdocdb.evidence:read-evidence path)))
          (unless (eq :running (getf data :status))
            (unless (member (getf data :status) '(:ok :failed :source-changed :complete))
              (error "Produttore non terminale: ~A" path))
            (push name roots)
            (when (eq :command-verification (getf data :kind)) (push (cons name data) commands)))))))
  (dolist (directory (uiop:subdirectories "spikes/out/"))
    (let ((name (car (last (pathname-directory directory)))))
      (when (some (lambda (prefix) (uiop:string-prefix-p prefix name))
                  '("get-series-coverage" "get-series-mutations" "get-series-bench" "get-series-self-test"))
        (let* ((argument (format nil "spikes/out/~A/" name))
               (producer (find-if (lambda (entry) (member argument (getf (cdr entry) :command) :test #'string=))
                                  commands)))
          (unless producer (error "Raw senza produttore chiuso: ~A" name))
          (push (list :directory name :producer-directory (car producer)) raw)))))
  (save-data (list :schema-version 1 :kind :get-series-publication-roots
                   :directories (nreverse roots) :raw-roots (nreverse raw))
             (concatenate 'string base "publication-roots.lisp"))
  (save-data (list :schema-version 1 :kind :get-series-command-manifest
                   :commands (mapcar (lambda (entry) (format nil "spikes/out/~A/report.lisp" (car entry)))
                                     (nreverse commands)))
             (concatenate 'string base "command-manifest.lisp"))
  (format t "Manifesto: ~D radici chiuse, ~D raw.~%" (length roots) (length raw)))
