;;;; C4: collega bytes raw e helper esclusi allo stdout del produttore registrato.
;;; REQ: REQ-VAL-001 REQ-AFF-012
(require :asdf)
(load "tools/evidence-storage.lisp")
(defun source-digests (argv)
  "Solo sorgenti invocati da SBCL, mai dati .lisp ricevuti o prodotti dal helper."
  (loop for argument in (remove-duplicates
                        (loop for tail on argv
                              when (member (first tail) '("--script" "--load" "-load") :test #'string=)
                                collect (second tail)) :test #'string=)
        when (and (equal "lisp" (pathname-type argument)) (probe-file argument)) collect
          (list :file argument :sha256 (arcdocdb.evidence:file-sha256 argument))))
(defun raw-digests (root)
  "Inventario bounded delle prove, escludendo checkout e cache ricostruibili."
  (let ((records nil) (count 0))
    (labels ((walk (directory)
               (dolist (file (uiop:directory-files directory))
                 (when (> (incf count) 8192) (error "Inventario raw oltre 8192 file."))
                 (push (list :file (enough-namestring file root)
                             :bytes (arcdocdb.evidence:file-bytes file)
                             :sha256 (arcdocdb.evidence:file-sha256 file)) records))
               (dolist (subdirectory (uiop:subdirectories directory))
                 (unless (member (car (last (pathname-directory subdirectory)))
                                 '("fasl" "src" "tests" "copies" "isolatedtests") :test #'string=)
                   (walk subdirectory)))))
      (when (and root (probe-file root)) (walk root)))
    (sort records #'string< :key (lambda (r) (getf r :file)))))
(defun check-source-digests (expected argv)
  (unless (equal expected (source-digests argv)) (error "Sorgente helper cambiato durante il comando.")))
(defun self-test ()
  (let* ((root (merge-pathnames (format nil "spikes/out/get-series-capture-self-~D/" (sb-posix:getpid))
                               (truename "./")))
         (path (merge-pathnames "source.lisp" root)) (checks 0))
    (when (probe-file root) (error "Namespace self-test esistente."))
    (ensure-directories-exist path)
    (with-open-file (s path :direction :output :if-exists :error) (write-string "(:a 1)" s))
    (let* ((data (merge-pathnames "data.lisp" root)) (output (merge-pathnames "output.lisp" root))
           (argv (list "sbcl" "--script" (namestring path) (namestring data) (namestring output)))
           (expected (source-digests argv)))
      (unless (= 1 (length expected)) (error "Helper invocato non riconosciuto.")) (incf checks)
      (check-source-digests expected argv) (incf checks)
      (unless (= 1 (length (raw-digests root))) (error "Inventario self-test errato.")) (incf checks)
      (with-open-file (s data :direction :output :if-exists :error) (write-string "(:input 1)" s))
      (with-open-file (s output :direction :output :if-exists :error) (write-string "(:output 1)" s))
      (check-source-digests expected argv) (incf checks)
      (with-open-file (s data :direction :output :if-exists :supersede) (write-string "(:input 2)" s))
      (check-source-digests expected argv) (incf checks)
      (unless (equal expected (source-digests (list "sbcl" "--load" (namestring path)
                                                   "--load" (namestring path))))
        (error "Helper --load ripetuto non riconosciuto.")) (incf checks)
      (with-open-file (s path :direction :output :if-exists :supersede) (write-string "(:a 2)" s))
      (let ((rejected nil))
        (handler-case (check-source-digests expected argv) (error () (setf rejected t)))
        (unless rejected (error "Mutazione helper accettata.")) (incf checks))
      (delete-file data) (delete-file output))
    (delete-file path) (sb-posix:rmdir (namestring root))
    (list :checks checks :helper-mutation-rejected t :raw-inventory-observed t)))
(defun main ()
  (let* ((args (uiop:command-line-arguments)) (separator (position "--" args :test #'string=))
         (argv (when separator (subseq args (1+ separator)))) (raw-root nil) (code 1)
         (record (list :schema-version 1 :kind :get-series-capture :status :running
                       :command argv :helper-sources-before nil :helper-sources-after nil :raw-root nil
                       :raw-files nil :exit-code nil :diagnostic nil
                       :limits '(:single-owner-frozen-helper-source :raw-digests-at-command-exit
                                 :not-hostile-host-authentication :nested-command-owns-timeout))))
    (handler-case
        (cond ((equal args '("--self-test"))
               (setf (getf record :self-test) (self-test) code 0 (getf record :status) :ok))
              (t
               (unless (and argv (or (= separator 0)
                                    (and (= separator 2) (string= "--raw-root" (first args)))))
                 (error "Uso: --self-test oppure [--raw-root directory/] -- argv..."))
               (when (= separator 2)
                 (setf (getf record :raw-root) (second args)
                       raw-root (merge-pathnames (uiop:ensure-directory-pathname (second args)) (truename "./")))
                 (unless (uiop:subpathp raw-root (merge-pathnames "spikes/out/" (truename "./")))
                   (error "Radice raw fuori da spikes/out.")))
               (setf (getf record :helper-sources-before) (source-digests argv))
               (multiple-value-bind (stdout stderr exit)
                   (uiop:run-program argv :output :string :error-output :string :ignore-error-status t)
                 (write-string stdout) (write-string stderr *error-output*) (setf code exit))
               (setf (getf record :helper-sources-after) (source-digests argv)
                     (getf record :raw-files) (raw-digests raw-root))
               (check-source-digests (getf record :helper-sources-before) argv)
               (setf (getf record :status) (if (zerop code) :ok :failed))))
      (error (condition) (setf code 1 (getf record :status) :failed
                              (getf record :diagnostic) (princ-to-string condition))))
    (setf (getf record :exit-code) code (getf record :finished-at-universal-time) (get-universal-time))
    (let ((*print-readably* nil) (*print-escape* t) (*print-pretty* nil) (*print-length* nil) (*print-level* nil))
      (format t "~&GET-SERIES-CAPTURE ~S~%" record))
    (unless (zerop code) (format *error-output* "tools/get-series-capture.lisp COD-61: comando fallito, exit ~D.~%" code))
    (sb-ext:exit :code code)))
(main)
