;;;; Raccolta di gruppi espliciti e completi, incluse cache FASL come soli dati.
(require :asdf)
(load "tools/evidence-storage.lisp")
(defpackage #:resolver.collection (:use #:cl))
(in-package #:resolver.collection)
(declaim (optimize (safety 3) (debug 3)))
(defparameter *collector-source* *load-truename*)
(defconstant +plain-limit+ 1048576)
(defconstant +gzip-limit+ 8388608)
(defconstant +expanded-limit+ 134217728)

(defun read-one (path)
  (arcdocdb.evidence:read-evidence path))
(defun save-new (path data)
  (ensure-directories-exist path)
  (with-open-file (stream path :direction :output :if-exists :error)
    (let ((*print-readably* t)) (write data :stream stream :pretty t) (terpri stream))))
(defun fingerprint (path)
  (list :bytes (arcdocdb.evidence:file-bytes path) :sha256 (arcdocdb.evidence:file-sha256 path)))
(defun leaf-p (value)
  (and (stringp value) (plusp (length value))
       (not (member value '("." "..") :test #'equal))
       (not (find-if (lambda (c) (or (find c "/\\:*?[]") (< (char-code c) 32))) value))))
(defun files-in (root)
  (let ((files nil) (count 0))
    (labels ((walk (directory depth)
               (when (> depth 64) (error "Directory troppo profonda."))
               (dolist (entry (directory (make-pathname :name :wild :type :wild :defaults directory)
                                         :resolve-symlinks nil))
                 (when (> (incf count) 100000) (error "Inventario troppo grande."))
                 (let* ((name (string-right-trim "/" (namestring entry)))
                        (mode (sb-posix:stat-mode (sb-posix:lstat name))))
                   (cond ((sb-posix:s-isdir mode) (walk entry (1+ depth)))
                         ((sb-posix:s-isreg mode) (push entry files))
                         (t (error "File speciale o symlink: ~A." entry)))))))
      (walk root 0))
    (sort files #'string< :key #'namestring)))

(defun copy-verified (source target root)
  (let* ((before (fingerprint source)) (size (getf before :bytes))
         (compressed (and (> size +plain-limit+) (not (equal "gz" (pathname-type source)))))
         (actual (if compressed (pathname (concatenate 'string (namestring target) ".gz")) target)))
    (unless (<= size +expanded-limit+) (error "Sorgente oltre budget: ~A." source))
    (when (probe-file actual) (error "Destinazione esistente: ~A." actual))
    (ensure-directories-exist actual)
    (if compressed
        (let ((decoded (pathname (concatenate 'string (namestring actual) ".decoded"))))
          (uiop:run-program (list "gzip" "-n" "-9" "-c" (namestring source)) :output actual)
          (unwind-protect
               (progn
                 (uiop:run-program (list "gzip" "-d" "-c" (namestring actual)) :output decoded)
                 (unless (equal before (fingerprint decoded)) (error "Espansione diversa: ~A." source)))
            (when (probe-file decoded) (delete-file decoded))))
        (progn (uiop:copy-file source actual)
               (unless (equal before (fingerprint actual)) (error "Copia diversa: ~A." source))))
    (unless (equal before (fingerprint source)) (error "Sorgente cambiata: ~A." source))
    (let ((stored (fingerprint actual)))
      (unless (<= (getf stored :bytes) (if (equal "gz" (pathname-type actual)) +gzip-limit+ +plain-limit+))
        (error "Artefatto oltre budget: ~A." actual))
      (append (list :path (enough-namestring actual root) :source (namestring source)) stored
              (when compressed (list :codec :gzip :uncompressed-bytes size
                                     :uncompressed-sha256 (getf before :sha256)))))))

(defun collect (manifest-path destination)
  (let* ((manifest (read-one manifest-path))
         (root (merge-pathnames (uiop:ensure-directory-pathname destination) (truename "./")))
         (files nil) (processes nil))
    (unless (and (eql 1 (getf manifest :schema-version)) (getf manifest :groups))
      (error "Manifesto invalido."))
    (when (probe-file root) (error "Archivio esistente: ~A." root))
    (dolist (group (getf manifest :groups))
      (destructuring-bind (label source-directory &optional process-p) group
        (unless (leaf-p label) (error "Nome gruppo invalido."))
        (let* ((base (truename (uiop:ensure-directory-pathname source-directory)))
               (target-base (merge-pathnames (format nil "raw/~A/" label) root))
               (before (files-in base)))
          (when (and process-p (> (arcdocdb.evidence:file-bytes (merge-pathnames "report.lisp" base))
                                 +plain-limit+))
            (error "Il produttore deve compattare il suo report prima della raccolta."))
          (dolist (file before)
            (push (copy-verified file (merge-pathnames (enough-namestring file base) target-base) root) files))
          (unless (equal before (files-in base)) (error "Insieme sorgente cambiato: ~A." base))
          (when process-p
            (let* ((source (merge-pathnames "report.lisp" base))
                   (target (merge-pathnames "report.lisp" target-base)) (data (read-one source)))
              (unless (and (eql 1 (getf data :schema-version)) (eq :command-verification (getf data :kind))
                           (member (getf data :status) '(:ok :failed :source-changed))
                           (equalp data (read-one target)))
                (error "Record incompleto o diverso: ~A." source))
              (push (list :label label :record-path (enough-namestring target root)
                          :command (getf data :command) :status (getf data :status)
                          :exit-code (getf data :exit-code) :source-consistency (getf data :source-consistency))
                    processes))))))
    (push (copy-verified (truename manifest-path) (merge-pathnames "collection-manifest.lisp" root) root) files)
    (push (copy-verified (truename *collector-source*) (merge-pathnames "collection-source.txt" root) root) files)
    (dolist (item (getf manifest :files))
      (destructuring-bind (label source) item
        (unless (leaf-p label) (error "Nome file invalido."))
        (push (copy-verified (truename source) (merge-pathnames label root) root) files)))
    (let ((paths (mapcar (lambda (entry) (getf entry :path)) files)))
      (unless (= (length paths) (length (remove-duplicates paths :test #'equal))) (error "Path duplicato.")))
    (save-new (merge-pathnames "archive-index.lisp" root)
              (list :schema-version 1 :kind :original-command-archive
                    :metadata-policy :read-without-result-inference :processes (nreverse processes)
                    :files (sort files #'string< :key (lambda (entry) (getf entry :path)))
                    :limits '(:complete-explicit-groups :raw-fasl-preserved-as-data
                              :source-and-copy-sha256 :lossless-gzip-for-large-raw :no-result-promotion)))
    (save-new (merge-pathnames "catalogo.lisp" root)
              (list :schema-version 1 :kind :evidence-catalog :pending nil :entries
                    (list (list :kind :original-command-archive :artifact "archive-index.lisp"
                                :artifact-schema-version 1 :artifact-kind :original-command-archive))))
    (format t "~D file conservati, ~D comandi.~%" (length files) (length processes))))

(defun append-group (destination label source-directory)
  "Aggiunge un gruppo completo; riscrive soltanto l'indice derivato, mai originali."
  (unless (leaf-p label) (error "Nome gruppo invalido."))
  (let* ((root (truename (uiop:ensure-directory-pathname destination)))
         (index-path (merge-pathnames "archive-index.lisp" root)) (index (read-one index-path))
         (base (truename (uiop:ensure-directory-pathname source-directory)))
         (target-base (merge-pathnames (format nil "raw/~A/" label) root))
         (source (merge-pathnames "report.lisp" base)) (data (read-one source))
         (before (files-in base)) (files nil))
    (when (> (arcdocdb.evidence:file-bytes source) +plain-limit+)
      (error "Il produttore deve compattare il suo report prima dell'aggiunta."))
    (when (probe-file target-base) (error "Gruppo esistente."))
    (unless (and (eq :original-command-archive (getf index :kind))
                 (eq :command-verification (getf data :kind))
                 (member (getf data :status) '(:ok :failed :source-changed)))
      (error "Archivio o comando incompleto."))
    (dolist (file before)
      (push (copy-verified file (merge-pathnames (enough-namestring file base) target-base) root) files))
    (unless (and (equal before (files-in base))
                 (equalp data (read-one (merge-pathnames "report.lisp" target-base))))
      (error "Sorgente/record diverso."))
    (setf (getf index :files) (append (getf index :files) (nreverse files))
          (getf index :processes)
          (append (getf index :processes)
                  (list (list :label label :record-path (enough-namestring
                                                        (merge-pathnames "report.lisp" target-base) root)
                              :command (getf data :command) :status (getf data :status)
                              :exit-code (getf data :exit-code) :source-consistency (getf data :source-consistency)))))
    (with-open-file (stream index-path :direction :output :if-exists :supersede)
      (let ((*print-readably* t)) (write index :stream stream :pretty t) (terpri stream)))
    (format t "Gruppo ~A aggiunto, ~D file.~%" label (length files))))

(require :sb-posix)
(let ((arguments (uiop:command-line-arguments)))
  (cond ((= 2 (length arguments)) (collect (first arguments) (second arguments)))
        ((and (= 4 (length arguments)) (equal "--append" (first arguments)))
         (append-group (second arguments) (third arguments) (fourth arguments)))
        (t (error "Uso: manifest destinazione oppure --append destinazione label source-directory."))))
