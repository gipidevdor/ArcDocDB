;;;; C4: sintesi della copertura HTML di quattro file esatti, senza eseguire prodotto.
;;; REQ: REQ-VAL-001 REQ-AFF-012
(require :asdf)
(load "tools/evidence-storage.lisp")
(defparameter *files* '("control/recovery-types.lisp" "control/recovery-read.lisp"
                       "control/recovery-publication.lisp" "recovery/manifest-transition.lisp"))
(defun demand (value detail) (unless value (error "coverage-summary COD-61: ~A" detail)))
(defun plain (text)
  (with-output-to-string (s)
    (loop with tag = nil for c across text do
      (cond ((char= c #\<) (setf tag t)) ((char= c #\>) (setf tag nil))
            ((not tag) (write-char c s))))))
(defun cells (row)
  (loop with start = 0 repeat (length row) for pos = (search "<td" row :start2 start)
        while pos collect
        (let* ((open (position #\> row :start pos)) (end (and open (search "</td>" row :start2 open))))
          (demand end "cellula HTML incompleta") (setf start (+ end 5))
          (string-trim '(#\Space #\Newline #\Return) (plain (subseq row (1+ open) end))))))
(defun rows (html)
  (demand (<= (length html) 1048576) "indice HTML oltre budget")
  (let ((directory nil) (result nil))
    (loop with start = 0 repeat (length html) for pos = (search "<tr" html :start2 start)
          while pos do
      (let ((end (search "</tr>" html :start2 pos)))
        (demand end "riga HTML incompleta")
        (let* ((row (subseq html pos end)) (columns (cells row)))
          (cond ((search "class='subheading'" row)
                 (demand (= 1 (length columns)) "directory HTML")
                 (let ((source (search "/src/" (first columns) :from-end t)))
                   (demand source "directory fuori dal prodotto")
                   (setf directory (subseq (first columns) (+ source 5)))))
                ((or (search "class='odd'" row) (search "class='even'" row))
                 (demand (and directory (= 7 (length columns))) "colonne HTML")
                 (let* ((name (concatenate 'string directory (first columns)))
                        (ec (parse-integer (second columns))) (et (parse-integer (third columns)))
                        (bc (parse-integer (fifth columns))) (bt (parse-integer (sixth columns))))
                   (demand (member name *files* :test #'string=) "file fuori scope")
                   (demand (and (<= 0 ec et) (<= 0 bc bt)) "contatori incoerenti")
                   (push (list :file name :expressions-covered ec :expressions-total et
                               :branches-covered bc :branches-total bt) result)))
                (t nil)))
        (setf start (+ end 5))))
    (demand (and (= 4 (length result))
                 (= 4 (length (remove-duplicates result :test #'equal :key (lambda (r) (getf r :file))))))
            "quattro file unici richiesti")
    (nreverse result)))
(defun rejected (thunk)
  (handler-case (progn (funcall thunk) nil) (error () t)))
(defun fixture (names)
  (with-output-to-string (s)
    (dolist (name names)
      (let ((slash (position #\/ name :from-end t)))
        (format s "<tr class='subheading'><td>/repo/src/~A</td></tr>"
                (subseq name 0 (1+ slash)))
        (format s "<tr class='odd'><td><a href='x.html'>~A</a></td><td>1</td><td>2</td><td>50</td><td>0</td><td>0</td><td>-</td></tr>"
                (subseq name (1+ slash)))))))
(defun self-test ()
  (demand (= 4 (length (rows (fixture *files*)))) "positivo")
  (demand (equal "a" (plain "<a>a</a>")) "tag")
  (demand (rejected (lambda () (rows ""))) "vuoto")
  (demand (rejected (lambda () (rows "<tr><td>"))) "HTML troncato")
  (demand (rejected (lambda () (rows (fixture (rest *files*))))) "mancante")
  (demand (rejected (lambda () (rows (fixture (cons (first *files*) *files*))))) "duplicato")
  (demand (rejected (lambda () (rows (fixture (cons "io/types.lisp" (rest *files*)))))) "fuori scope")
  (let ((html (fixture *files*)))
    (setf html (concatenate 'string "<tr class='odd'><td>a</td></tr>" html))
    (demand (rejected (lambda () (rows html))) "riga senza directory"))
  (write '(:schema-version 1 :kind :control-recovery-coverage-summary-self-test :status :ok :checks 8))
  (terpri))
(defun summarize (args)
  (demand (= 3 (length args)) "indice producer output")
  (let* ((html (first args)) (producer (second args))
         (record (arcdocdb.evidence:read-evidence producer))
         (data (list :schema-version 1 :kind :control-recovery-coverage-summary :status :ok
                     :rows (rows (uiop:read-file-string html :external-format :utf-8))
                     :inputs (loop for file in (list html producer) collect
                                   (list :file file :sha256 (arcdocdb.evidence:file-sha256 file)))
                     :limits '(:sb-cover-forms-and-branches :not-mcdc :local-frozen-source))))
    (demand (and (eq :command-verification (getf record :kind)) (eq :ok (getf record :status))
                 (eq :stable (getf record :source-consistency)) (eql 0 (getf record :exit-code))
                 (integerp (getf record :finished-at-universal-time))) "producer non concluso riuscito stabile")
    (with-open-file (out (third args) :direction :output :if-exists :error)
      (write data :stream out :pretty t) (terpri out))
    (write data :pretty t) (terpri)))
(let ((args (uiop:command-line-arguments)))
  (if (equal args '("--self-test")) (self-test) (summarize args)))
