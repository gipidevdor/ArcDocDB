;;;; C4: integra foglie esplicite e due sintesi, senza includere il digest del catalogo in se stesso.
(require :asdf)
(load "tools/evidence-storage.lisp")
(defparameter *base* "spikes/results/2026-10-14-control-transizione/")
(defparameter *leaves*
  '("README.md" "report.md" "revisione-autore.json" "revisione-indipendente.json"
    "revisione-benchmark.json" "riparazioni-benchmark.json" "collect.lisp" "collect-source.txt"
    "supervise.lisp" "supervise-source.txt" "coverage-summary.lisp" "coverage-summary-source.txt"
    "bench-summary.lisp" "bench-summary-source.txt" "benchmark-attempt-A-source.txt"
    "benchmark-attempt-B-source.txt" "benchmark-attempt-C-source.txt" "finish.lisp" "finish-source.txt"))
(defun local-name-p (s)
  (and (stringp s) (plusp (length s)) (not (find #\/ s)) (not (find #\\ s))
       (not (member s '("." "..") :test #'string=))))
(defun summary-valid-p (data kind)
  (and (eql 1 (getf data :schema-version)) (eq kind (getf data :kind)) (eq :ok (getf data :status))))
(defun save (data path)
  (with-open-file (s path :direction :output :if-exists :supersede)
    (write data :stream s :pretty t) (terpri s)))
(defun integrate ()
  (let* ((path (merge-pathnames "catalogo.lisp" *base*)) (catalog (arcdocdb.evidence:read-evidence path)))
    (unless (eq :evidence-catalog (getf catalog :kind)) (error "finish COD-61: catalogo mancante."))
    (dolist (leaf *leaves*)
      (unless (local-name-p leaf) (error "finish COD-61: foglia non locale."))
      (let* ((file (merge-pathnames leaf *base*))
             (entry (list :file leaf :bytes (arcdocdb.evidence:file-bytes file)
                          :sha256 (arcdocdb.evidence:file-sha256 file) :source (namestring (truename file)))))
        (setf (getf catalog :associated-raw-files)
              (cons entry (remove leaf (getf catalog :associated-raw-files)
                                  :test #'string= :key (lambda (r) (getf r :file)))))))
    (dolist (spec '(("coverage.lisp" :control-recovery-coverage-summary)
                   ("benchmark.lisp" :control-transition-derived-results)))
      (let* ((name (first spec)) (file (merge-pathnames name *base*))
             (data (arcdocdb.evidence:read-evidence file)))
        (unless (summary-valid-p data (second spec)) (error "finish COD-61: sintesi invalida ~A." name))
        (setf (getf catalog :entries)
              (cons (list :artifact name :kind (getf data :kind) :status :ok :scope :control-transizione
                          :sha256 (arcdocdb.evidence:file-sha256 file))
                    (remove name (getf catalog :entries) :test #'string= :key (lambda (r) (getf r :artifact)))))))
    (save catalog path)
    (write (list :schema-version 1 :kind :control-transition-publication-completion :status :ok
                 :records (length (getf catalog :entries)) :raw-files (length (getf catalog :associated-raw-files))
                 :catalog-sha256 (arcdocdb.evidence:file-sha256 path)
                 :limits '(:explicit-leaves :local-copy-integrity :no-self-certification))) (terpri)))
(defun self-test ()
  (unless (and (every #'local-name-p *leaves*)
               (= (length *leaves*) (length (remove-duplicates *leaves* :test #'string=)))
               (not (local-name-p "../a")) (not (local-name-p "a/b")) (not (local-name-p ".."))
               (not (local-name-p ""))
               (summary-valid-p '(:schema-version 1 :kind :fixture :status :ok) :fixture)
               (not (summary-valid-p '(:schema-version 1 :kind :other :status :ok) :fixture))
               (not (summary-valid-p '(:schema-version 1 :kind :fixture :status :failed) :fixture)))
    (error "finish COD-60: selezione/classificazione."))
  (write '(:schema-version 1 :kind :control-transition-publication-self-test :status :ok :checks 8)) (terpri))
(let ((args (uiop:command-line-arguments)))
  (cond ((equal args '("--self-test")) (self-test)) ((equal args '("--finish")) (integrate))
        (t (error "finish COD-61: --self-test o --finish."))))
