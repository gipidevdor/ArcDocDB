;;;; Raccolta di gruppi espliciti e completi, incluse cache FASL come soli dati.
;;;; Derivato batch da collect-R2 SHA256 638945f457ad4c5aaec72a26ef2b708249ed84ae2964ac157af9daf243bbb9b0.
;;;; Le copie normali sono accettate soltanto dopo snapshot SHA source/copy post-join.
(require :asdf)
(load "tools/evidence-storage.lisp")
(defpackage #:resolver.collection (:use #:cl))
(in-package #:resolver.collection)
(declaim (optimize (safety 3) (debug 3)))
(defparameter *collector-source* *load-truename*)
(defconstant +plain-limit+ 1048576)
(defconstant +gzip-limit+ 8388608)
(defconstant +expanded-limit+ 134217728)
(defvar *source-fingerprints* nil)
(defvar *defer-normal-sha* nil)
(defparameter *batch-library-sha256* "PENDING-LIBRARY-SHA256")

(defun read-one (path)
  (arcdocdb.evidence:read-evidence path))
(defun save-new (path data)
  (ensure-directories-exist path)
  (with-open-file (stream path :direction :output :if-exists :error)
    (let ((*print-readably* t)) (write data :stream stream :pretty t) (terpri stream))))
(defun native-fingerprint (path)
  (list :bytes (arcdocdb.evidence:file-bytes path) :sha256 (arcdocdb.evidence:file-sha256 path)))
(defun fingerprint (path)
  (let ((row (and *source-fingerprints*
                  (gethash (namestring (merge-pathnames path)) *source-fingerprints*))))
    (if row (list :bytes (getf row :bytes) :sha256 (getf row :sha256)) (native-fingerprint path))))
(defun batch-snapshot (paths)
  (uiop:symbol-call :preflight.batch.sha :snapshot paths))
(defun snapshot-table (rows)
  (uiop:symbol-call :preflight.batch.sha :table rows))
(defun row-fingerprint (row)
  (unless row (error "Path assente dallo snapshot batch."))
  (list :bytes (getf row :bytes) :sha256 (getf row :sha256)))
(defun leaf-p (value)
  (and (stringp value) (plusp (length value))
       (not (member value '("." "..") :test #'equal))
       (not (find-if (lambda (c) (or (find c "/\\:*?[]") (< (char-code c) 32))) value))))
(defun files-in (root)
  (let ((files nil) (count 0))
    (labels ((walk (directory depth)
               (when (> depth 64) (error "Directory troppo profonda."))
               (dolist (entry (directory (make-pathname :name :wild :type :wild :defaults directory)
                                         :resolve-symlinks nil))
                 (when (> (incf count) 100000) (error "Inventario troppo grande."))
                 (let* ((name (string-right-trim "/" (namestring entry)))
                        (mode (sb-posix:stat-mode (sb-posix:lstat name))))
                   (cond ((sb-posix:s-isdir mode) (walk entry (1+ depth)))
                         ((sb-posix:s-isreg mode) (push entry files))
                         (t (error "File speciale o symlink: ~A." entry)))))))
      (walk root 0))
    (sort files #'string< :key #'namestring)))

(defun copy-verified (source target root)
  (let* ((before (fingerprint source)) (size (getf before :bytes))
         (compressed (and (> size +plain-limit+) (not (equal "gz" (pathname-type source)))))
         (actual (if compressed (pathname (concatenate 'string (namestring target) ".gz")) target)))
    (unless (<= size +expanded-limit+) (error "Sorgente oltre budget: ~A." source))
    (when (probe-file actual) (error "Destinazione esistente: ~A." actual))
    (ensure-directories-exist actual)
    (if compressed
        (let ((decoded (pathname (concatenate 'string (namestring actual) ".decoded"))))
          (uiop:run-program (list "gzip" "-n" "-9" "-c" (namestring source)) :output actual)
          (unwind-protect
               (progn
                 (uiop:run-program (list "gzip" "-d" "-c" (namestring actual)) :output decoded)
                 (unless (equal before (native-fingerprint decoded)) (error "Espansione diversa: ~A." source)))
            (when (probe-file decoded) (delete-file decoded))))
        (progn (uiop:copy-file source actual)
               (unless (or *defer-normal-sha* (equal before (fingerprint actual)))
                 (error "Copia diversa: ~A." source))))
    (unless (or (and *defer-normal-sha* (not compressed))
                (equal before (native-fingerprint source))) (error "Sorgente cambiata: ~A." source))
    ;; For normal deferred files this is an expectation, not an accepted index entry.
    (let ((stored (if (and *defer-normal-sha* (not compressed)) before (native-fingerprint actual))))
      (unless (<= (getf stored :bytes) (if (equal "gz" (pathname-type actual)) +gzip-limit+ +plain-limit+))
        (error "Artefatto oltre budget: ~A." actual))
      (append (list :path (enough-namestring actual root) :source (namestring source)) stored
              (when compressed (list :codec :gzip :uncompressed-bytes size
                                     :uncompressed-sha256 (getf before :sha256)))))))

(defun parallel-map-four (function items)
  "Ordered results, at most four workers; join all and propagate every failure."
  (let* ((input (coerce items 'vector)) (size (length input)) (workers (min 4 size))
         (results (make-array size)) (failures (make-array size :initial-element nil))
         (threads nil) (launch-failure nil) (join-failures nil)
         (defaults *default-pathname-defaults*) (package *package*)
         (stdout *standard-output*) (stderr *error-output*)
         (source-fingerprints *source-fingerprints*) (defer-normal-sha *defer-normal-sha*))
    (unwind-protect
         (handler-case
             (dotimes (worker workers)
               (let ((start worker))
                 (push (sb-thread:make-thread
                        (lambda ()
                          (let ((*default-pathname-defaults* defaults) (*package* package)
                                (*standard-output* stdout) (*error-output* stderr)
                                (*source-fingerprints* source-fingerprints) (*defer-normal-sha* defer-normal-sha))
                            (loop for index from start below size by workers do
                              (handler-case (setf (aref results index) (funcall function (aref input index)))
                                (serious-condition (condition) (setf (aref failures index) condition))))))
                        :name (format nil "publication-worker-~D" start)) threads)))
           (serious-condition (condition) (setf launch-failure condition)))
      (sb-sys:without-interrupts
        (dolist (thread threads)
          (handler-case (sb-thread:join-thread thread)
            (serious-condition (condition) (push condition join-failures))))))
    (let ((errors (loop for condition across failures for index from 0 when condition
                        collect (list index (princ-to-string condition)))))
      (when (or launch-failure join-failures errors)
        (error "Parallel publication failed after joining all workers: launch=~A joins=~S entries=~S."
               launch-failure (mapcar #'princ-to-string join-failures) errors)))
    (coerce results 'list)))

(defun collect (manifest-path destination)
  (let* ((manifest (read-one manifest-path))
         (root (merge-pathnames (uiop:ensure-directory-pathname destination) (truename "./")))
         (manifest-before (native-fingerprint manifest-path))
         (collector-before (native-fingerprint *collector-source*))
         (library (merge-pathnames "batch-sha.lisp" *collector-source*))
         (library-before (native-fingerprint library))
         (files nil) (processes nil))
    (unless (equal *batch-library-sha256* (getf library-before :sha256))
      (error "Libreria batch diversa dalla fonte preregistrata."))
    (load library :verbose nil :print nil)
    (unless (and (eql 1 (getf manifest :schema-version)) (getf manifest :groups))
      (error "Manifesto invalido."))
    (when (probe-file root) (error "Archivio esistente: ~A." root))
    (dolist (group (getf manifest :groups))
      (destructuring-bind (label source-directory &optional process-p) group
        (unless (leaf-p label) (error "Nome gruppo invalido."))
        (let* ((base (truename (uiop:ensure-directory-pathname source-directory)))
               (target-base (merge-pathnames (format nil "raw/~A/" label) root))
               (before (files-in base))
               (source-before (batch-snapshot before))
               (*source-fingerprints* (snapshot-table source-before))
               (*defer-normal-sha* t))
          (unless (equal (mapcar #'namestring before)
                         (mapcar (lambda (row) (getf row :path)) source-before))
            (error "Snapshot sorgente non esatto: ~A." label))
          (when (and process-p (> (getf (fingerprint (merge-pathnames "report.lisp" base)) :bytes)
                                 +plain-limit+))
            (error "Il produttore deve compattare il suo report prima della raccolta."))
          (let* ((targets (mapcar (lambda (file) (merge-pathnames (enough-namestring file base) target-base)) before))
                 (owned (loop for file in before for target in targets append
                          (if (and (> (getf (fingerprint file) :bytes) +plain-limit+)
                                   (not (equal "gz" (pathname-type file))))
                              (list (concatenate 'string (namestring target) ".gz")
                                    (concatenate 'string (namestring target) ".gz.decoded"))
                              (list (namestring target))))))
            (unless (= (length owned) (length (remove-duplicates owned :test #'equal)))
              (error "Destinazioni/temp paralleli duplicati: ~A." label))
            (dolist (name owned) (ensure-directories-exist name))
            (let* ((pending (parallel-map-four (lambda (pair) (copy-verified (car pair) (cdr pair) root))
                                              (mapcar #'cons before targets)))
                   (copy-paths (sort (mapcar (lambda (entry) (merge-pathnames (getf entry :path) root)) pending)
                                     #'string< :key #'namestring))
                   (source-after (files-in base)))
              (unless (and (equal before source-after)
                           (equal copy-paths (files-in target-base)))
                (error "Insieme sorgente/copie cambiato dopo join: ~A." label))
              (let* ((copy-after (batch-snapshot copy-paths))
                     (source-after-hashes (batch-snapshot source-after))
                     (copy-table (snapshot-table copy-after)))
                (unless (and (equal source-before source-after-hashes)
                             (equal (mapcar #'namestring copy-paths)
                                    (mapcar (lambda (row) (getf row :path)) copy-after)))
                  (error "Sorgenti mutate o snapshot copie incompleto: ~A." label))
                (dolist (entry pending)
                  (let* ((source (gethash (getf entry :source) *source-fingerprints*))
                         (copy (gethash (namestring (merge-pathnames (getf entry :path) root)) copy-table))
                         (compressed (eq :gzip (getf entry :codec))))
                    (unless (and (equal (row-fingerprint copy) (list :bytes (getf entry :bytes)
                                                                   :sha256 (getf entry :sha256)))
                                 (equal (row-fingerprint source)
                                        (list :bytes (getf entry (if compressed :uncompressed-bytes :bytes))
                                              :sha256 (getf entry (if compressed :uncompressed-sha256 :sha256)))))
                      (error "Copia batch diversa dalle aspettative originarie: ~A." (getf entry :source)))))
                (unless (and (equal before (files-in base)) (equal copy-paths (files-in target-base)))
                  (error "Inventario sorgente/copie mutato durante hash batch: ~A." label))
                ;; Only fully verified groups can enter the future archive index.
                (setf files (nconc pending files)))))
          (when process-p
            (let* ((source (merge-pathnames "report.lisp" base))
                   (target (merge-pathnames "report.lisp" target-base)) (data (read-one source)))
              (unless (and (eql 1 (getf data :schema-version)) (eq :command-verification (getf data :kind))
                           (member (getf data :status) '(:ok :failed :source-changed))
                           (equalp data (read-one target)))
                (error "Record incompleto o diverso: ~A." source))
              (push (list :label label :record-path (enough-namestring target root)
                          :command (getf data :command) :status (getf data :status)
                          :exit-code (getf data :exit-code) :source-consistency (getf data :source-consistency))
                    processes))))))
    ;; Outside group bindings all original per-file checks remain active, as in append.
    (push (copy-verified (truename manifest-path) (merge-pathnames "collection-manifest.lisp" root) root) files)
    (push (copy-verified (truename *collector-source*) (merge-pathnames "collection-source.txt" root) root) files)
    (dolist (item (getf manifest :files))
      (destructuring-bind (label source) item
        (unless (leaf-p label) (error "Nome file invalido."))
        (push (copy-verified (truename source) (merge-pathnames label root) root) files)))
    (unless (and (equal manifest-before (native-fingerprint manifest-path))
                 (equal collector-before (native-fingerprint *collector-source*))
                 (equal library-before (native-fingerprint library)))
      (error "Manifesto, collector o libreria mutati prima dell'indice finale."))
    (let ((paths (mapcar (lambda (entry) (getf entry :path)) files)))
      (unless (= (length paths) (length (remove-duplicates paths :test #'equal))) (error "Path duplicato.")))
    (save-new (merge-pathnames "archive-index.lisp" root)
              (list :schema-version 1 :kind :original-command-archive
                    :metadata-policy :read-without-result-inference :processes (reverse processes)
                    :files (sort (copy-list files) #'string< :key (lambda (entry) (getf entry :path)))
                    :limits '(:complete-explicit-groups :raw-fasl-preserved-as-data
                              :source-and-copy-sha256 :lossless-gzip-for-large-raw :no-result-promotion)))
    (save-new (merge-pathnames "catalogo.lisp" root)
              (list :schema-version 1 :kind :evidence-catalog :pending nil :entries
                    (list (list :kind :original-command-archive :artifact "archive-index.lisp"
                                :artifact-schema-version 1 :artifact-kind :original-command-archive))))
    (format t "~D file conservati, ~D comandi.~%" (length files) (length processes))))

(defun append-group (destination label source-directory)
  "Aggiunge un gruppo completo; riscrive soltanto l'indice derivato, mai originali."
  (unless (leaf-p label) (error "Nome gruppo invalido."))
  (let* ((root (truename (uiop:ensure-directory-pathname destination)))
         (index-path (merge-pathnames "archive-index.lisp" root)) (index (read-one index-path))
         (base (truename (uiop:ensure-directory-pathname source-directory)))
         (target-base (merge-pathnames (format nil "raw/~A/" label) root))
         (source (merge-pathnames "report.lisp" base)) (data (read-one source))
         (before (files-in base)) (files nil))
    (when (> (arcdocdb.evidence:file-bytes source) +plain-limit+)
      (error "Il produttore deve compattare il suo report prima dell'aggiunta."))
    (when (probe-file target-base) (error "Gruppo esistente."))
    (unless (and (eq :original-command-archive (getf index :kind))
                 (eq :command-verification (getf data :kind))
                 (member (getf data :status) '(:ok :failed :source-changed)))
      (error "Archivio o comando incompleto."))
    (dolist (file before)
      (push (copy-verified file (merge-pathnames (enough-namestring file base) target-base) root) files))
    (unless (and (equal before (files-in base))
                 (equalp data (read-one (merge-pathnames "report.lisp" target-base))))
      (error "Sorgente/record diverso."))
    (setf (getf index :files) (append (getf index :files) (nreverse files))
          (getf index :processes)
          (append (getf index :processes)
                  (list (list :label label :record-path (enough-namestring
                                                        (merge-pathnames "report.lisp" target-base) root)
                              :command (getf data :command) :status (getf data :status)
                              :exit-code (getf data :exit-code) :source-consistency (getf data :source-consistency)))))
    (with-open-file (stream index-path :direction :output :if-exists :supersede)
      (let ((*print-readably* t)) (write index :stream stream :pretty t) (terpri stream)))
    (format t "Gruppo ~A aggiunto, ~D file.~%" label (length files))))

(defun directory-prefix-p (root path)
  (uiop:string-prefix-p (namestring (uiop:ensure-directory-pathname root)) (namestring path)))

(defun byte-inventory (root)
  "Inventario ordinato di path e bytes originali; non interpreta i file."
  (loop for file in (files-in root)
        collect (append (list :path (enough-namestring file root)) (fingerprint file))))

(defun size-inventory (root)
  "I file già archiviati sono censiti per path e dimensione, senza nuovo SHA.
L'audit finale controlla separatamente tutti i loro hash in due passaggi."
  (loop for file in (files-in root)
        collect (list :path (enough-namestring file root)
                      :bytes (arcdocdb.evidence:file-bytes file))))

(defun append-files (destination label source-directory)
  "Aggiunge solo bytes di un gruppo completo, senza inferire report o risultati.
Le copie parziali e gli indici preparatori restano disponibili in caso di errore."
  (unless (leaf-p label) (error "Nome gruppo invalido."))
  (let* ((root (truename (uiop:ensure-directory-pathname destination)))
         (base (truename (uiop:ensure-directory-pathname source-directory)))
         (index-path (merge-pathnames "archive-index.lisp" root))
         (index (read-one index-path))
         (prefix (format nil "raw/~A/" label))
         (target-base (merge-pathnames prefix root))
         (helper-base (uiop:pathname-directory-pathname (truename *collector-source*)))
         (backup (merge-pathnames (format nil "append-~A-index-before.lisp.txt" label) helper-base))
         (candidate (merge-pathnames (format nil "append-~A-index-next.lisp.txt" label) helper-base))
         (collector-before (fingerprint *collector-source*))
         (index-before (fingerprint index-path))
         (source-before nil) (archive-before nil) (expected-archive nil) (index-next nil) (files nil))
    (unless (and (eql 1 (getf index :schema-version))
                 (eq :original-command-archive (getf index :kind))
                 (eq :read-without-result-inference (getf index :metadata-policy))
                 (listp (getf index :files)) (getf index :files))
      (error "Indice archivio invalido."))
    (when (or (directory-prefix-p root base) (directory-prefix-p base root)
              (directory-prefix-p root helper-base) (directory-prefix-p base helper-base))
      (error "Archivio, sorgente e helper devono avere perimetri separati."))
    (when (or (probe-file target-base)
              (some (lambda (entry) (uiop:string-prefix-p prefix (getf entry :path)))
                    (getf index :files))
              (find label (getf index :processes) :key (lambda (entry) (getf entry :label)) :test #'equal))
      (error "Gruppo esistente: ~A." label))
    (when (or (probe-file backup) (probe-file candidate))
      (error "Prove append esistenti: ~A." label))
    (setf source-before (byte-inventory base) archive-before (size-inventory root))
    (unless source-before (error "Gruppo sorgente vuoto: ~A." base))
    ;; Conserva l'indice precedente fuori dai gruppi e dall'archivio.
    (uiop:copy-file index-path backup)
    (unless (equal index-before (fingerprint backup)) (error "Backup indice diverso."))
    (dolist (entry source-before)
      (let* ((relative (getf entry :path))
             (copy (copy-verified (merge-pathnames relative base)
                                  (merge-pathnames relative target-base) root))
             (compressed (eq :gzip (getf copy :codec))))
        (unless (and (= (getf entry :bytes)
                        (getf copy (if compressed :uncompressed-bytes :bytes)))
                     (equal (getf entry :sha256)
                            (getf copy (if compressed :uncompressed-sha256 :sha256))))
          (error "Copia diversa dall'inventario sorgente iniziale: ~A." relative))
        (push copy files)))
    (setf files (sort files #'string< :key (lambda (entry) (getf entry :path))))
    (unless (and (equal source-before (byte-inventory base))
                 (equal collector-before (fingerprint *collector-source*)))
      (error "Gruppo sorgente o collector mutato durante append."))
    (let* ((new-inventory (mapcar (lambda (entry)
                                   (list :path (getf entry :path) :bytes (getf entry :bytes))) files))
           (all-paths (append (mapcar (lambda (entry) (getf entry :path)) (getf index :files))
                              (mapcar (lambda (entry) (getf entry :path)) files))))
      (setf expected-archive (sort (append (copy-list archive-before) new-inventory)
                                  #'string< :key (lambda (entry) (getf entry :path))))
      (unless (= (length all-paths) (length (remove-duplicates all-paths :test #'equal)))
        (error "Path indice duplicato."))
      (unless (equal expected-archive (size-inventory root))
        (error "Archivio precedente o nuove copie mutati durante append.")))
    ;; Il gruppo è generico: :processes e tutti gli originali rimangono identici.
    (setf (getf index :files) (sort (append (copy-list (getf index :files)) files)
                                  #'string< :key (lambda (entry) (getf entry :path))))
    (save-new candidate index)
    (setf index-next (fingerprint candidate))
    (unless (and (equalp index (read-one candidate))
                 (equal index-before (fingerprint index-path))
                 (equal index-before (fingerprint backup))
                 (equal source-before (byte-inventory base))
                 (equal collector-before (fingerprint *collector-source*)))
      (error "Indice preparato o sorgenti mutati prima della pubblicazione."))
    ;; Rinomina atomica sullo stesso filesystem: un errore conserva il candidato.
    (sb-posix:rename (namestring candidate) (namestring index-path))
    (setf expected-archive
          (mapcar (lambda (entry)
                    (if (equal "archive-index.lisp" (getf entry :path))
                        (list :path "archive-index.lisp" :bytes (getf index-next :bytes)) entry)) expected-archive))
    (unless (and (equalp index (read-one index-path))
                 (equal index-next (fingerprint index-path))
                 (equal expected-archive (size-inventory root))
                 (equal source-before (byte-inventory base))
                 (equal collector-before (fingerprint *collector-source*))
                 (equal index-before (fingerprint backup)))
      (error "Indice pubblicato, copie o sorgenti diversi dopo append."))
    (format t "Gruppo raw ~A aggiunto, ~D file; nessuna inferenza di risultato.~%" label (length files))
    (format t "Limite append: file preesistenti controllati per path/dimensione; SHA completi nell'audit finale.~%")))

(require :sb-posix)
(let ((arguments (uiop:command-line-arguments)))
  (cond ((= 2 (length arguments)) (collect (first arguments) (second arguments)))
        ((and (= 4 (length arguments)) (equal "--append" (first arguments)))
         (append-group (second arguments) (third arguments) (fourth arguments)))
        ((and (= 4 (length arguments)) (equal "--append-files" (first arguments)))
         (append-files (second arguments) (third arguments) (fourth arguments)))
        (t (error "Uso: manifest destinazione oppure --append/--append-files destinazione label source-directory."))))
