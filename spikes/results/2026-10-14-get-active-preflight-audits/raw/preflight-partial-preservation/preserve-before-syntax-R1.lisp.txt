;;;; Preserva directory parziali come tar.gz opachi: nessun contenuto Lisp viene letto.
;;;; USTAR deterministico di soli file regolari/directory; estrazione stretta in staging separato.
(require :asdf)
(require :sb-posix)
(defparameter *helper-source* *load-truename*)
(defun demand (condition control &rest args)
  (unless condition (error (apply #'format nil control args))))
(defun path (base leaf) (merge-pathnames leaf base))
(defun emit (record file)
  (with-open-file (s file :direction :output :if-exists :error :if-does-not-exist :create)
    (with-standard-io-syntax (let ((*print-readably* t)) (write record :stream s :pretty t) (terpri s)))))
(defun options ()
  (let ((o nil) (args (uiop:command-line-arguments)))
    (demand (evenp (length args)) "Options key/value.")
    (loop for (key value) on args by #'cddr
          for k = (and (uiop:string-prefix-p "--" key) (intern (string-upcase (subseq key 2)) :keyword))
          do (demand (and (member k '(:workspace :commit :library :library-sha256 :destination :staging :manifest))
                          (not (getf o k))) "Unknown/duplicate option ~A." key)
             (setf (getf o k) value))
    (dolist (k '(:workspace :commit :library :library-sha256 :destination :staging :manifest))
      (demand (getf o k) "Missing ~A." k))
    (setf (getf o :workspace) (truename (uiop:ensure-directory-pathname (getf o :workspace))))
    (dolist (k '(:library :manifest :destination :staging))
      (setf (getf o k) (path (getf o :workspace) (getf o k))))
    (dolist (k '(:destination :staging)) (setf (getf o k) (uiop:ensure-directory-pathname (getf o k)))) o))
(defun git (o &rest args)
  (string-right-trim '(#\Newline #\Return)
    (uiop:run-program (append '("git") args) :directory (getf o :workspace) :output :string :error-output :string)))
(defun source-state (o)
  (let ((state (list :head (git o "rev-parse" "HEAD") :status (git o "status" "--porcelain=v1" "--untracked-files=no"))))
    (demand (and (equal (getf o :commit) (getf state :head)) (equal "" (getf state :status))) "Frozen Git identity changed.") state))
(defun sha (p) (uiop:symbol-call :arcdocdb.evidence :file-sha256 p))
(defun bytes (p) (uiop:symbol-call :arcdocdb.evidence :file-bytes p))
(defun safe-path-p (name directory-p)
  (and (stringp name) (plusp (length name)) (not (char= #\/ (char name 0)))
       (not (find #\\ name)) (notany (lambda (c) (< (char-code c) 32)) name)
       (let* ((body (if directory-p (string-right-trim "/" name) name))
              (parts (uiop:split-string body :separator '(#\/))))
         (and (plusp (length body))
              (every (lambda (p) (and (plusp (length p)) (not (member p '("." "..") :test #'equal)))) parts)
              (eq directory-p (char= #\/ (char name (1- (length name)))))))))
(defun tree-shape (base)
  (let ((files nil) (dirs nil) (n 0))
    (labels ((walk (dir depth)
               (demand (< depth 128) "Directory depth.")
               (dolist (p (directory (make-pathname :name :wild :type :wild :defaults dir) :resolve-symlinks nil))
                 (demand (< (incf n) 50000) "Directory entry budget.")
                 (let ((mode (sb-posix:stat-mode (sb-posix:lstat (string-right-trim "/" (namestring p)))))
                       (name (enough-namestring p base)))
                   (cond ((sb-posix:s-isdir mode)
                          (demand (safe-path-p name t) "Unsafe directory ~A." name)
                          (push name dirs) (walk (uiop:ensure-directory-pathname p) (1+ depth)))
                         ((sb-posix:s-isreg mode)
                          (demand (safe-path-p name nil) "Unsafe file ~A." name) (push name files))
                         (t (demand nil "Symlink/special source ~A." p)))))))
      (demand (sb-posix:s-isdir (sb-posix:stat-mode (sb-posix:lstat (string-right-trim "/" (namestring base)))))) "Source root not a plain directory.")
      (walk base 0))
    (list :directories (sort dirs #'string<) :files (sort files #'string<))))
(defun source-snapshot (base shape)
  (let ((rows (uiop:symbol-call :preflight.batch.sha :snapshot
                                (mapcar (lambda (f) (path base f)) (getf shape :files)))))
    (loop for r in rows collect (list :path (enough-namestring (getf r :path) base)
                                      :bytes (getf r :bytes) :sha256 (getf r :sha256)))))
(defun octets (s) (sb-ext:string-to-octets s :external-format :utf-8))
(defun header-text (header at width text)
  (let ((b (octets text))) (demand (<= (length b) width) "USTAR text field overflow.")
    (replace header b :start1 at)))
(defun header-octal (header at width value)
  (let ((s (format nil "~O" value)))
    (demand (< (length s) width) "USTAR integer overflow.")
    (loop for i from at below (+ at width -1) do (setf (aref header i) 48))
    (header-text header (+ at width -1 (- (length s))) (length s) s)))
(defun tar-header (name size directory-p)
  (demand (safe-path-p name directory-p) "Unsafe tar member.")
  (let ((header (make-array 512 :element-type '(unsigned-byte 8) :initial-element 0)) (leaf name) (prefix ""))
    (when (> (length (octets name)) 100)
      (let ((split (loop for p from (1- (length name)) downto 1
                         when (and (char= #\/ (char name p))
                                   (<= (length (octets (subseq name 0 p))) 155)
                                   (<= 1 (length (octets (subseq name (1+ p)))) 100)) return p)))
        (demand split "USTAR path exceeds prefix/name fields: ~A." name)
        (setf prefix (subseq name 0 split) leaf (subseq name (1+ split)))))
    (header-text header 0 100 leaf) (header-text header 345 155 prefix)
    (header-octal header 100 8 (if directory-p #o755 #o644))
    (header-octal header 108 8 0) (header-octal header 116 8 0)
    (header-octal header 124 12 size) (header-octal header 136 12 0)
    (loop for i from 148 below 156 do (setf (aref header i) 32))
    (setf (aref header 156) (char-code (if directory-p #\5 #\0)))
    (header-text header 257 6 (concatenate 'string "ustar" (string #\Null)))
    (header-text header 263 2 "00")
    (header-text header 148 6 (format nil "~6,'0O" (reduce #'+ header)))
    (setf (aref header 154) 0 (aref header 155) 32) header))
(defun pad-size (size) (mod (- size) 512))
(defun write-tar (source shape rows output)
  (let ((buffer (make-array 65536 :element-type '(unsigned-byte 8)))
        (zeros (make-array 1024 :element-type '(unsigned-byte 8) :initial-element 0)) (total 0))
    (with-open-file (out output :direction :output :element-type '(unsigned-byte 8) :if-exists :error)
      (dolist (name (getf shape :directories)) (write-sequence (tar-header name 0 t) out) (incf total 512))
      (loop for name in (getf shape :files) for row in rows
            for size = (getf row :bytes) do
        (demand (equal name (getf row :path)) "Snapshot/source order.")
        (write-sequence (tar-header name size nil) out)
        (with-open-file (in (path source name) :element-type '(unsigned-byte 8))
          (loop with left = size while (plusp left) for want = (min left (length buffer))
                for got = (read-sequence buffer in :end want)
                do (demand (= got want) "Source shortened during tar.")
                   (write-sequence buffer out :end got) (decf left got))
          (demand (eq :eof (read-byte in nil :eof)) "Source grew during tar."))
        (write-sequence zeros out :end (pad-size size)) (incf total (+ 512 size (pad-size size)))
        (demand (<= total 134216704) "Tar expansion exceeds budget."))
      (write-sequence zeros out) (incf total 1024)) total))
(defun field-string (header at width)
  (let ((end (or (position 0 header :start at :end (+ at width)) (+ at width))))
    (sb-ext:octets-to-string (subseq header at end) :external-format :utf-8)))
(defun field-octal (header at width)
  (let ((text (string-trim '(#\Space #\Null) (map 'string #'code-char (subseq header at (+ at width))))))
    (demand (and (plusp (length text)) (every (lambda (c) (digit-char-p c 8)) text)) "Malformed tar numeric field.")
    (parse-integer text :radix 8)))
(defun read-tar (input staging expected-shape expected-rows)
  (let ((header (make-array 512 :element-type '(unsigned-byte 8)))
        (buffer (make-array 65536 :element-type '(unsigned-byte 8)))
        (members nil) (files nil) (dirs nil) (total 0))
    (with-open-file (in input :element-type '(unsigned-byte 8))
      (loop
        (demand (= 512 (read-sequence header in)) "Truncated tar header.") (incf total 512)
        (when (every #'zerop header)
          (demand (and (= 512 (read-sequence header in)) (every #'zerop header)
                       (eq :eof (read-byte in nil :eof))) "Require exact two zero tar end blocks and EOF.")
          (incf total 512) (return))
        (let* ((leaf (field-string header 0 100)) (prefix (field-string header 345 155))
               (name (if (equal prefix "") leaf (concatenate 'string prefix "/" leaf)))
               (type (aref header 156)) (directory-p (= type 53)) (size (field-octal header 124 12))
               (sum (field-octal header 148 8)) (check (copy-seq header)))
          (loop for i from 148 below 156 do (setf (aref check i) 32))
          (demand (and (= sum (reduce #'+ check))
                       (equal "ustar" (field-string header 257 6)) (equal "00" (field-string header 263 2))
                       (member type '(48 53)) (safe-path-p name directory-p)
                       (equal "" (field-string header 157 100))
                       (not (member name members :test #'equal))
                       (<= (+ total size (pad-size size)) 134216704)) "Unsafe/malformed/duplicate tar header.")
          (push name members)
          (if directory-p
              (progn (demand (and (zerop size) (member name (getf expected-shape :directories) :test #'equal)) "Unexpected tar directory.")
                     (ensure-directories-exist (path staging (concatenate 'string name ".directory-marker"))) (push name dirs))
              (let ((row (find name expected-rows :test #'equal :key (lambda (r) (getf r :path))))
                    (dest (path staging name)))
                (demand (and row (= size (getf row :bytes))) "Unexpected tar member/size ~A." name)
                (ensure-directories-exist dest)
                (with-open-file (out dest :direction :output :element-type '(unsigned-byte 8) :if-exists :error)
                  (loop with left = size while (plusp left) for want = (min left (length buffer))
                        for got = (read-sequence buffer in :end want)
                        do (demand (= want got) "Truncated tar payload.") (write-sequence buffer out :end got) (decf left got)))
                (push name files)))
          (let* ((padding (pad-size size)) (got (read-sequence header in :end padding)))
            (demand (and (= padding got) (every #'zerop (subseq header 0 padding))) "Bad tar padding."))
          (incf total (+ size (pad-size size))))))
    (demand (and (equal (reverse members) (append (getf expected-shape :directories) (getf expected-shape :files)))
                 (equal (sort dirs #'string<) (getf expected-shape :directories))
                 (equal (sort files #'string<) (getf expected-shape :files))) "Exact tar member list differs.")
    (list :members (length members) :files (length files) :directories (length dirs) :expanded-bytes total)))
(defun same-bytes-p (a b)
  (let ((x (make-array 65536 :element-type '(unsigned-byte 8)))
        (y (make-array 65536 :element-type '(unsigned-byte 8))))
    (with-open-file (sx a :element-type '(unsigned-byte 8))
      (with-open-file (sy b :element-type '(unsigned-byte 8))
        (loop for nx = (read-sequence x sx) for ny = (read-sequence y sy)
              always (and (= nx ny) (equalp (subseq x 0 nx) (subseq y 0 ny)))
              until (zerop nx))))))
(defun archive-one (o name)
  (let* ((workspace (getf o :workspace))
         (source (path workspace (format nil "spikes/out/preflight-publication-partial-R2/~A/" name)))
         (stage (path (getf o :staging) (format nil "~A/" name)))
         (extract (path stage "extracted/")) (raw (path stage "original.tar")) (expanded (path stage "expanded.tar"))
         (payload (path (getf o :destination) (format nil "~A-partial-unqualified.tar.gz" name)))
         (shape-before (tree-shape source)) (before (source-snapshot source shape-before))
         (expanded-size nil) (members nil) (payload-before nil))
    (demand before "Empty partial source ~A." name)
    (emit (list :kind :partial-unqualified-source-inventory :name name :source (namestring source)
                :directories (getf shape-before :directories) :files before)
          (path (uiop:pathname-directory-pathname (getf o :manifest)) (format nil "~A-before.lisp" name)))
    (ensure-directories-exist (path stage ".stage-owned"))
    (setf expanded-size (write-tar source shape-before before raw))
    (demand (and (= expanded-size (bytes raw)) (<= expanded-size 134217728)) "Tar expanded size/budget.")
    (uiop:run-program (list "gzip" "-n" "-9" "-c" (namestring raw))
                      :output payload :error-output :string :if-output-exists :error)
    (demand (<= (bytes payload) 8388608) "Gzip payload exceeds 8MiB.")
    (setf payload-before (uiop:symbol-call :preflight.batch.sha :snapshot (list payload)))
    (uiop:run-program (list "gzip" "-d" "-c" (namestring payload))
                      :output expanded :error-output :string :if-output-exists :error)
    (demand (and (= (bytes expanded) expanded-size) (<= (bytes expanded) 134217728)
                 (same-bytes-p raw expanded)) "Gzip roundtrip/expansion differs.")
    (setf members (read-tar expanded extract shape-before before))
    (let* ((extracted-shape (tree-shape extract))
           (extracted (source-snapshot extract extracted-shape))
           (shape-after (tree-shape source)) (after (source-snapshot source shape-after))
           (payload-after (uiop:symbol-call :preflight.batch.sha :snapshot (list payload))))
      (demand (and (equal shape-before shape-after) (equal before after)
                   (equal shape-before extracted-shape) (equal before extracted)
                   (equal payload-before payload-after)) "Source/archive/member inventory changed.")
      (emit (list :kind :partial-unqualified-source-inventory :name name :source (namestring source)
                  :directories (getf shape-after :directories) :files after
                  :verified-extracted-files extracted :verified-tar-members members)
            (path (uiop:pathname-directory-pathname (getf o :manifest)) (format nil "~A-after.lisp" name)))
      (list :name name :qualification :partial-unqualified :reason :collector-stopped-for-batch-digest-replacement
            :source (namestring source) :payload (first payload-after) :expanded-bytes expanded-size
            :regular-files (length before) :directories (length (getf shape-before :directories))
            :source-bytes (reduce #'+ before :key (lambda (r) (getf r :bytes)))
            :canonical-tar-paths t :symlinks-or-special-files nil :tar-members members
            :source-before-after :identical :extracted-vs-source :identical :gzip-roundtrip :identical))))
(defun main ()
  (let* ((o (options)) (workspace (getf o :workspace))
         (reader (path workspace "tools/evidence-storage.lisp"))
         (library (getf o :library)) (destination (getf o :destination)) (stage (getf o :staging))
         (source-root (path workspace "spikes/out/preflight-publication-partial-R2/"))
         (manifest (getf o :manifest)) (state-before (source-state o))
         (helper-hash nil) (library-hash nil) (reader-hash nil) (archives nil))
    (demand (and (not (probe-file destination)) (not (probe-file stage)) (not (probe-file manifest))
                 (uiop:subpathp destination (uiop:pathname-directory-pathname *helper-source*))
                 (uiop:subpathp manifest (uiop:pathname-directory-pathname *helper-source*))
                 (uiop:subpathp stage (path workspace "spikes/out/"))
                 (not (uiop:subpathp stage (uiop:pathname-directory-pathname *helper-source*)))
                 (not (uiop:subpathp source-root destination)) (not (uiop:subpathp destination source-root))
                 (not (uiop:subpathp source-root stage)) (not (uiop:subpathp stage source-root))) "Fresh disjoint destinations required.")
    (demand (equal (git o "rev-parse" (format nil "~A:tools/evidence-storage.lisp" (getf o :commit)))
                   (git o "hash-object" "--" (namestring reader))) "Trusted reader not frozen Git blob.")
    (load reader :verbose nil :print nil)
    (setf helper-hash (sha *helper-source*) library-hash (sha library) reader-hash (sha reader))
    (demand (equal library-hash (getf o :library-sha256)) "Library SHA differs.")
    (load library :verbose nil :print nil)
    (ensure-directories-exist (path destination ".destination-owned"))
    (ensure-directories-exist (path stage ".stage-owned"))
    (format t "Opaque partial inputs; staging outside publication group: ~A~%" stage)
    (dolist (name '("mutation" "bootstrap")) (push (archive-one o name) archives))
    (setf archives (reverse archives))
    (demand (and (equal helper-hash (sha *helper-source*)) (equal library-hash (sha library))
                 (equal reader-hash (sha reader)) (equal state-before (source-state o))) "Helper/library/reader/Git changed.")
    (emit (list :schema-version 1 :kind :opaque-partial-collection-preservation :status :ok
                :qualification :partial-unqualified :reason :collector-stopped-for-batch-digest-replacement
                :commit (getf o :commit) :git-before state-before :git-after (source-state o)
                :helper (list :path (namestring *helper-source*) :sha256 helper-hash)
                :batch-library (list :path (namestring library) :sha256 library-hash)
                :trusted-reader (list :path (namestring reader) :sha256 reader-hash)
                :source-root (namestring source-root) :archives archives
                :staging (namestring stage) :staging-removal :after-successful-verification
                :limits '(:opaque-bytes-only :no-source-report-or-fasl-load :no-lisp-content-read
                          :canonical-ustar-regular-files-and-directories-only :gzip-per-payload-8mib
                          :expanded-tar-per-payload-128mib :non-atomic-two-pass-observation)) manifest)
    ;; L'unica rimozione riguarda le copie temporanee create da questa invocazione.
    (uiop:delete-directory-tree stage :validate t :if-does-not-exist :error)
    (demand (not (probe-file stage)) "Staging still exists after positive verification.")
    (emit (list :status :ok :staging (namestring stage) :removed t :originals-and-payloads-preserved t)
          (path (uiop:pathname-directory-pathname manifest) "staging-removal.lisp"))
    (format t "OK: ~D opaque archives; source/member byte inventories exact; staging removed.~%" (length archives))))
(handler-case (main)
  (serious-condition (condition)
    (format *error-output* "Partial preservation failed; staging is retained if created: ~A~%" condition)
    (sb-ext:exit :code 1)))
