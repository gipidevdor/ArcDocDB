;;;; Audit indipendente: archivio e report come dati, FASL soltanto come bytes.
(require :asdf)
(require :sb-posix)
(load "tools/evidence-storage.lisp")
(defpackage #:active.archive.audit (:use #:cl))
(in-package #:active.archive.audit)
(declaim (optimize (safety 3) (debug 3)))
(defconstant +expanded-limit+ 134217728)
(defvar *archive*)
(defparameter *auditor-source* *load-truename*)
(defparameter *audit-directory* (make-pathname :name nil :type nil :defaults *load-truename*))
(defun require-check (predicate control &rest arguments)
  (unless predicate (error (apply #'format nil control arguments)))
  t)
(defun read-data (path)
  (arcdocdb.evidence:read-evidence path :max-expanded-bytes +expanded-limit+))
(defun fingerprint (path)
  (list :bytes (arcdocdb.evidence:file-bytes path)
        :sha256 (arcdocdb.evidence:file-sha256 path)))
(defun leaf-p (value)
  (and (stringp value) (plusp (length value))
       (not (member value '("." "..") :test #'equal))
       (not (find-if (lambda (character)
                       (or (find character "/\\:*?[]") (< (char-code character) 32))) value))))
(defun relative-p (value)
  (and (stringp value) (plusp (length value))
       (not (char= #\/ (char value 0)))
       (every #'leaf-p (uiop:split-string value :separator "/"))))
(defun files-in (root)
  "Inventario indipendente, rifiuta symlink e file speciali a ogni livello."
  (let ((files nil) (count 0))
    (labels ((walk (directory depth)
               (require-check (<= depth 64) "Profondità eccessiva: ~A." directory)
               (dolist (entry (directory (make-pathname :name :wild :type :wild :defaults directory)
                                         :resolve-symlinks nil))
                 (require-check (<= (incf count) 100000) "Inventario oltre budget.")
                 (let ((mode (sb-posix:stat-mode
                              (sb-posix:lstat (string-right-trim "/" (namestring entry))))))
                   (cond ((sb-posix:s-isdir mode) (walk entry (1+ depth)))
                         ((sb-posix:s-isreg mode) (push entry files))
                         (t (error "Symlink/file speciale: ~A." entry)))))))
      (walk root 0))
    (sort files #'string< :key #'namestring)))
(defun relative-files (root)
  (mapcar (lambda (path) (enough-namestring path root)) (files-in root)))
(defun head ()
  (string-trim '(#\Space #\Tab #\Newline #\Return)
               (uiop:run-program '("git" "rev-parse" "HEAD") :output :string)))
(defun directory-prefix-p (root path)
  (uiop:string-prefix-p (namestring (uiop:ensure-directory-pathname root)) (namestring path)))
(defun owned-output-path (output archive groups)
  "Il parent deve già esistere: resolve prima di autorizzare qualsiasi write."
  (let* ((path (merge-pathnames output (truename "./")))
         (parent (truename (make-pathname :name nil :type nil :defaults path)))
         (actual (merge-pathnames (file-namestring path) parent))
         (temporary-directory (truename *audit-directory*)))
    (require-check (and (leaf-p (file-namestring actual))
                        (not (directory-prefix-p archive actual))
                        (not (directory-prefix-p archive temporary-directory)))
                   "Output/temp dentro archivio: ~A." output)
    (dolist (group groups)
      (let ((source (truename (uiop:ensure-directory-pathname (second group)))))
        (require-check (and (not (directory-prefix-p source actual))
                            (not (directory-prefix-p source temporary-directory)))
                       "Output/temp dentro gruppo sorgente: ~A." (first group))))
    (require-check (not (probe-file actual)) "Output già presente: ~A." actual)
    actual))
(defun sha-p (value)
  (and (stringp value) (= 64 (length value))
       (every (lambda (character) (or (char<= #\0 character #\9)
                                     (char<= #\a character #\f))) value)))
(defun expected-fingerprint (entry &optional expanded-p)
  (list :bytes (getf entry (if expanded-p :uncompressed-bytes :bytes))
        :sha256 (getf entry (if expanded-p :uncompressed-sha256 :sha256))))
(defun cleanup-process (process)
  (when process
    (unwind-protect
         (progn
           (when (member (sb-ext:process-status process) '(:running :stopped))
             (sb-ext:process-kill process sb-posix:sigkill))
           (sb-ext:process-wait process))
      (sb-ext:process-close process))))
(defun expanded-fingerprint (path expected)
  "gzip è espanso con limite prima di ogni write, mai interpretato come Lisp."
  (require-check (typep expected `(integer 0 ,+expanded-limit+)) "Dimensione gzip invalida.")
  (multiple-value-bind (fd name)
      (sb-posix:mkstemp (namestring (merge-pathnames "decoded-XXXXXX" *audit-directory*)))
    (let ((stream nil) (process nil) (temporary (pathname name)))
      (unwind-protect
           (progn
             (setf stream (sb-sys:make-fd-stream fd :input t :output t
                                               :element-type '(unsigned-byte 8) :auto-close t)
                   fd nil)
             (with-open-file (input path :direction :input :element-type '(unsigned-byte 8))
               (setf process (sb-ext:run-program "gzip" '("-dc") :search t :wait nil
                                                :input input :output :stream :error nil))
               (let ((buffer (make-array 65536 :element-type '(unsigned-byte 8))) (total 0))
                 (loop for count = (read-sequence buffer (sb-ext:process-output process)
                                                  :end (min 65536 (1+ (- expected total))))
                       until (zerop count) do
                   (require-check (<= (+ total count) expected) "gzip supera i byte attesi: ~A." path)
                   (write-sequence buffer stream :end count) (incf total count))
                 (require-check (= total expected) "gzip incompleto: ~A." path))
               (sb-ext:process-wait process)
               (require-check (and (eq :exited (sb-ext:process-status process))
                                   (eql 0 (sb-ext:process-exit-code process))) "gzip fallito: ~A." path))
             (finish-output stream)
             (fingerprint temporary))
        (unwind-protect (cleanup-process process)
          (unwind-protect
               (if stream (close stream) (when fd (sb-posix:close fd)))
            (when (probe-file temporary) (delete-file temporary))))))))

(defun gzip-size (path)
  "ISIZE del singolo gzip raccolto; l'espansione controlla davvero il limite."
  (with-open-file (stream path :direction :input :element-type '(unsigned-byte 8))
    (require-check (>= (file-length stream) 18) "gzip troppo corto: ~A." path)
    (file-position stream (- (file-length stream) 4))
    (loop for shift from 0 below 32 by 8 sum (ash (read-byte stream) shift))))
(defun validate-entry (entry)
  (let* ((relative (getf entry :path)) (source (getf entry :source))
         (codec (getf entry :codec)) (path (merge-pathnames relative *archive*))
         (gzip-p (equal "gz" (pathname-type path))) (expanded nil))
    (require-check (and (relative-p relative) (stringp source)
                        (typep (getf entry :bytes) '(integer 0 8388608))
                        (sha-p (getf entry :sha256)) (member codec '(nil :gzip)))
                   "Entry invalida: ~S." relative)
    (require-check (equal (expected-fingerprint entry) (fingerprint path))
                   "Copia/hash diversi: ~A." relative)
    (require-check (<= (getf entry :bytes) (if gzip-p 8388608 1048576)) "Budget copia: ~A." relative)
    (when codec
      (setf expanded (expanded-fingerprint path (getf entry :uncompressed-bytes)))
      (require-check (and (sha-p (getf entry :uncompressed-sha256))
                          gzip-p (equal (expected-fingerprint entry t) expanded))
                     "gzip espanso diverso: ~A." relative))
    (when (and gzip-p (null codec))
      (setf expanded (expanded-fingerprint path (gzip-size path))))
    (let ((available (probe-file source)))
      (when available
        (require-check (sb-posix:s-isreg (sb-posix:stat-mode (sb-posix:lstat source)))
                       "Sorgente speciale: ~A." source)
        (require-check (equal (expected-fingerprint entry (eq codec :gzip)) (fingerprint source))
                       "Sorgente diversa dalla copia: ~A." source)
        (when (and gzip-p (null codec))
          (require-check (equal expanded (expanded-fingerprint source (gzip-size source)))
                         "gzip sorgente/copia espansi diversi: ~A." relative)))
      (list :path relative :stored-bytes (getf entry :bytes) :codec codec
            :expanded-fingerprint expanded :source-available (not (null available)) :verified t))))
(defun stored-report (root relative)
  (require-check (relative-p relative) "Path report invalido: ~S." relative)
  (read-data (merge-pathnames relative root)))
(defun check-process (process contract)
  (let* ((relative (getf process :record-path)) (label (getf process :label))
         (data (stored-report *archive* relative))
         (expected (assoc label (getf contract :expected-processes) :test #'equal)))
    (require-check (and expected (eql 1 (getf data :schema-version))
                        (eq :command-verification (getf data :kind))
                        (eq (second expected) (getf data :status))
                        (eql (third expected) (getf data :exit-code))
                        (eq (fourth expected) (getf data :source-consistency)))
                   "Contratto comando diverso: ~A." label)
    (dolist (key '(:command :status :exit-code :source-consistency))
      (require-check (equal (getf process key) (getf data key)) "Metadati indice/report: ~A ~S." label key))
    (list :label label :status (getf data :status) :exit-code (getf data :exit-code)
          :source-consistency (getf data :source-consistency) :read-as-data t)))

(defun parallel-map-four (function items)
  "Ordered results, at most four workers; join all and propagate every failure."
  (let* ((input (coerce items 'vector)) (size (length input)) (workers (min 4 size))
         (results (make-array size)) (failures (make-array size :initial-element nil))
         (threads nil) (launch-failure nil) (join-failures nil)
         (defaults *default-pathname-defaults*) (package *package*)
         (stdout *standard-output*) (stderr *error-output*) (archive *archive*) (audit-directory *audit-directory*))
    (unwind-protect
         (handler-case
             (dotimes (worker workers)
               (let ((start worker))
                 (push (sb-thread:make-thread
                        (lambda ()
                          (let ((*default-pathname-defaults* defaults) (*package* package)
                                (*standard-output* stdout) (*error-output* stderr)
                                (*archive* archive) (*audit-directory* audit-directory))
                            (loop for index from start below size by workers do
                              (handler-case (setf (aref results index) (funcall function (aref input index)))
                                (serious-condition (condition) (setf (aref failures index) condition))))))
                        :name (format nil "publication-worker-~D" start)) threads)))
           (serious-condition (condition) (setf launch-failure condition)))
      (sb-sys:without-interrupts
        (dolist (thread threads)
          (handler-case (sb-thread:join-thread thread)
            (serious-condition (condition) (push condition join-failures))))))
    (let ((errors (loop for condition across failures for index from 0 when condition
                        collect (list index (princ-to-string condition)))))
      (when (or launch-failure join-failures errors)
        (error "Parallel publication failed after joining all workers: launch=~A joins=~S entries=~S."
               launch-failure (mapcar #'princ-to-string join-failures) errors)))
    (coerce results 'list)))

(defun metadata-storage ()
  "Only the two metadata leaves may authorize their own descriptor payload."
  (loop for leaf in '("archive-index.lisp" "catalogo.lisp")
        for path = (merge-pathnames leaf *archive*) collect
    (let* ((stored (fingerprint path))
           (raw (progn
                  (require-check (<= (getf stored :bytes) +expanded-limit+) "Metadata raw budget.")
                  (with-standard-io-syntax
                    (let ((*read-eval* nil) (eof (gensym)))
                      (with-open-file (stream path :direction :input)
                        (let* ((data (read stream nil eof)) (size (and (consp data) (list-length data))))
                          (require-check (and size (evenp size) (eq eof (read stream nil eof)))
                                         "Metadata requires one form and EOF.") data))))))
           (compressed (eq :compressed-evidence (getf raw :kind)))
           (payload (when compressed (getf raw :payload))) (payload-bytes nil))
      (when compressed
        (require-check (and (<= (getf stored :bytes) 1048576) (leaf-p payload)
                            (equal payload (concatenate 'string leaf ".gz"))) "Metadata descriptor payload leaf.")
        (read-data path) ; trusted reader validates exact descriptor shape, bytes, SHA and expansion.
        (setf payload-bytes (fingerprint (merge-pathnames payload *archive*)))
        (require-check (equal payload-bytes (list :bytes (getf raw :compressed-bytes)
                                                 :sha256 (string-downcase (getf raw :compressed-sha256))))
                       "Metadata payload differs from descriptor."))
      (append (list :path leaf :raw-fingerprint stored)
              (when compressed (list :descriptor raw :payload payload :payload-fingerprint payload-bytes))))))

(defun audit (manifest-path archive-path contract-path output)
  (let* ((*archive* (truename (uiop:ensure-directory-pathname archive-path)))
         (manifest (read-data manifest-path)) (contract (read-data contract-path))
         (manifest-before (fingerprint manifest-path)) (contract-before (fingerprint contract-path))
         (reader-before (fingerprint "tools/evidence-storage.lisp"))
         (auditor-before (fingerprint *auditor-source*))
         (catalog-path (merge-pathnames "catalogo.lisp" *archive*)) (catalog-before (fingerprint catalog-path))
         (output (owned-output-path output *archive* (getf manifest :groups)))
         (index-path (merge-pathnames "archive-index.lisp" *archive*))
         (index-before (fingerprint index-path)) (index (read-data index-path))
         (entries (getf index :files)) (paths (mapcar (lambda (entry) (getf entry :path)) entries))
         (initial-head (head)) (entry-results nil) (group-results nil) (process-results nil)
         (native-results nil) (report-records nil) (initial-inventory (relative-files *archive*))
         (metadata-before (metadata-storage)))
    (require-check (and (eql 1 (getf manifest :schema-version))
                        (eql 1 (getf contract :schema-version))
                        (eq :archive-audit-contract (getf contract :kind))
                        (equal initial-head (getf contract :source-commit))
                        (eql 1 (getf index :schema-version))
                        (eq :original-command-archive (getf index :kind))
                        (eq :read-without-result-inference (getf index :metadata-policy)))
                   "Schema/freeze dell'archivio invalido.")
    (require-check (= (length paths) (length (remove-duplicates paths :test #'equal))) "Entry duplicate.")
    (require-check (equal (sort (copy-list initial-inventory) #'string<)
                          (sort (append (copy-list paths) (list "archive-index.lisp" "catalogo.lisp")
                                        (remove nil (mapcar (lambda (row) (getf row :payload)) metadata-before))) #'string<))
                   "Inventario globale archivio diverso dall'indice.")
    (let ((catalog (read-data catalog-path)))
      (require-check (and (eql 1 (getf catalog :schema-version)) (eq :evidence-catalog (getf catalog :kind))
                          (null (getf catalog :pending))
                          (equal '("archive-index.lisp")
                                 (mapcar (lambda (entry) (getf entry :artifact)) (getf catalog :entries))))
                     "Catalogo invalido."))
    (setf entry-results (reverse (parallel-map-four #'validate-entry entries)))
    ;; Anche report figli/fixture non qualificati: vengono soltanto letti come
    ;; dati e censiti. Status e kind non sono promossi a risultati di prodotto.
    (dolist (entry entries)
      (when (and (null (getf entry :codec))
                 (equal "report.lisp" (file-namestring (pathname (getf entry :path)))))
        (let ((data (stored-report *archive* (getf entry :path))))
          (require-check (eql 1 (getf data :schema-version)) "Schema report figlio diverso: ~A." (getf entry :path))
          (push (list :path (getf entry :path) :kind (getf data :kind)
                      :status (getf data :status) :read-as-data t :result-inference nil) report-records))))
    (let ((labels (mapcar #'first (getf manifest :groups))))
      (require-check (and (= (length labels) (length (remove-duplicates labels :test #'equal)))
                          (equal (sort (copy-list labels) #'string<)
                                 (sort (copy-list (getf contract :expected-groups)) #'string<)))
                     "Gruppi espliciti diversi dal contratto."))
    (dolist (group (getf manifest :groups))
      (destructuring-bind (label source-directory &optional process-p) group
        (declare (ignore process-p))
        (require-check (leaf-p label) "Label invalida.")
        (let* ((base (truename (uiop:ensure-directory-pathname source-directory)))
               (source-before (relative-files base))
               (prefix (format nil "raw/~A/" label))
               (selected (remove-if-not (lambda (entry) (uiop:string-prefix-p prefix (getf entry :path))) entries))
               (source-expected (mapcar (lambda (entry)
                                         (enough-namestring (pathname (getf entry :source)) base)) selected))
               (stored (mapcar (lambda (entry) (subseq (getf entry :path) (length prefix))) selected))
               (target (merge-pathnames prefix *archive*))
               (fasl (count-if (lambda (name) (equal "fasl" (pathname-type (pathname name)))) source-before)))
          (dolist (entry selected)
            (let ((relative (enough-namestring (pathname (getf entry :source)) base)))
              (require-check (and (relative-p relative)
                                  (equal (getf entry :path)
                                         (concatenate 'string prefix relative
                                                      (if (eq :gzip (getf entry :codec)) ".gz" ""))))
                             "Mapping sorgente/copia diverso: ~A." (getf entry :path))))
          (require-check (equal (sort (copy-list source-before) #'string<) (sort source-expected #'string<))
                         "Gruppo sorgente incompleto: ~A." label)
          (require-check (equal (relative-files target) (sort stored #'string<))
                         "Gruppo copia incompleto: ~A." label)
          (when (member label (getf contract :required-fasl-groups) :test #'equal)
            (require-check (plusp fasl) "FASL raw assenti: ~A." label))
          (require-check (equal source-before (relative-files base)) "Inventario sorgente mutato: ~A." label)
          (push (list :label label :files (length source-before) :raw-fasl fasl :exact-inventories t) group-results))))
    (let ((processes (getf index :processes)))
      (require-check (equal (sort (mapcar (lambda (item) (getf item :label)) processes) #'string<)
                            (sort (mapcar #'first (getf contract :expected-processes)) #'string<))
                     "Inventario comandi diverso.")
      (dolist (process processes) (push (check-process process contract) process-results)))
    (dolist (native (getf contract :native-reports))
      (destructuring-bind (relative kind &optional status) native
        (let ((data (stored-report *archive* relative)))
          (require-check (and (eql 1 (getf data :schema-version)) (eq kind (getf data :kind))
                              (or (null status) (eq status (getf data :status))))
                         "Contratto report nativo diverso: ~A." relative)
          (push (list :path relative :kind kind :status (getf data :status)
                      :diagnostic (getf data :diagnostic) :read-as-data t) native-results))))
    ;; Secondo pass byte/SHA: stessi path non bastano a escludere una mutazione.
    (parallel-map-four (lambda (entry)
      (require-check (equal (expected-fingerprint entry)
                            (fingerprint (merge-pathnames (getf entry :path) *archive*)))
                     "Copia mutata durante audit: ~A." (getf entry :path))
      (let ((before (find (getf entry :path) entry-results :test #'equal
                          :key (lambda (item) (getf item :path))))
            (source (getf entry :source)))
        (require-check (eql (getf before :source-available) (not (null (probe-file source))))
                       "Disponibilità sorgente mutata: ~A." source)
        (when (getf before :source-available)
          (require-check (equal (expected-fingerprint entry (eq :gzip (getf entry :codec)))
                                (fingerprint source)) "Sorgente mutata durante audit: ~A." source)))) entries)
    (require-check (equal metadata-before (metadata-storage)) "Metadata descriptor/payload mutated during audit.")
    (require-check (and (equal initial-head (head)) (equal index-before (fingerprint index-path))
                        (equal catalog-before (fingerprint catalog-path))
                        (equal manifest-before (fingerprint manifest-path))
                        (equal contract-before (fingerprint contract-path))
                        (equal reader-before (fingerprint "tools/evidence-storage.lisp"))
                        (equal auditor-before (fingerprint *auditor-source*))
                        (equal initial-inventory (relative-files *archive*))) "Archivio/HEAD mutato durante audit.")
    (let ((report (list :schema-version 1 :kind :independent-active-archive-audit :status :ok
                        :source-commit initial-head :reader-sha256 (arcdocdb.evidence:file-sha256 "tools/evidence-storage.lisp")
                        :manifest-sha256 (arcdocdb.evidence:file-sha256 manifest-path)
                        :archive-index-sha256 (getf index-before :sha256) :metadata-storage metadata-before
                        :catalog-sha256 (getf catalog-before :sha256)
                        :auditor-sha256 (getf auditor-before :sha256)
                        :contract-sha256 (arcdocdb.evidence:file-sha256 contract-path)
                        :file-count (length entries) :stored-bytes (reduce #'+ entries :key (lambda (entry) (getf entry :bytes)))
                        :source-available-count (count-if (lambda (entry) (getf entry :source-available)) entry-results)
                        :groups (reverse group-results) :processes (reverse process-results)
                        :report-records (reverse report-records)
                        :native-reports (reverse native-results) :entries (reverse entry-results)
                        :limits '(:read-evidence-only :raw-fasl-as-bytes :complete-explicit-groups
                                  :no-result-promotion :no-performance-or-mcdc-claim))))
      (ensure-directories-exist output)
      (with-open-file (stream output :direction :output :if-exists :error)
        (let ((*print-readably* t)) (write report :stream stream :pretty t) (terpri stream)))
      (format t "STATUS OK: ~D entry, ~D gruppi, ~D comandi, ~D report nativi.~%"
              (length entries) (length group-results) (length process-results) (length native-results)))))

(let ((arguments (uiop:command-line-arguments)))
  (require-check (= 4 (length arguments)) "Uso: manifest archive audit-contract output.")
  (apply #'audit arguments))
