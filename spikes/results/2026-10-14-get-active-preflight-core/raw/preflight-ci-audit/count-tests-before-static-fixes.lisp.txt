;;;; Static Git-only test inventory. No systems, product or evidence are LOADed.
;;;; Reuses frozen-suite-summaries/source-token/tree/ASD walking from preflight-local-audit.
;;;; CLI: --workspace DIR --commit SHA. Output is a single data record.
(require :asdf)
(declaim (optimize (safety 3) (debug 3)))
(defparameter *helper-source* *load-truename*)
(defparameter *git-cache* (make-hash-table :test #'equal))
(defun demand (condition control &rest arguments)
  (unless condition (error (apply #'format nil control arguments))))

(defun git-output (workspace &rest arguments)
  (let ((key (cons (namestring workspace) arguments)))
    (or (gethash key *git-cache*)
        (setf (gethash key *git-cache*)
              (uiop:run-program (append '("git") arguments) :directory workspace
                                :output :string :error-output :string)))))

(defun line-output (workspace &rest arguments)
  (string-right-trim '(#\Newline #\Return) (apply #'git-output workspace arguments)))


(defun expected-source-tree (workspace commit)
  "Require every native source path and blob to equal the explicit Git tree."
  (let ((entries nil))
    (dolist (entry (uiop:split-string
                   (git-output workspace "ls-tree" "-r" "-z" "--full-tree" commit)
                   :separator '(#\Null)))
      (when (plusp (length entry))
        (let* ((tab (position #\Tab entry))
               (metadata (and tab (uiop:split-string (subseq entry 0 tab) :separator '(#\Space))))
               (path (and tab (subseq entry (1+ tab)))))
          (demand (and (= 3 (length metadata)) (string= "blob" (second metadata)) path)
                  "Unsupported Git tree entry ~S." entry)
          (unless (uiop:string-prefix-p "spikes/results/" path)
            (push (list :path path :git-blob (third metadata)) entries)))))
    (demand entries "Explicit expected source tree is empty.")
    (sort entries #'string< :key (lambda (entry) (getf entry :path)))))


(defun source-token (line prefix)
  "Extract only the current frozen source's simple token spelling; no Lisp reader."
  (demand (uiop:string-prefix-p prefix line) "Source prefix mismatch: ~S." line)
  (let* ((start (length prefix))
         (end (position-if (lambda (character) (find character " ()")) line :start start))
         (token (subseq line start (or end (length line)))))
    (demand (plusp (length token)) "Missing source token: ~S." line)
    (string-upcase token)))


(defun frozen-suite-summaries (options tree)
  "Derive every suite count/template and ASD execution order from explicit Git source text."
  (let ((counts (make-hash-table :test #'equal)) (templates (make-hash-table :test #'equal))
        (names (make-hash-table :test #'equal)) (order nil) (rows nil))
    (dolist (entry tree)
      (let ((path (getf entry :path)))
        (when (and (uiop:string-prefix-p "tests/" path) (uiop:string-suffix-p path ".lisp"))
          (let ((package nil))
            (dolist (raw (uiop:split-string
                          (git-output (getf options :workspace) "show"
                                      (format nil "~A:~A" (getf options :commit) path))
                          :separator '(#\Newline)))
              (let ((line (string-trim '(#\Space #\Tab #\Return) raw)))
                (when (uiop:string-prefix-p "(in-package #:" line)
                  (setf package (source-token line "(in-package #:")))
                (when (uiop:string-prefix-p "(deftest " raw)
                  (demand package "Test before in-package: ~A." path)
                  (let ((identity (list package (source-token raw "(deftest "))))
                    (unless (gethash identity names)
                      (setf (gethash identity names) t)
                      (incf (gethash package counts 0)))))
                (when (uiop:string-prefix-p "(format t \"~D test " line)
                  (let ((end (position #\" line :start 11)))
                    (demand (and package end (not (gethash package templates)))
                            "Missing/duplicate suite template: ~A." path)
                    (setf (gethash package templates) (subseq line 11 end))))))))))
    (dolist (line (uiop:split-string
                  (git-output (getf options :workspace) "show"
                              (format nil "~A:arcdocdb.asd" (getf options :commit)))
                  :separator '(#\Newline)))
      (let ((trimmed (string-trim '(#\Space #\Tab #\Return) line)))
        (when (uiop:string-prefix-p "(uiop:symbol-call '#:" trimmed)
          (push (source-token trimmed "(uiop:symbol-call '#:") order))))
    (setf order (nreverse order))
    (demand (and (equal "ARCDOCDB.TESTS" (first order))
                 (= (length order) (1+ (hash-table-count counts)))
                 (= (length order) (length (remove-duplicates order :test #'equal))))
            "Frozen ASD suite order/count is inconsistent.")
    (dolist (package (rest order))
      (let ((count (gethash package counts)) (template (gethash package templates)))
        (demand (and count template) "Missing frozen suite count/template for ~A." package)
        (push (list :package package :count count
                    :native-line (string-right-trim '(#\Newline #\Return) (format nil template count))) rows)))
    (setf rows (nreverse rows))
    (let ((total (reduce #'+ rows :key (lambda (row) (getf row :count)))))
      (when (getf options :expect-tests)
        (demand (= (getf options :expect-tests) total) "Frozen test count differs from --expect-tests.")))
    rows))


(defun frozen-test-source-order (options tree)
  "Adapt the reviewed ASD data walker; preserve serial test component order."
  (let* ((text (git-output (getf options :workspace) "show"
                           (format nil "~A:arcdocdb.asd" (getf options :commit))))
         (systems nil) (files nil))
    (demand (<= (length text) 1048576) "Frozen ASD source exceeds budget.")
    (with-standard-io-syntax
      (let ((*read-eval* nil) (*package* (find-package :asdf-user)) (eof (gensym "EOF")))
        (with-input-from-string (stream text)
          (loop for form = (read stream nil eof) until (eq eof form)
                do (when (and (consp form) (symbolp (first form))
                              (equal "DEFSYSTEM" (symbol-name (first form)))
                              (equal "arcdocdb/tests" (second form)))
                     (push form systems))))))
    (demand (= 1 (length systems)) "Require exactly one frozen tests ASDF system.")
    (labels ((leaf-p (name)
               (and (stringp name) (plusp (length name))
                    (not (find-if (lambda (c) (find c "/\\:*?[] .")) name))))
             (walk (components base depth)
               (demand (and components (listp components) (<= depth 12)) "Frozen ASDF components/depth.")
               (loop for component in components append
                 (progn
                   (demand (and (consp component) (leaf-p (second component))
                                (null (getf (cddr component) :pathname))
                                (null (getf (cddr component) :depends-on)))
                           "Unsupported frozen ASDF component ~S." component)
                   (ecase (first component)
                     (:file (list (concatenate 'string base (second component) ".lisp")))
                     (:module
                      (demand (eq t (getf (cddr component) :serial)) "Nonserial frozen test module.")
                      (walk (getf (cddr component) :components)
                            (concatenate 'string base (second component) "/") (1+ depth))))))))
      (let* ((system (first systems)) (base (getf (cddr system) :pathname)))
        (demand (and (equal "tests/" base) (eq t (getf (cddr system) :serial)))
                "Unsupported frozen tests pathname/serialization.")
        (setf files (walk (getf (cddr system) :components) base 0))))
    (demand (and files (<= (length files) 4096)
                 (= (length files) (length (remove-duplicates files :test #'equal)))
                 (every (lambda (path) (find path tree :test #'equal :key (lambda (entry) (getf entry :path)))) files))
            "Frozen ASDF files duplicate or absent from Git source tree.")
    files))

(defun ordered-suite-names (options files rows)
  "Literal deftest source tokens only; model the two existing registry macros."
  (let ((names (make-hash-table :test #'equal)) (policies (make-hash-table :test #'equal))
        (suite-files (make-hash-table :test #'equal)) (duplicates nil) (smoke nil))
    (dolist (file files)
      (let* ((text (git-output (getf options :workspace) "show" (format nil "~A:~A" (getf options :commit) file)))
             (package nil) (macro (search "(defmacro deftest " text))
             (end (and macro (search (format nil "~%(def") text :start2 (1+ macro))))
             (definition (and macro (subseq text macro end))))
        (demand (<= (length text) 2097152) "Frozen test source budget: ~A." file)
        (dolist (raw (uiop:split-string text :separator '(#\Newline)))
          (let ((line (string-trim '(#\Space #\Tab #\Return) raw)))
            (when (uiop:string-prefix-p "(in-package #:" line)
              (setf package (source-token line "(in-package #:"))
              (when definition
                (let ((policy (cond ((search "pushnew ',name " definition) :first-definition)
                                    ((and (search "acons ',name #',name" definition)
                                          (search "remove ',name " definition)) :last-definition))))
                  (demand (and policy (not (gethash package policies)))
                          "Unknown/duplicate registry macro: ~A in ~A." package file)
                  (setf (gethash package policies) policy))))
            (when (uiop:string-prefix-p "(deftest " raw)
              (demand (and package (gethash package policies)) "No supported registry before test: ~A." file)
              (pushnew file (gethash package suite-files) :test #'equal)
              (let* ((name (source-token raw "(deftest ")))
                     (previous (member name (gethash package names) :test #'equal)))
                (when previous (push (list :package package :name name :file file) duplicates))
                (cond ((eq :last-definition (gethash package policies))
                       (setf (gethash package names)
                             (append (remove name (gethash package names) :test #'equal) (list name))))
                      ((not previous)
                       (setf (gethash package names) (append (gethash package names) (list name)))))))
            (when (and (equal package "ARCDOCDB.TESTS") (uiop:string-prefix-p "(list (check \"" line))
              (push (subseq line 14 (position #\" line :start 14)) smoke))
            (when (and (equal package "ARCDOCDB.TESTS") (uiop:string-prefix-p "(check \"" line))
              (push (subseq line 8 (position #\" line :start 8)) smoke))))))
    (demand (= (hash-table-count names) (length rows)) "Registered named suites differ from all frozen source suites.")
    (dolist (row rows)
      (let ((package (getf row :package)))
        (demand (= (getf row :count) (length (gethash package names)))
                "Serial ASDF registry count differs from frozen source count: ~A." package)
        (setf (getf row :test-names) (gethash package names)
              (getf row :test-source-files) (reverse (gethash package suite-files))
              (getf row :duplicate-policy) (gethash package policies))))
    (values rows (nreverse duplicates) (nreverse smoke))))

(defun options ()
  (let ((args (uiop:command-line-arguments)) (workspace nil) (commit nil))
    (loop while args for key = (pop args) do
      (demand args "Missing option value.")
      (cond ((equal "--workspace" key) (demand (null workspace) "Duplicate workspace.") (setf workspace (pop args)))
            ((equal "--commit" key) (demand (null commit) "Duplicate commit.") (setf commit (pop args)))
            (t (demand nil "Unknown option: ~A." key))))
    (demand (and workspace commit) "Usage --workspace DIR --commit SHA.")
    (setf workspace (truename (uiop:ensure-directory-pathname workspace)))
    (let ((resolved (line-output workspace "rev-parse" "--verify" (concatenate 'string commit "^{commit}"))))
      (demand (and (= 40 (length resolved)) (every (lambda (c) (digit-char-p c 16)) resolved)) "Commit SHA shape.")
      (list :workspace workspace :commit resolved))))

(handler-case
    (let* ((options (options)) (workspace (getf options :workspace)) (commit (getf options :commit))
           (tree (expected-source-tree workspace commit))
           (rows (frozen-suite-summaries options tree))
           (files (frozen-test-source-order options tree)))
      (multiple-value-bind (named duplicates smoke) (ordered-suite-names options files rows)
        (let ((record (list :schema-version 1 :kind :frozen-git-test-inventory :status :ok
                            :workspace (namestring workspace) :commit commit
                            :git-tree (line-output workspace "rev-parse" (concatenate 'string commit "^{tree}"))
                            :module-test-total (reduce #'+ named :key (lambda (row) (getf row :count)))
                            :named-suite-count (length named) :smoke-check-names smoke :smoke-count (length smoke)
                            :suite-execution-order (mapcar (lambda (row) (getf row :package)) named)
                            :serial-test-source-files files :suites named :duplicate-definitions duplicates
                            :limits '(:static-literal-source-tokens :asd-read-eval-disabled
                                      :serial-asdf-components-only :two-existing-registry-macro-shapes
                                      :no-source-evaluation :no-systems-loaded :no-evidence-loaded :no-tests-executed))))
          (let ((*print-readably* t) (*print-length* nil) (*print-level* nil)) (write record :pretty t) (terpri)))))
  (error (condition)
    (let ((*print-readably* t))
      (write (list :schema-version 1 :kind :frozen-git-test-inventory :status :failed
                   :diagnostic (princ-to-string condition)) :pretty t) (terpri))
    (sb-ext:exit :code 1)))
