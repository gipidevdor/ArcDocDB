;;;; Independent byte audit. Only evidence-storage.lisp is LOADed, after Git anchoring.
;;;; No product/test/driver/FASL code is executed. Python companion hashes bytes only.

(defparameter *auditor-source* *load-truename*)
(define-condition ci-audit-error (error)
 ((message :initarg :message :reader audit-message))
 (:report (lambda (c s) (write-string (audit-message c) s))))
(defun demand (condition control &rest arguments)
  (unless condition
    (error 'ci-audit-error :message (apply #'format nil control arguments))))


(defun read-native (path)
  (uiop:symbol-call :arcdocdb.evidence :read-evidence path))


(defun native-sha256 (path)
  (uiop:symbol-call :arcdocdb.evidence :file-sha256 path))



(defun git-output (workspace &rest arguments)
  (uiop:run-program (append '("git") arguments) :directory workspace
                    :output :string :error-output :string))


(defun line-output (workspace &rest arguments)
  (string-right-trim '(#\Newline #\Return) (apply #'git-output workspace arguments)))


(defun expected-source-tree (workspace commit)
  "Require every native source path and blob to equal the explicit Git tree."
  (let ((entries nil))
    (dolist (entry (uiop:split-string
                   (git-output workspace "ls-tree" "-r" "-z" "--full-tree" commit)
                   :separator '(#\Null)))
      (when (plusp (length entry))
        (let* ((tab (position #\Tab entry))
               (metadata (and tab (uiop:split-string (subseq entry 0 tab) :separator '(#\Space))))
               (path (and tab (subseq entry (1+ tab)))))
          (demand (and (= 3 (length metadata)) (string= "blob" (second metadata))
                           (member (first metadata) '("100644" "100755") :test #'equal) path)
                  "Unsupported Git tree entry ~S." entry)
          (unless (uiop:string-prefix-p "spikes/results/" path)
            (push (list :path path :git-blob (third metadata)) entries)))))
    (demand entries "Explicit expected source tree is empty.")
    (sort entries #'string< :key (lambda (entry) (getf entry :path)))))


(defun source-token (line prefix)
  "Extract only the current frozen source's simple token spelling; no Lisp reader."
  (demand (uiop:string-prefix-p prefix line) "Source prefix mismatch: ~S." line)
  (let* ((start (length prefix))
         (end (position-if (lambda (character) (find character " ()")) line :start start))
         (token (subseq line start (or end (length line)))))
    (demand (plusp (length token)) "Missing source token: ~S." line)
    (string-upcase token)))


(defun frozen-asdf-source-files (options tree)
  "Read frozen ASD as data, never evaluate/load it; derive registered source files."
  (let* ((text (git-output (getf options :workspace) "show"
                           (format nil "~A:arcdocdb.asd" (getf options :commit))))
         (systems nil) (files nil))
    (demand (<= (length text) 1048576) "Frozen ASD source exceeds budget.")
    (with-standard-io-syntax
      (let ((*read-eval* nil) (*package* (find-package :asdf-user)) (eof (gensym "EOF")))
        (with-input-from-string (stream text)
          (loop for form = (read stream nil eof) until (eq eof form)
                do (when (and (consp form) (symbolp (first form))
                              (equal "DEFSYSTEM" (symbol-name (first form)))
                              (member (second form) '("arcdocdb" "arcdocdb/tests") :test #'equal))
                     (push form systems))))))
    (demand (and (= 2 (length systems))
                 (= 2 (length (remove-duplicates systems :key #'second :test #'equal))))
            "Require exactly the frozen product and test ASDF systems.")
    (labels ((leaf-p (name)
               (and (stringp name) (plusp (length name))
                    (not (find-if (lambda (c) (find c "/\\:*?[] .")) name))))
             (walk (components base)
               (demand (and components (listp components)) "Missing frozen ASDF components.")
               (loop for component in components append
                 (progn
                   (demand (and (consp component) (leaf-p (second component))
                                (null (getf (cddr component) :pathname)))
                           "Unsupported frozen ASDF component ~S." component)
                   (ecase (first component)
                     (:file (list (concatenate 'string base (second component) ".lisp")))
                     (:module (walk (getf (cddr component) :components)
                                    (concatenate 'string base (second component) "/"))))))))
      (dolist (system systems)
        (let ((base (getf (cddr system) :pathname)))
          (demand (equal base (if (equal "arcdocdb" (second system)) "src/" "tests/"))
                  "Unsupported frozen ASDF system pathname.")
          (setf files (append files (walk (getf (cddr system) :components) base))))))
    (demand (and files (= (length files) (length (remove-duplicates files :test #'equal)))
                 (every (lambda (path)
                          (find path tree :test #'equal :key (lambda (entry) (getf entry :path)))) files))
            "Frozen ASDF files duplicate or differ from the complete Git source tree.")
    (sort files #'string<)))


(defun audit-wrapper (path options expected-tree &optional (status :ok))
  (let* ((record (read-native path)) (environment (getf record :environment))
         (before (getf record :source-blobs-before)) (after (getf record :source-blobs-after))
         (sorted (sort (copy-list before) #'string< :key (lambda (entry) (getf entry :path)))))
    (demand (and (eql 1 (getf record :schema-version)) (eq :command-verification (getf record :kind))
                 (eq status (getf record :status)) (eq :stable (getf record :source-consistency))
                 (equal before after) (equal sorted expected-tree)
                 (equal (getf options :commit) (getf environment :commit))
                 (stringp (getf environment :working-tree))
                 (zerop (length (getf environment :working-tree)))
                 (equal (namestring (getf options :workspace)) (getf environment :cwd))
                 (typep (getf record :started-at-universal-time) '(integer 1))
                 (typep (getf record :finished-at-universal-time) '(integer 1))
                 (<= (getf record :started-at-universal-time) (getf record :finished-at-universal-time))
                 (stringp (getf record :stdout)) (stringp (getf record :stderr))
                 (if (eq status :ok) (eql 0 (getf record :exit-code))
                     (and (integerp (getf record :exit-code)) (plusp (getf record :exit-code)))))
            "Wrapper native identity, finality, status or full source tree differs: ~A." path)
    record))


(defun read-data-string (text)
  (demand (<= (length text) 134217728) "Data string budget.")
  (with-standard-io-syntax
    (let ((*read-eval* nil) (*package* (find-package :cl-user)) (eof (gensym)))
      (with-input-from-string (s text)
        (let ((value (read s nil eof)))

          value)))))

(defun options ()
  (let ((o nil) (args (uiop:command-line-arguments)))
    (demand (evenp (length args)) "Options must be key/value pairs.")
    (loop for (k v) on args by #'cddr
          for key = (and (uiop:string-prefix-p "--" k) (intern (string-upcase (subseq k 2)) :keyword))
          do (demand (member key '(:workspace :commit :preflight-dir :preflight-command :bootstrap-dir
                                  :bootstrap-command :unknown-command :duplicate-command
                                  :preflight-selftest-command :bootstrap-selftest-command :output)) "Unknown option ~A." k)
             (demand (not (getf o key)) "Duplicate option ~A." k) (setf (getf o key) v))
    (dolist (key '(:workspace :commit :preflight-dir :preflight-command :bootstrap-dir :bootstrap-command
                  :unknown-command :duplicate-command :preflight-selftest-command :bootstrap-selftest-command :output))
      (demand (getf o key) "Missing ~A." key))
    (setf (getf o :workspace) (truename (uiop:ensure-directory-pathname (getf o :workspace))))
    (dolist (key '(:preflight-dir :bootstrap-dir))
      (setf (getf o key) (truename (uiop:ensure-directory-pathname (path (getf o :workspace) (getf o key))))))
    (dolist (key '(:preflight-command :bootstrap-command :unknown-command :duplicate-command
                  :preflight-selftest-command :bootstrap-selftest-command))
      (setf (getf o key) (truename (path (getf o :workspace) (getf o key)))))
    (setf (getf o :output) (path (getf o :workspace) (getf o :output))) o))

(defun walk-files (directory)
  "Inventory all regular files, including hidden files; reject symlinks/special files."
  (let ((files nil) (count 0))
    (labels ((walk (base depth)
               (demand (<= depth 128) "Artifact directory depth exceeds limit.")
               (dolist (path (directory (make-pathname :name :wild :type :wild :defaults base)
                                        :resolve-symlinks nil))
                 (demand (<= (incf count) 100000) "Artifact inventory exceeds entry limit.")
                 (let ((mode (sb-posix:stat-mode (sb-posix:lstat (string-right-trim "/" (namestring path))))))
                   (cond ((sb-posix:s-isdir mode) (walk (uiop:ensure-directory-pathname path) (1+ depth)))
                         ((sb-posix:s-isreg mode) (push path files))
                         (t (demand nil "Artifact contains a symlink or special file: ~A." path)))))))
      (walk directory 0))
    (sort files #'string< :key #'namestring)))

(defparameter +sha-batch-code+ "import os,sys,stat,hashlib
for name in sys.argv[1:]:
 fd=os.open(name,os.O_RDONLY|os.O_NOFOLLOW|os.O_NONBLOCK)
 try:
  before=os.fstat(fd)
  if not stat.S_ISREG(before.st_mode) or not 0<=before.st_size<=16777216: raise ValueError('file shape/size')
  digest=hashlib.sha256(); count=0
  for i in range(1+before.st_size//65536):
   chunk=os.read(fd,65536); digest.update(chunk); count+=len(chunk)
  after=os.fstat(fd)
  identity=lambda s:(s.st_dev,s.st_ino,s.st_size,s.st_mtime_ns,s.st_ctime_ns)
  if count!=before.st_size or identity(before)!=identity(after): raise ValueError('source drift')
  print(digest.hexdigest())
 finally: os.close(fd)")
(defun snapshot (paths)
  (let* ((files (sort (remove-duplicates
                       (loop for p in paths append
                         (if (uiop:directory-pathname-p p) (walk-files p) (list p)))
                       :test #'equal :key #'namestring) #'string< :key #'namestring))
         (rows nil))
    (demand (<= (length files) 50000) "Snapshot file budget.")
    (loop for tail on files by (lambda (v) (nthcdr (min 512 (length v)) v))
          for batch = (subseq tail 0 (min 512 (length tail)))
          do (let* ((out (uiop:run-program
                         (append (list "python3" "-c" +sha-batch-code+) (mapcar #'namestring batch))
                         :output :string :error-output :string))
                    (hashes (remove "" (uiop:split-string out :separator '(#\Newline)) :test #'equal)))
               (demand (and (= (length hashes) (length batch))
                            (every (lambda (h) (and (= 64 (length h)) (every (lambda (c) (digit-char-p c 16)) h))) hashes))
                       "Batch SHA256 cardinality/format.")
               (loop for p in batch for h in hashes
                     do (push (list :path (namestring p)
                                    :bytes (uiop:symbol-call :arcdocdb.evidence :file-bytes p) :sha256 h) rows))))
    (reverse rows)))

(defun lookup-byte (inventory p)
  (or (find (namestring p) inventory :test #'equal :key (lambda (e) (getf e :path)))
      (error 'ci-audit-error :message (format nil "Uninventoried file ~A." p))))
(defun file-sha (inventory p) (getf (lookup-byte inventory p) :sha256))
(defun manifest-at (inventory files root)
  (loop for f in files collect (list :file f :sha256 (file-sha inventory (path root f)))))
(defun unique-replacement (source before after)
  (let ((n (search before source)))
    (demand (and n (not (search before source :start2 (1+ n)))) "Mutation anchor missing/nonunique.")
    (concatenate 'string (subseq source 0 n) after (subseq source (+ n (length before))))))
(defun expected-patch (file source before after)
  ;; Derive changed line block and hunk coordinates independently from frozen bytes.
  (let* ((n (search before source))
         (a (1+ (or (position #\Newline source :end n :from-end t) -1)))
         (b (or (position #\Newline source :start (+ n (length before))) (length source)))
         (old (uiop:split-string (subseq source a b) :separator '(#\Newline)))
         (new (uiop:split-string (concatenate 'string (subseq source a n) after
                                            (subseq source (+ n (length before)) b)) :separator '(#\Newline)))
         (line (1+ (count #\Newline source :end a))))
    (with-output-to-string (s)
      (format s "--- a/~A~%+++ b/~A~%@@ -~D,~D +~D,~D @@~%" file file line (length old) line (length new))
      (dolist (v old) (format s "-~A~%" v)) (dolist (v new) (format s "+~A~%" v)))))
(defun log-check (inventory evidence)
  (let ((notes 0) (bytes 0))
    (dolist (leaf '("stdout.log" "stderr.log"))
      (let* ((p (path evidence leaf)) (text (text-file p)))
        (incf bytes (getf (lookup-byte inventory p) :bytes))
        (dolist (line (uiop:split-string text :separator '(#\Newline)))
          (demand (not (or (search "WARNING:" line) (search "STYLE-WARNING:" line))) "Compiler warning in ~A." p)
          (when (search "note:" (string-downcase line)) (incf notes)))))
    (list :log-bytes bytes :note-lines notes :warnings 0)))
(defun audit-child (o campaign parent entry mutant preflight tests files compiled inventory canonical)

         (delta (loop for a in canonical for b in copied unless (equal a b) collect (getf b :file)))
         (config-sha (file-sha inventory (path evidence "config.lisp")))

                  (namestring (make-pathname :type "fasl" :defaults (path (path copy "fasl/") f))))))
    (check-equal entry result "embedded / separately closed result")
    (check-equal (getf result :child-report) child "embedded / separate child")
    (check-equal (getf result :name) name "child name")
    (check-equal (getf result :file) (getf mutant :file) "child mutated file")

    (check-equal (getf result :expected-failure) (getf mutant :expected-failure) "preregistered reason")

                 (eq (if mutant :detected :baseline-passed) (getf result :result))
                 (getf result :child-launched) (getf result :child-reaped)
                 (getf result :producer-closed) (getf result :artifacts-verified)
                 (null (getf result :timed-out)) (null (getf result :signal))
                 (null (getf result :diagnostic)) (eq :reaped (getf result :cleanup))
                 (eql (if mutant 1 0) (getf result :exit-code))
                 (<= 0 (getf result :deadline-seconds) 120)
                 (plusp (getf result :deadline-seconds))) "Child status/reap/deadline ~A." name)
    (check-equal (getf result :copy-directory) (namestring copy) "copy location")
    (check-equal (getf result :evidence-directory) (namestring evidence) "evidence location")
    (check-equal (getf result :producer-record) (namestring (path campaign "report.lisp")) "producer identity")
    (check-equal (getf result :argv)
                 (list "sbcl" "--noinform" "--no-userinit" "--no-sysinit" "--disable-debugger"
                       "--script" (namestring (path evidence "driver.lisp")) "--child"
                       (namestring (path evidence "config.lisp"))) "child argv")

      (check-equal (getf r :expected-tests) expected "executed selection")
      (check-equal (getf r :source-manifest-before) copied "copy before manifest")
      (check-equal (getf r :config-sha256) config-sha "config initial bytes")
      (check-equal (getf r :config-sha256-after) config-sha "config final bytes"))
    (check-equal (getf result :source-manifest-after) copied "parent copy after manifest")
    (check-equal (getf child :source-manifest) copied "child copy final manifest")
    (check-equal delta (when mutant (list (getf mutant :file))) "one changed source")
    (check-equal (getf result :changed-paths) delta "result changed paths")
    (demand (and (eql 1 (getf config :schema-version)) (eql 1 (getf child :schema-version))
                 (eq (if preflight :get-active-preflight-mutation-config :get-active-init-mutation-config) (getf config :kind))
                 (eq (if preflight :get-active-preflight-mutation-child :get-active-init-mutation-child) (getf child :kind))) "Config/child schema.")
    (dolist (pair (list (list :files files) (list :baseline-manifest canonical) (list :manifest copied)
                       (list :changed-paths delta) (list :mutation mutant) (list :expected-tests expected)
                       (list :test-suites (suite-selection preflight)) (list :compiled-source-files compiled)
                       (list :child-report (namestring (path evidence "child-report.lisp")))))
      (check-equal (getf config (first pair)) (second pair) (first pair)))
    (check-equal (getf child :registered-tests) tests "whole registry")
    (when preflight
      (dolist (r (list result child config)) (check-equal (getf r :suite-tests) tests "whole ACTIVE suite"))
      (dolist (r (list result child)) (check-equal (getf r :executed-tests) expected "actual ACTIVE execution"))
      (check-equal (getf config :profile) :preflight "config profile"))
    (demand (and (getf child :compiled) (getf child :suite-compiled)
                 (<= (getf parent :started-at-universal-time) (getf child :finished-at-universal-time)
                     (getf parent :finished-at-universal-time))) "Compiled/final child time.")
    (check-equal (getf child :loaded-source-files) compiled "all ASDF source definitions")
    (demand (= (length compiled) (length (getf child :compiled-files))) "Compiled entry count.")
    (check-equal (sort (mapcar (lambda (e) (getf e :file)) (getf child :compiled-files)) #'string<) compiled "compiled source set")
    (dolist (c (getf child :compiled-files))
      (let* ((f (getf c :file)) (fasl (make-pathname :type "fasl" :defaults (path (path copy "fasl/") f))))
        (check-equal (getf c :fasl) (namestring fasl) "private FASL mapping")
        (check-equal (getf c :source-sha256) (file-sha inventory (path copy f)) "compiled source bytes")
        (check-equal (getf c :fasl-sha256) (file-sha inventory fasl) "FASL bytes")))
    (check-equal (sort (loop for b in inventory for p = (getf b :path)
                             when (uiop:string-prefix-p (namestring copy) p) collect p) #'string<)
                 (sort (append (mapcar (lambda (f) (namestring (path copy f))) files) fasls) #'string<)
                 "exact isolated copy and private FASL inventory")
    (dolist (pair '(("driver.lisp" "tools/get-active-init-mutation.lisp")
                    ("evidence-storage.lisp" "tools/evidence-storage.lisp")))
      (check-equal (file-sha inventory (path evidence (first pair)))
                   (file-sha inventory (path (getf o :workspace) (second pair))) "driver/reader frozen bytes"))
    (when mutant
      (let* ((file (getf mutant :file)) (original (source-text o file))
             (changed (unique-replacement original (getf mutant :before) (getf mutant :after))))
        (check-equal (text-file (path copy file)) changed "single frozen source substitution")
        (check-equal (text-file (path evidence "mutated-source.lisp")) changed "retained mutation source")
        (check-equal (text-file (path evidence "mutation.patch"))
                     (expected-patch file original (getf mutant :before) (getf mutant :after)) "exact preregistered patch")))
    (if mutant
        (let ((target (getf mutant :test)))
          (demand (member target tests :test #'equal) "Target not passed in baseline.")
          (check-equal (getf child :status) :failed "mutant status")
          (check-equal (getf child :phase) :tests "mutant failed during test")
          (check-equal (getf child :failed-test) target "actual failed target")
          (check-equal (getf child :passed-tests) (subseq expected 0 (position target expected :test #'equal)) "passed prefix")
          (if (and preflight (getf mutant :allowed-product-condition))
              (progn
                (check-equal (getf child :failure-kind) :product-condition "specific product failure")
                (check-equal (list (getf child :condition-package) (getf child :condition-name) (getf child :condition-reason))
                             '("ARCDOCDB.CONDITIONS" "RESOURCE-EXHAUSTED" :active-batch-record-budget) "exact allowed condition"))
              (if preflight
                  (check-equal (getf child :failure-kind) :assertion "assertion detection")
                  (demand (member (getf child :failure-kind) '(:assertion :product-condition)) "Bootstrap semantic failure."))))
        (progn (check-equal (getf child :status) :ok "baseline status")
               (check-equal (getf child :phase) :complete "baseline complete")
               (check-equal (getf child :passed-tests) tests "every baseline test passed")
               (demand (null (getf child :failed-test)) "Baseline failed test.")))
    (append (list :name name :result (getf result :result) :registered (length tests) :executed (length expected)
                  :failed-test (getf child :failed-test) :failure-kind (getf child :failure-kind)
                  :private-fasl (length compiled)) (log-check inventory evidence))))

(defun operation-audit (campaign inventory preflight)
  (let ((records (loop for b in inventory for p = (getf b :path)
                       when (and (uiop:string-prefix-p (namestring campaign) p)
                                 (search "/raw/operations/" p) (uiop:string-suffix-p p "/execution.lisp")) collect p))
        (counts (make-hash-table :test #'equal)))
    (unless preflight (demand (null records) "Bootstrap unexpectedly contains preflight operation records."))
    (when preflight (demand records "Missing bounded operation records."))
    (dolist (p records)
      (let* ((r (read-native p)) (argv (getf r :argv))
             (base (uiop:pathname-directory-pathname p))
             (parts (pathname-directory base)) (n (car (last parts)))
             (scope (namestring (make-pathname :directory (butlast parts) :defaults base))))
        (demand (and (eql 0 (getf r :exit-code)) (null (getf r :signal))
                     (null (getf r :timed-out)) (eq :ok (getf r :status)) (eq :reaped (getf r :cleanup))
                     (typep (getf r :deadline-seconds) '(real (0) 120))) "Operation not bounded/reaped ~A." p)
        (demand (or (and (equal (first argv) "git")
                        (member (rest argv) '(("rev-parse" "HEAD") ("rev-parse" "HEAD^{tree}")
                                              ("status" "--porcelain=v1" "--untracked-files=no")) :test #'equal))
                    (and (equal (subseq argv 0 (min 2 (length argv))) '("python3" "-c"))
                         (equal +sha-batch-code+ (third argv))
                         (every (lambda (f) (and (stringp f)
                                                (uiop:string-prefix-p (namestring (getf *audit-options* :workspace)) f)))
                                (nthcdr 3 argv)))) "Operation argv shape ~A." p)
        (dolist (leaf '("stdout.log" "stderr.log"))
          (demand (<= (getf (lookup-byte inventory (path base leaf)) :bytes) 2097152) "Operation log exceeds budget."))
        (demand (and (stringp n) (every #'digit-char-p n)) "Operation serial format.")
        (push (parse-integer n) (gethash scope counts))))
    (when preflight
      (demand (= 6 (hash-table-count counts)) "Expected one parent + five separate child operation scopes.")
      (maphash (lambda (scope serials)
                 (declare (ignore scope))
                 (check-equal (sort serials #'<) (loop for n from 1 to (length serials) collect n) "contiguous operation scope")) counts))
    (list :operations (length records) :separate-scopes (hash-table-count counts))))

(defun audit-campaign (o tree inventory preflight)
  (let* ((directory (getf o (if preflight :preflight-dir :bootstrap-dir)))
         (wrapper (audit-wrapper (getf o (if preflight :preflight-command :bootstrap-command)) o tree))
         (r (read-native (path directory "report.lisp")))
         (tail (command-tail wrapper)) (tests (frozen-catalog o preflight))
         (targets (mutants preflight)) (files (inventory-files tree))
         (canonical (manifest-at inventory files (getf o :workspace)))
         (compiled (frozen-asdf-source-files o tree)) (children nil))
    (demand (and (= (if preflight 42 25) (length tests))
                 (= (length tests) (length (remove-duplicates tests :test #'equal)))) "Frozen test cardinality/uniqueness.")
    (check-equal (first tail) "--check" "campaign mode")
    (check-equal (namestring (resolve-directory o (second tail))) (namestring directory) "native campaign destination")
    (check-equal (cddr tail) (when preflight '("--suite" "preflight")) "explicit preflight/default bootstrap CLI")
    (check-equal (getf r :argv) tail "native/report arguments")
    (check-equal (read-data-string (getf wrapper :stdout)) r "wrapper stdout / final report")
    (demand (and (eql 1 (getf r :schema-version)) (eq :ok (getf r :status))
                 (eq (if preflight :get-active-preflight-mutations :get-active-init-mutations) (getf r :kind))
                 (eq :stable (getf r :source-consistency))
                 (<= (getf wrapper :started-at-universal-time) (getf r :started-at-universal-time)
                     (getf r :finished-at-universal-time) (getf wrapper :finished-at-universal-time))
                 (<= (- (getf r :finished-at-universal-time) (getf r :started-at-universal-time)) 902)) "Campaign identity/finality/time.")
    (when preflight
    (check-equal (getf r :canonical-source-before) (getf r :canonical-source-after) "source identity stable")
    (check-equal (getf r :canonical-source-before)
                 (list :commit (getf o :commit) :tree (line-output (getf o :workspace) "rev-parse" (format nil "~A^{tree}" (getf o :commit))) :working-tree "") "freeze identity"))
    (dolist (key (if preflight '(:frozen-manifest :canonical-source-manifest :source-manifest-after) '(:frozen-manifest :source-manifest-after)))
      (check-equal (getf r key) canonical key))
    (check-equal (getf r :expected-tests) tests "baseline source identities")
    (when preflight (check-equal (getf r :suite-tests) tests "parent whole registry") (check-equal (getf r :profile) :preflight "parent profile"))
    (check-equal (getf r :test-selection) (suite-selection preflight) "ASDF-consistent registry catalog")
    (check-equal (getf r :targets) targets "preregistered four targets")
    (check-equal (getf r :collector-exclusions) (unless preflight '("raw/isolatedtests/")) "declared collection scope")
    (check-equal (getf r :child-deadline-seconds) 120 "child deadline")
    (check-equal (getf r :campaign-deadline-seconds) 900 "campaign deadline")
    (check-equal (getf r :counters) '(:baseline-passed 1 :baseline-failed 0 :detected 4 :survived 0
                                   :compilation-error 0 :infrastructure-error 0 :invalid 0) "terminal counters")
    (demand (and (= 4 (length (getf r :mutants)))
                 (= 4 (length (remove-duplicates (getf r :mutants) :key (lambda (e) (getf e :name)) :test #'equal)))) "Four unique mutants.")
    (push (audit-child o directory r (getf r :baseline) nil preflight tests files compiled inventory canonical) children)
    (dolist (m targets)
      (let ((entry (find (getf m :name) (getf r :mutants) :test #'equal :key (lambda (e) (getf e :name)))))
        (demand entry "Missing expected mutant.")
        (push (audit-child o directory r entry m preflight tests files compiled inventory canonical) children)))
    (list :profile (if preflight :preflight :bootstrap) :baseline (length tests) :detected 4
          :source-files (length files) :compiled-per-child (length compiled)
          :tests tests :children (reverse children) :operations (operation-audit directory inventory preflight))))
(defun audit-negative (o tree key duplicate)

    (check-equal (first tail) "--check" "negative mode")
    (check-equal (cddr tail) (if duplicate '("--suite" "preflight" "--suite" "preflight") '("--suite" "unknown")) "negative guard arguments")
    (demand (and (eql 2 (getf r :exit-code)) (equal "" (getf r :stdout))
                 (equal (format nil "tools/get-active-init-mutation.lisp COD-61: usage/unknown or duplicate suite.~%")
                        (getf r :stderr))
                 (not (probe-file (resolve-directory o (second tail))))) "Expected usage error / absent output directory.")
    (list :case (if duplicate :duplicate-suite :unknown-suite) :exit 2 :campaign-created nil :classification :usage)))

(defun check-current-git (o tree)
  (check-equal (line-output (getf o :workspace) "rev-parse" "HEAD") (getf o :commit) "current HEAD")
  (check-equal (git-output (getf o :workspace) "status" "--porcelain=v1" "--untracked-files=no") "" "clean tracked source")
  (let* ((input (format nil "~{~A~%~}" (mapcar (lambda (r) (getf r :path)) tree)))
         (stdout (uiop:run-program '("git" "hash-object" "--stdin-paths") :directory (getf o :workspace)
                                  :input (make-string-input-stream input) :output :string :error-output :string))
         (hashes (remove "" (uiop:split-string stdout :separator '(#\Newline)) :test #'equal)))
    (check-equal hashes (mapcar (lambda (r) (getf r :git-blob)) tree) "every actual source path / frozen Git blob")))
(defun main ()
  (let* ((o (options)) (*audit-options* o) (workspace (getf o :workspace))
         (commit (getf o :commit)) (reader (path workspace "tools/evidence-storage.lisp"))
         (tree (expected-source-tree workspace commit))
         (reader-entry (find "tools/evidence-storage.lisp" tree :test #'equal :key (lambda (e) (getf e :path))))
         (output (getf o :output)) (before-path (pathname (concatenate 'string (namestring output) ".before.lisp")))
         (after-path (pathname (concatenate 'string (namestring output) ".after.lisp"))))
    (demand (and (= 40 (length commit)) reader-entry
                 (equal (getf reader-entry :git-blob) (line-output workspace "hash-object" (namestring reader))))
            "Trusted reader is not the explicit frozen Git blob.")
    (demand (and (uiop:subpathp output (uiop:pathname-directory-pathname *auditor-source*))
                 (not (probe-file output)) (not (probe-file before-path)) (not (probe-file after-path))) "Output must be fresh under auditor folder.")
    (load reader :verbose nil :print nil)
    (check-current-git o tree)
    (let* ((wrapper-keys '(:preflight-command :bootstrap-command :unknown-command :duplicate-command
                          :preflight-selftest-command :bootstrap-selftest-command))
           (paths (append (mapcar (lambda (e) (path workspace (getf e :path))) tree)
                          (list *auditor-source* (getf o :preflight-dir) (getf o :bootstrap-dir)
                                (selftest-directory o (getf o :preflight-selftest-command))
                                (selftest-directory o (getf o :bootstrap-selftest-command)))
                          (mapcar (lambda (k) (uiop:pathname-directory-pathname (getf o k))) wrapper-keys)))
           (before (snapshot paths)) (report nil))
      (write-data (list :status :captured :files before) before-path)
      (check-equal +sha-batch-code+
                   (let* ((text (source-text o "tools/get-active-init-mutation.lisp"))
                          (start (search "\"import os,sys,stat,hashlib" text)))
                     (demand start "Frozen hash utility literal absent.")
                     (let ((*read-eval* nil)) (read-from-string text t nil :start start)))
                   "batch hash utility exactly from frozen source")
      (setf report (list :schema-version 1 :kind :independent-preflight-mutation-audit :status :ok
                         :commit commit :tree (line-output workspace "rev-parse" (format nil "~A^{tree}" commit))
                         :full-source-files (length tree)
                         :preflight (audit-campaign o tree before t)
                         :bootstrap (audit-campaign o tree before nil)
                         :negative (list (audit-negative o tree :unknown-command nil) (audit-negative o tree :duplicate-command t))
                         :self-tests (list (audit-selftest o tree :preflight-selftest-command t)
                                           (audit-selftest o tree :bootstrap-selftest-command nil))))
      (let ((after (snapshot paths)))
        (write-data (list :status :captured :files after) after-path)
        (check-equal before after "two-pass complete source/campaign/wrapper/helper byte inventory")
        (check-current-git o tree)
        (setf report (append report (list :byte-inventory-files (length before)
                                         :byte-inventory-bytes (reduce #'+ before :key (lambda (e) (getf e :bytes)))
                                         :two-pass :stable :before (namestring before-path) :after (namestring after-path)
                                         :auditor-sha256 (file-sha before *auditor-source*)
                                         :reader-sha256 (file-sha before reader)
                                         :limits '(:read-eval-nil-and-eof :reports-and-fasl-data-only
                                                   :selected-four-mutations-each :not-mcdc :no-requirement-promotion
                                                   :non-atomic-two-pass-observation :operation-cleanup-from-recorded-contract
                                                   :bootstrap-has-no-bounded-operation-records))))))
      (write-data report output)
      (write report :pretty t) (terpri))))
