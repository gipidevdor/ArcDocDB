+(defun parallel-map-four (function items)
+  "Ordered results, at most four workers; join all and propagate every failure."
+  (let* ((input (coerce items 'vector)) (size (length input)) (workers (min 4 size))
+         (results (make-array size)) (failures (make-array size :initial-element nil))
+         (threads nil) (launch-failure nil) (join-failures nil)
+         (defaults *default-pathname-defaults*) (package *package*)
+         (stdout *standard-output*) (stderr *error-output*) (archive *archive*) (audit-directory *audit-directory*))
+    (unwind-protect
+         (handler-case
+             (dotimes (worker workers)
+               (let ((start worker))
+                 (push (sb-thread:make-thread
+                        (lambda ()
+                          (let ((*default-pathname-defaults* defaults) (*package* package)
+                                (*standard-output* stdout) (*error-output* stderr)
+                                (*archive* archive) (*audit-directory* audit-directory))
+                            (loop for index from start below size by workers do
+                              (handler-case (setf (aref results index) (funcall function (aref input index)))
+                                (serious-condition (condition) (setf (aref failures index) condition)))))
+                        :name (format nil "publication-worker-~D" start)) threads)))
+           (serious-condition (condition) (setf launch-failure condition)))
+      (sb-sys:without-interrupts
+        (dolist (thread threads)
+          (handler-case (sb-thread:join-thread thread)
+            (serious-condition (condition) (push condition join-failures))))))
+    (let ((errors (loop for condition across failures for index from 0 when condition
+                        collect (list index (princ-to-string condition)))))
+      (when (or launch-failure join-failures errors)
+        (error "Parallel publication failed after joining all workers: launch=~A joins=~S entries=~S."
+               launch-failure (mapcar #'princ-to-string join-failures) errors)))
+    (coerce results 'list)))
