;;;; C4: derivazione soltanto da produttori chiusi; nessuna esecuzione del prodotto.
(require :asdf)
(load "tools/evidence-storage.lisp")
(defun require-proof (truth detail) (unless truth (error "summary.lisp COD-61: ~A" detail)))
(defun median-proof (numbers)
  (require-proof (and numbers (oddp (length numbers))) "mediana richiede campioni dispari")
  (nth (floor (length numbers) 2) (sort (copy-list numbers) #'<)))
(defun successful-proof (path)
  (let ((record (arcdocdb.evidence:read-evidence path)))
    (require-proof (eq :ok (getf record :status)) "input non riuscito") record))
(defun row-proof (windows mode readers)
  (let* ((selected (remove-if-not (lambda (w) (and (eq mode (getf w :mode))
                                                  (= readers (getf w :readers)))) windows))
         (rates (mapcar (lambda (w) (when (plusp (getf w :seconds))
                                     (/ (getf w :completed-renames) (getf w :seconds)))) selected))
         (heaps (mapcar (lambda (w) (getf w :heap-bytes)) selected)))
    (require-proof (= 5 (length selected)) "cinque repliche richieste")
    (list :mode mode :readers readers :replicas 5
          :renames-per-second-min (when (every #'numberp rates) (reduce #'min rates))
          :renames-per-second-median (when (every #'numberp rates) (median-proof rates))
          :renames-per-second-max (when (every #'numberp rates) (reduce #'max rates))
          :heap-bytes-min (reduce #'min heaps) :heap-bytes-median (median-proof heaps)
          :heap-bytes-max (reduce #'max heaps))))
(let ((args (uiop:command-line-arguments)))
  (if (equal args '("--self-test"))
      (progn
        (require-proof (= 3 (median-proof '(5 1 3))) "mediana")
        (require-proof (= 1 (median-proof '(1))) "singolo campione")
        (let ((caught nil)) (handler-case (median-proof nil) (error () (setf caught t)))
          (require-proof caught "mediana vuota accettata"))
        (write '(:schema-version 1 :kind :io-rename-summary-self-test :status :ok :checks 3)) (terpri))
      (progn
        (require-proof (= 6 (length args)) "bench mutation bench-producer mutation-producer check output")
        (let* ((bench (successful-proof (first args))) (mutations (successful-proof (second args)))
               (producers (mapcar #'successful-proof (subseq args 2 5))) (windows (getf bench :windows))
               (renames (reduce #'+ windows :key (lambda (w) (getf w :completed-renames))))
          (require-proof (every (lambda (r) (and (eq :stable (getf r :source-consistency))
                                                (eql 0 (getf r :exit-code)))) producers) "produttori stabili/chiusi")
          (require-proof (and (= 30 (length windows)) (= 30 (length (remove-duplicates windows
                              :key (lambda (w) (list (getf w :mode) (getf w :readers) (getf w :replica))) :test #'equal)))
                              (every (lambda (w) (and (eq :ok (getf w :status))
                                                      (= (getf w :requested-renames) (getf w :completed-renames)))) windows)
                              (= 17920 renames)) "matrice/oracolo rinomine")
          (require-proof (and (eq :baseline-passed (getf (getf mutations :baseline) :result))
                              (= 68 (getf mutations :test-count)) (= 4 (length (getf mutations :mutants)))
                              (every (lambda (m) (eq :detected (getf m :result))) (getf mutations :mutants)))
                         "baseline/mutanti")
          (let ((record (list :schema-version 1 :kind :io-rename-derived-results :status :ok
                              :test-count 68 :mutants-detected 4 :mutants-total 4 :windows 30
                              :renames renames :directory-fsyncs
                              (reduce #'+ windows :key (lambda (w) (getf w :directory-fsyncs)))
                              :rows (loop for mode in '(:rename-only :rename-and-directory-fsync) append
                                      (loop for readers in '(1 2 4) collect (row-proof windows mode readers)))
                              :inputs (mapcar (lambda (f) (list :file f :sha256 (arcdocdb.evidence:file-sha256 f)))
                                              (subseq args 0 5))
                              :limits '(:local-machine :whole-process-heap :no-database-throughput
                                        :no-zero-heap-or-power-loss-or-mcdc-qualification))))
            (require-proof (= 8960 (getf record :directory-fsyncs)) "fsync count")
            (with-open-file (out (sixth args) :direction :output :if-exists :error)
              (write record :stream out :pretty t) (terpri out))
            (write record :pretty t) (terpri))))))
