(require :asdf)
(require :sb-posix)
(load "tools/evidence-storage.lisp")
(defun save-data (data path)
  (with-open-file (stream path :direction :output :if-exists :supersede)
    (let ((*print-readably* nil) (*print-escape* t)) (write data :stream stream :pretty t) (terpri stream))))
(defun copy-exact (source target)
  (ensure-directories-exist target)
  (if (probe-file target)
      (unless (equal (arcdocdb.evidence:file-sha256 source)
                     (arcdocdb.evidence:file-sha256 target))
        (error "Target diverso, nessuna sovrascrittura: ~A" target))
      (uiop:copy-file source target))
  (unless (equal (arcdocdb.evidence:file-sha256 source)
                 (arcdocdb.evidence:file-sha256 target))
    (error "Copia non integra: ~A" target))
  target)
(defun descriptor (path)
  (with-open-file (stream path)
    (let ((*read-eval* nil)) (read stream))))
(defun copy-record (source target &optional parsed-original)
  (let ((data (descriptor source)))
    (if (eq (getf data :kind) :compressed-evidence)
        (let* ((payload (merge-pathnames (getf data :payload) source))
               (new-name (concatenate 'string (file-namestring target) ".gz"))
               (new-payload (merge-pathnames new-name target)))
          (copy-exact payload new-payload)
          (setf (getf data :payload) new-name)
          (when (probe-file target)
            (unless (equal (descriptor target) data) (error "Descriptor diverso: ~A" target)))
          (save-data data target))
        (copy-exact source target)))
  (let ((original (or parsed-original (arcdocdb.evidence:read-evidence source)))
        (copy (arcdocdb.evidence:read-evidence target)))
    (unless (equalp original copy) (error "Record risolti diversi: ~A" target)))
  target)

(defparameter *base* "spikes/results/2026-10-15-active-integrazione/")
(defun root-name-p (name)
  (and (stringp name) (<= 1 (length name) 96)
       (every (lambda (c) (or (char<= #\a c #\z) (char<= #\0 c #\9) (char= c #\-))) name)))
(defun within-limit (path)
  (unless (<= (arcdocdb.evidence:file-bytes path)
              (if (equal "gz" (pathname-type path)) 8388608 1048576))
    (error "Integrazione COD-61: file oltre limite: ~A" path))
  path)
(defun terminal-p (data)
  (and (eql 1 (getf data :schema-version))
       (if (eq :command-verification (getf data :kind))
           (and (member (getf data :status) '(:ok :failed :source-changed))
                (integerp (getf data :finished-at-universal-time)))
           (member (getf data :status) '(:complete :failed)))))
(defun raw (leaf)
  (let ((path (merge-pathnames leaf *base*)))
    (within-limit path)
    (list :file leaf :bytes (arcdocdb.evidence:file-bytes path)
          :sha256 (arcdocdb.evidence:file-sha256 path))))
(defun publish ()
  "Produttori chiusi espliciti, copie/digest e limiti locali; nessuna promozione degli esiti."
  (let* ((selection (arcdocdb.evidence:read-evidence (merge-pathnames "selection.lisp" *base*)))
         (roots (getf selection :roots)) (entries nil))
    (unless (and (eq :active-integration-selection (getf selection :kind))
                 (listp roots) (<= 1 (length roots) 32) (every #'root-name-p roots)
                 (= (length roots) (length (remove-duplicates roots :test #'equal))))
      (error "Integrazione COD-61: selezione invalida."))
    (dolist (name roots)
      (let* ((directory (format nil "spikes/out/~A/" name))
             (producer (arcdocdb.evidence:read-evidence (merge-pathnames "report.lisp" directory))))
        (unless (terminal-p producer) (error "Integrazione COD-61: produttore non chiuso: ~A" name))
        (dolist (leaf (append '("report.lisp" "conservazione.lisp" "self-test-metrics.lisp" "self-test-sensor.lisp")
                             (loop for i from 1 to 10 collect (format nil "SPK-~2,'0D.lisp" i))))
          (let ((source (merge-pathnames leaf directory)))
            (when (probe-file source)
              (let* ((target-name (format nil "~A-~A" name leaf))
                     (target (merge-pathnames target-name *base*))
                     (data (arcdocdb.evidence:read-evidence source)))
                (within-limit source) (copy-record source target data) (within-limit target)
                (when (eq :compressed-evidence (getf (descriptor target) :kind))
                  (within-limit (merge-pathnames (getf (descriptor target) :payload) target)))
                (push (list :artifact target-name :kind (getf data :kind) :status (getf data :status)
                            :source (namestring (truename source)) :scope :active-pool-integration
                            :sha256 (arcdocdb.evidence:file-sha256 target)) entries)))))))
    (copy-exact *load-truename* (merge-pathnames "publish-source.txt" *base*))
    (save-data (list :schema-version 1 :kind :evidence-catalog :scope :active-pool-integration
                     :entries (nreverse entries)
                     :associated-raw-files (mapcar #'raw '("report.md" "publish.lisp" "publish-source.txt" "selection.lisp"))
                     :limits '(:closed-producers :local-copy-integrity :size-limits :no-engine-qualification))
               (merge-pathnames "catalogo.lisp" *base*))
    (format t "~S~%" (list :schema-version 1 :kind :active-integration-publication :status :ok
                           :records (length entries)))))
(defun audit ()
  (let* ((catalog (arcdocdb.evidence:read-evidence (merge-pathnames "catalogo.lisp" *base*)))
         (entries (getf catalog :entries)) (files (getf catalog :associated-raw-files)))
    (dolist (entry entries)
      (let ((path (merge-pathnames (getf entry :artifact) *base*)))
        (within-limit path)
        (unless (equal (getf entry :sha256) (arcdocdb.evidence:file-sha256 path))
          (error "Integrazione COD-61: digest record diverso."))
        (arcdocdb.evidence:read-evidence path)))
    (dolist (entry files)
      (let ((path (merge-pathnames (getf entry :file) *base*)))
        (within-limit path)
        (unless (and (= (getf entry :bytes) (arcdocdb.evidence:file-bytes path))
                     (equal (getf entry :sha256) (arcdocdb.evidence:file-sha256 path)))
          (error "Integrazione COD-61: raw diverso."))))
    (format t "~S~%" (list :schema-version 1 :kind :active-integration-audit :status :ok
                           :records (length entries) :raw-files (length files)))))
(defun self-test ()
  (let ((checks 0))
    (flet ((check (x) (unless x (error "Integrazione COD-60: self-test.")) (incf checks)))
      (check (root-name-p "4001000000-command-1-0"))
      (dolist (bad '("" "../a" "a/b" "A" "a.b")) (check (not (root-name-p bad))))
      (check (terminal-p '(:schema-version 1 :kind :command-verification :status :ok :finished-at-universal-time 1)))
      (check (not (terminal-p '(:schema-version 1 :kind :command-verification :status :running))))
      (check (not (terminal-p '(:schema-version 1 :kind :command-verification :status :ok))))
      (check (terminal-p '(:schema-version 1 :status :complete)))
      (let* ((directory (format nil "spikes/out/active-integration-self-~D/" (sb-posix:getpid)))
             (source (merge-pathnames "source.lisp" directory)) (copy (merge-pathnames "copy.lisp" directory)))
        (sb-posix:mkdir directory #o700)
        (unwind-protect
             (progn
               (save-data '(:schema-version 1 :kind :fixture :status :ok) source)
               (copy-record source copy) (check (equalp (descriptor source) (descriptor copy)))
               (save-data '(:schema-version 1 :kind :fixture :status :changed) copy)
               (check (handler-case (progn (copy-record source copy) nil) (error () t))))
          (dolist (path (list source copy)) (when (probe-file path) (delete-file path)))
          (sb-posix:rmdir directory)))
      (format t "~S~%" (list :schema-version 1 :kind :active-integration-self-test :status :ok :checks checks)))))
(let ((args (uiop:command-line-arguments)))
  (cond ((equal args '("--self-test")) (self-test))
        ((equal args '("--publish")) (publish)) ((equal args '("--audit")) (audit))
        (t (error "Integrazione COD-61: --self-test, --publish o --audit."))))
