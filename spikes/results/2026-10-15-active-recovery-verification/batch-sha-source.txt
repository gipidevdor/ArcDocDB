;;;; Byte inventory only; same bounded hashlib literal as frozen Git8c, no evidence LOAD.
(require :asdf)
(require :sb-posix)
(defpackage #:preflight.batch.sha (:use #:cl) (:export #:snapshot #:table))
(in-package #:preflight.batch.sha)
(declaim (optimize (safety 3) (debug 3)))
(defparameter *workspace* (truename "./"))
(defparameter *anchor* "8c0866711c50d047f87b9c484ce56d42a87ba19c")
(defparameter *source* (merge-pathnames "tools/get-active-init-mutation.lisp" *workspace*))
(defun git-line (&rest arguments)
  (string-right-trim '(#\Newline #\Return)
    (uiop:run-program (cons "git" arguments) :directory *workspace* :output :string :error-output :string)))
(defparameter *source-blob* (git-line "rev-parse" (concatenate 'string *anchor* ":tools/get-active-init-mutation.lisp")))
(defparameter *sha-code*
  (progn
    (unless (equal *source-blob* (git-line "hash-object" "--" (namestring *source*)))
      (error "Batch hash source is not frozen Git8c blob."))
    (let* ((text (uiop:read-file-string *source*)) (anchor "\"import os,sys,stat,hashlib")
           (position (search anchor text)))
      (unless (and (<= (length text) 1048576) position (not (search anchor text :start2 (1+ position))))
        (error "Frozen hash literal absent/nonunique or source budget."))
      (let ((*read-eval* nil) (*readtable* (copy-readtable nil)))
        (multiple-value-bind (code end) (read-from-string text t nil :start position)
          (declare (ignore end))
          (unless (and (stringp code) (<= (length code) 8192)) (error "Frozen hash literal shape."))
          code)))))
(defun sha-p (value)
  (and (stringp value) (= 64 (length value))
       (every (lambda (c) (or (char<= #\0 c #\9) (char<= #\a c #\f))) value)))
(defun checked-path (value)
  (let* ((path (merge-pathnames value *workspace*)) (parts (pathname-directory path)))
    (unless (and (uiop:absolute-pathname-p path) (not (wild-pathname-p path))
                 (notany (lambda (p) (member p '(:up :back :wild :wild-inferiors))) parts))
      (error "Unsupported inventory path: ~A." value))
    (loop for n from 1 to (length parts) do
      (let* ((parent (make-pathname :directory (subseq parts 0 n) :name nil :type nil :defaults path))
             (name (string-right-trim "/" (namestring parent))))
        (unless (sb-posix:s-isdir (sb-posix:stat-mode (sb-posix:lstat (if (equal name "") "/" name))))
          (error "Symlink/special inventory ancestor: ~A." parent))))
    (unless (sb-posix:s-isreg (sb-posix:stat-mode (sb-posix:lstat (namestring path))))
      (error "Symlink/special inventory file: ~A." path))
    (namestring (truename path))))
(defun file-identity (path)
  (let ((s (sb-posix:lstat path)))
    (unless (and (sb-posix:s-isreg (sb-posix:stat-mode s)) (<= 0 (sb-posix:stat-size s) 16777216))
      (error "Inventory file shape/16MiB budget: ~A." path))
    (list (sb-posix:stat-dev s) (sb-posix:stat-ino s) (sb-posix:stat-mode s)
          (sb-posix:stat-size s) (sb-posix:stat-mtime s) (sb-posix:stat-ctime s))))
(defun argv-bytes (argv)
  (loop for value in argv sum (1+ (length (sb-ext:string-to-octets value :external-format :utf-8)))))
(defun snapshot (paths)
  "Fresh sorted path/byte/SHA inventory; caller LOADs the trusted evidence oracle."
  (unless (<= (length paths) 50000) (error "Inventory path count exceeds 50000."))
  (let* ((files (sort (remove-duplicates (mapcar #'checked-path paths) :test #'equal) #'string<))
         (before (mapcar #'file-identity files)) (pending (copy-list files)) (hashes nil)
         (prefix (list "python3" "-c" *sha-code*)))
    (loop while pending do
      (let ((batch nil) (bytes (argv-bytes prefix)))
        (loop while (and pending (< (length batch) 512)
                         (<= (+ bytes (argv-bytes (list (first pending)))) 160000))
              do (incf bytes (argv-bytes (list (first pending)))) (push (pop pending) batch))
        (unless batch (error "Single inventory path exceeds argv budget."))
        (setf batch (nreverse batch))
        (multiple-value-bind (stdout stderr exit)
            (uiop:run-program (append prefix batch) :output :string :error-output :string :ignore-error-status t)
          (unless (eql 0 exit) (error "Batch SHA failed: ~A" stderr))
          (let ((values (remove "" (uiop:split-string stdout :separator '(#\Newline)) :test #'equal)))
            (unless (and (= (length batch) (length values)) (every #'sha-p values))
              (error "Batch SHA cardinality/64hex mismatch."))
            (setf hashes (nconc hashes values))))))
    (unless (and (equal files (mapcar #'checked-path files)) (equal before (mapcar #'file-identity files)))
      (error "Inventory path/identity drift during batch SHA."))
    (let ((rows (loop for file in files for stat in before for hash in hashes
                      collect (list :path file :bytes (fourth stat) :sha256 hash))))
      (dolist (index (remove-duplicates (when rows (list 0 (floor (length rows) 2) (1- (length rows))))))
        (let ((row (nth index rows)))
          (unless (equal (getf row :sha256)
                         (uiop:symbol-call :arcdocdb.evidence :file-sha256 (getf row :path)))
            (error "Batch SHA/native oracle sample mismatch: ~A." (getf row :path)))))
      rows)))
(defun table (rows)
  "No hashing: EQUAL table keyed by absolute canonical path, value is full row."
  (let ((table (make-hash-table :test #'equal)))
    (dolist (row rows)
      (let ((path (getf row :path)))
        (unless (and (stringp path) (uiop:absolute-pathname-p (pathname path))
                     (typep (getf row :bytes) '(integer 0)) (sha-p (getf row :sha256))
                     (not (gethash path table))) (error "Invalid/duplicate snapshot row."))
        (setf (gethash path table) row)))
    table))
