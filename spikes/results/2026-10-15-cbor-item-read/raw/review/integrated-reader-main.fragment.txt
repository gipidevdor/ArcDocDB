(defun integers-in (line from until)
  (let ((start (search from line)) (values nil))
    (assert start) (incf start (length from))
    (let ((end (or (and until (search until line :start2 start)) (length line))))
      (loop while (< start end)
            do (if (digit-char-p (char line start))
                   (multiple-value-bind (value next) (parse-integer line :start start :end end :junk-allowed t)
                     (push value values) (setf start next)) (incf start))))
    (nreverse values)))
