(require :asdf)
(load "tools/evidence-storage.lisp")
(let* ((args (uiop:command-line-arguments))
       (r (arcdocdb.evidence:read-evidence (first args)))
       (lines (uiop:split-string (getf r :stdout) :separator '(#\Newline)))
       (count (count-if (lambda (line)
                          (and (uiop:string-prefix-p "ok    " line)
                               (uiop:string-prefix-p "TEST-" (string-upcase (subseq line 6))))) lines))
       (summary (list :schema-version 1 :kind :directory-id-verification-summary
                      :record (first args) :status (getf r :status) :exit-code (getf r :exit-code)
                      :consistency (getf r :source-consistency) :wall-seconds (getf r :wall-seconds)
                      :source-count (length (getf r :source-blobs-after)) :named-tests count
                      :stderr-characters (length (getf r :stderr)))))
  (unless (and (eq :ok (getf r :status)) (zerop (getf r :exit-code))
               (eq :stable (getf r :source-consistency))) (error "Registro non concluso con successo."))
  (write summary :pretty nil) (terpri)
  (dolist (line lines)
    (when (and (plusp (length line)) (digit-char-p (char line 0))
               (or (search "test recovery superati" line) (search "test verticali superati" line)
                   (search "requisiti," line) (search "link controllati" line)
                   (search "file, 0 violazioni" line) (search "spike completati" line)))
      (write-line line)))
  (with-open-file (output (second args) :direction :output :if-exists :error)
    (write summary :stream output :pretty t) (terpri output)))
