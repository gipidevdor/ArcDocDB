(require :asdf)
(load "tools/evidence-storage.lisp")
(let ((observations nil))
  (dolist (entry '(("coverage-initial" "4001154281-command-84702-0")
                   ("mutation-self-test" "4001154308-command-85197-0")
                   ("coverage-final" "4001154353-command-85838-0")
                   ("vertical" "4001154377-command-86278-0")
                   ("mutation" "4001154425-command-87352-0")))
    (let* ((path (format nil "spikes/out/~A/report.lisp" (second entry)))
           (r (arcdocdb.evidence:read-evidence path))
           (observation (list :label (first entry) :path path :status (getf r :status)
                              :exit-code (getf r :exit-code) :consistency (getf r :source-consistency)
                              :wall-seconds (getf r :wall-seconds)
                              :sources (length (getf r :source-blobs-after))
                              :stderr-characters (length (getf r :stderr)))))
      (write observation :pretty nil) (terpri)
      (dolist (line (uiop:split-string (getf r :stdout) :separator '(#\Newline)))
        (when (or (search "test recovery superati" line) (search "test verticali superati" line))
          (write-line line)))
      (push observation observations)))
  (with-open-file (output "spikes/out/directory-id-audit/development-records.lisp"
                          :direction :output :if-exists :error)
    (write (nreverse observations) :stream output :pretty t) (terpri output)))
