;;;; C4: contatori nativi e HTML, nessun reset, esclusione o modifica dei bit.
(require :asdf)
(defun did-counts (row)
  (let ((paths (second row)) (bits (cddr row))
        (covered 0) (forms 0) (taken 0) (branches 0) (missing nil))
    (unless (and (vectorp paths) (typep bits 'bit-vector) (= (length paths) (length bits)))
      (error "Dimensioni native discordanti."))
    (loop for path across paths for bit across bits for index from 0 do
      (if (member (first path) '(:then :else))
          (progn (incf branches) (incf taken bit))
          (progn (incf forms) (incf covered bit)))
      (when (zerop bit) (push (list index path) missing)))
    (list :forms covered :total-forms forms :branches taken :total-branches branches
          :missing (nreverse missing))))
(defun did-self-test ()
  (unless (equal (did-counts '("fixture" #((0) (:then 1) (:else 1)) . #*110))
                 '(:forms 1 :total-forms 1 :branches 1 :total-branches 2 :missing ((2 (:else 1)))))
    (error "Oracolo indipendente dei contatori fallito."))
  (let ((rejected nil))
    (handler-case (did-counts '("fixture" #((0)) . #*10)) (error () (setf rejected t)))
    (unless rejected (error "Dimensioni discordanti non rifiutate."))))
(did-self-test)
(let* ((*read-eval* nil)
       (rows (with-open-file (input "spikes/out/sources-coverage-final/coverage-state.lisp") (read input)))
       (html (uiop:read-file-string "spikes/out/sources-coverage-final/cover-index.html"))
       (observed nil) (forms 0) (total-forms 0) (branches 0) (total-branches 0))
  (dolist (name '("inventory-sources-types.lisp" "inventory-sources-query.lisp"
                  "inventory-sources-build.lisp" "inventory-sources.lisp"))
    (let* ((suffix (concatenate 'string "/src/recovery/" name))
           (selected (remove-if-not (lambda (row) (uiop:string-suffix-p (first row) suffix)) rows)))
      (unless (= 1 (length selected)) (error "Sorgente assente o ambiguo: ~A" name))
      (let* ((counts (did-counts (first selected)))
             (position (search (concatenate 'string name "</a>") html))
             (start (and position (search "<tr" html :end2 position :from-end t)))
             (end (and position (search "</tr>" html :start2 position)))
             (row (and start end (subseq html start (+ end 5)))))
        (dolist (pair '((:forms :total-forms) (:branches :total-branches)))
          (unless (and row (search (format nil "<td>~D</td><td>~D</td>"
                                          (getf counts (first pair)) (getf counts (second pair))) row))
            (error "HTML e dati nativi discordanti: ~A" name)))
        (incf forms (getf counts :forms)) (incf total-forms (getf counts :total-forms))
        (incf branches (getf counts :branches)) (incf total-branches (getf counts :total-branches))
        (push (list :file (subseq suffix 1) :counts counts :html-row row) observed))))
  (let ((result (list :schema-version 1 :kind :recovery-sources-coverage-audit :status :ok
                      :self-test :passed :scope :four-new-product-files :files (nreverse observed)
                      :forms forms :total-forms total-forms :branches branches :total-branches total-branches
                      :exclusions nil :qualification :open)))
    (with-open-file (output "spikes/out/sources-audit/coverage-result-final.lisp"
                            :direction :output :if-exists :error)
      (write result :stream output :pretty t) (terpri output))
    (write result :pretty t) (terpri)))
