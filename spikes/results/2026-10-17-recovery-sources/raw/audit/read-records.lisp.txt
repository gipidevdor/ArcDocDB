(require :asdf)
(load "tools/evidence-storage.lisp")
(let ((observations nil))
  (dolist (entry '(
("initial-compile" "4001217236-command-75881-0")
("initial-budgets" "4001217279-command-77284-0")
("mutation-self-initial" "4001217286-command-77574-0")
("targeted-first" "4001217356-command-78886-0")
("targeted-symlink" "4001217410-command-79559-0")
("targeted-before-nesting-fix" "4001217433-command-79967-0")
("mutation-before-nesting-fix" "4001217447-command-80244-0")
("coverage-before-nesting-fix" "4001217447-command-80246-0")
("check-interrupted" "4001217447-command-80269-0")
("targeted-u64-history" "4001217621-command-84148-0")
("targeted-final" "4001217691-command-85566-0")
("mutation-self-merge-diagnostic" "4001217712-command-85921-0")
("mutation-self-final" "4001217724-command-86188-0")
("mutation-final" "4001217731-command-86323-0")
("coverage-final" "4001217731-command-86325-0")
("check-final" "4001217731-command-86347-0")
("targeted-count-final" "4001218074-command-95965-0")))
    (let* ((path (format nil "spikes/out/~A/report.lisp" (second entry)))
           (r (arcdocdb.evidence:read-evidence path))
           (stdout (getf r :stdout))
           (observation
            (list :label (first entry) :record (second entry) :status (getf r :status)
                  :exit-code (getf r :exit-code) :consistency (getf r :source-consistency)
                  :commit (getf (getf r :environment) :commit)
                  :working-tree (getf (getf r :environment) :working-tree)
                  :wall-seconds (getf r :wall-seconds)
                  :sources (length (getf r :source-blobs-after))
                  :stderr-characters (length (getf r :stderr))
                  :result-lines
                  (when stdout
                    (remove-if-not
                     (lambda (line)
                       (or (search "test " line) (search "tests-complete" line)
                           (search "FI, " line) (search "Markdown" line) (search "file verificati" line)
                           (search "req=" line) (search "catalog" line) (search "record: " line)))
                     (uiop:split-string stdout :separator '(#\Newline)))))))
      (write (list (first entry) (getf observation :status) (getf observation :consistency)
                   (getf observation :wall-seconds)) :pretty nil) (terpri)
      (push observation observations)))
  (with-open-file (output "spikes/out/sources-audit/process-summary.lisp"
                          :direction :output :if-exists :supersede)
    (write (nreverse observations) :stream output :pretty t) (terpri output)))
