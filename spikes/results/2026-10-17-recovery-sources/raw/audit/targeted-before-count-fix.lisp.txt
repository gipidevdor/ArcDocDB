(require :asdf)
(setf asdf:*compile-file-failure-behaviour* :error asdf:*compile-file-warnings-behaviour* :error)
(handler-bind ((warning (lambda (condition)
                          (unless (typep condition 'sb-kernel:redefinition-warning) (error condition)))))
  (asdf:load-asd (merge-pathnames "arcdocdb.asd" (truename "./")))
  (asdf:load-system "arcdocdb/tests"))
(let ((*read-eval* nil) (*package* (find-package :arcdocdb.recovery.tests)) (tests nil))
  (dolist (file '("tests/recovery/inventory-sources.lisp" "tests/recovery/inventory-sources-native.lisp"))
    (with-open-file (input file :external-format :utf-8)
      (loop for form = (read input nil :eof) until (eq form :eof)
            when (and (consp form) (eq (car form) (find-symbol "DEFTEST" *package*)))
              do (push (second form) tests))))
  (unless (= 16 (length tests)) (error "Attesi 16 test autonomi; trovati ~D" (length tests)))
  (unless (every #'fboundp tests) (error "Test non definiti al load"))
  (dolist (test (nreverse tests))
    (format t "~&sources-test-start ~A~%" test) (finish-output)
    (funcall test) (format t "ok ~A~%" test))
  (format t "~&sources-tests-complete ~D~%" (length tests)))
