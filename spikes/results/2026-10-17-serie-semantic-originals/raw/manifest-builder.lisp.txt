(require :asdf)
(let* ((root (truename "./")) (files nil) (processes nil))
  (labels ((add (source path) (push (list :source (namestring source) :path path) files))
           (original (source path)
             (add source path)
             (let ((*read-eval* nil))
               (with-open-file (stream source)
                 (let ((data (read stream)))
                   (when (eq (getf data :kind) :compressed-evidence)
                     (add (merge-pathnames (getf data :payload) source)
                          (namestring (merge-pathnames (getf data :payload) path)))))))))
    (loop for (label relative) on (cdr (uiop:command-line-arguments)) by #'cddr
          do (let* ((source (merge-pathnames relative root))
                    (target (format nil "processes/~A/report.lisp" label)))
               (original source target)
               (original (merge-pathnames "conservazione.lisp" source)
                         (format nil "processes/~A/conservazione.lisp" label))
               (push (list :label label :record-path target) processes)))
    (dolist (pair '(("/tmp/arcdocdb-serie-semantic-worker-logs.tar.gz" "raw/evidence-worker-logs.tar.gz")
                    ("/tmp/arcdocdb-serie-semantic-compile.lisp" "raw/compile.lisp.txt")
                    ("/tmp/arcdocdb-serie-semantic-disassemble.lisp" "raw/disassemble.lisp.txt")
                    ("/tmp/arcdocdb-serie-semantic-manifest.lisp" "raw/manifest-builder.lisp.txt")))
      (add (pathname (first pair)) (second pair)))
    (with-open-file (stream (first (uiop:command-line-arguments)) :direction :output :if-exists :error)
      (write (list :schema-version 1 :kind :command-archive-manifest :date "2026-10-17"
                   :files (nreverse files) :processes (nreverse processes))
             :stream stream :readably t :pretty t)
      (terpri stream))))
