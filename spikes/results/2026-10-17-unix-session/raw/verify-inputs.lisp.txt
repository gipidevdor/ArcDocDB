(defun git-output (&rest args)
  (string-right-trim '(#\Newline #\Return)
                    (uiop:run-program (cons "git" args) :output :string :error-output :string)))

(defun executable-input-p (path)
  (or (member path '("Makefile" "arcdocdb.asd") :test #'string=)
      (some (lambda (prefix) (uiop:string-prefix-p prefix path))
            '("src/" "tests/" "tools/" "data/" ".github/" "spikes/SPK-"))))

(defun ancestor-p (sha)
  (multiple-value-bind (out err code)
      (uiop:run-program (list "git" "merge-base" "--is-ancestor" sha "HEAD")
                        :output :string :error-output :string :ignore-error-status t)
    (unless (and (string= out "") (string= err "")) (error "Diagnostica Git inattesa"))
    (zerop code)))

(defun same-rows-p (left right)
  (equal (sort (copy-list left) #'string< :key (lambda (row) (getf row :path)))
         (sort (copy-list right) #'string< :key (lambda (row) (getf row :path)))))

(let* ((args (uiop:command-line-arguments)) (source (first args)) (destination (second args))
       (record (arcdocdb.evidence:read-evidence source))
       (rows (remove-if-not (lambda (row) (executable-input-p (getf row :path)))
                            (getf record :source-blobs-after)))
       (paths (remove-if-not #'executable-input-p
                            (remove "" (uiop:split-string
                                         (uiop:run-program '("git" "ls-files" "-z" "--cached" "--others" "--exclude-standard")
                                                           :output :string)
                                         :separator '(#\Null)) :test #'string=)))
       (hashes (uiop:split-string (apply #'git-output "hash-object" "--" paths) :separator '(#\Newline)))
       (actual (loop for path in paths for hash in hashes collect (list :path path :git-blob hash)))
       (heads nil) (branches nil))
  (unless (and (eq :ok (getf record :status)) (eql 0 (getf record :exit-code))
               (eq :stable (getf record :source-consistency))
               (member "check-core" (getf record :command) :test #'equal))
    (error "Verifica completa non OK/STABLE/check-core"))
  (unless (and (= (length paths) (length hashes)) (same-rows-p rows actual))
    (error "Ingressi finali diversi dalla verifica completa"))
  (when (or (same-rows-p '((:path "src/x" :git-blob "old")) '((:path "src/x" :git-blob "new")))
            (same-rows-p '((:path "src/x" :git-blob "old"))
                         '((:path "src/x" :git-blob "old") (:path "src/y" :git-blob "new"))))
    (error "Controlli negativi del confronto falliti"))
  (unless (string= "" (git-output "stash" "list")) (error "Stash ancora presente"))
  (dolist (field (uiop:split-string (uiop:run-program '("git" "worktree" "list" "--porcelain" "-z") :output :string)
                                  :separator '(#\Null)))
    (when (uiop:string-prefix-p "worktree " field)
      (let* ((path (subseq field 9)) (sha (git-output "-C" path "rev-parse" "HEAD"))
             (dirty (git-output "-C" path "status" "--porcelain")))
        (unless (string= dirty "") (error "Worktree modificato: ~A~%~A" path dirty))
        (unless (ancestor-p sha) (error "HEAD non integrata: ~A ~A" path sha))
        (push (list :path path :head sha :ancestor t :clean t) heads))))
  (dolist (line (uiop:split-string
                 (git-output "for-each-ref" "--format=%(refname) %(objectname)" "refs/heads" "refs/remotes")
                 :separator '(#\Newline)))
    (let* ((split (position #\Space line)) (name (subseq line 0 split)) (sha (subseq line (1+ split))))
      (unless (ancestor-p sha) (error "Branch non integrato: ~A ~A" name sha))
      (push (list :ref name :head sha :ancestor t) branches)))
  (setf heads (nreverse heads) branches (nreverse branches))
  (with-open-file (stream destination :direction :output :if-exists :error)
    (write (list :schema-version 1 :kind :final-integration-input-comparison :status :ok
                 :tested-record (namestring (truename source)) :integrated-head (git-output "rev-parse" "HEAD")
                 :input-count (length rows) :inputs actual :negative-controls 2 :stash-empty t
                 :worktrees heads :branches-before-publication branches
                 :limits '(:exact-executable-input-equality :documentation-and-archives-checked-separately
                           :snapshot-after-task-closure :no-product-qualification-claim))
           :stream stream :pretty t)
    (terpri stream))
  (format t "~D ingressi identici; ~D worktree e ~D ref integrati, puliti e senza stash.~%"
          (length rows) (length heads) (length branches)))
