(defun git-output (&rest args)
  (string-right-trim '(#\Newline #\Return)
                     (uiop:run-program (cons "git" args) :output :string :error-output :string)))

(defun executable-input-p (path)
  (or (member path '("Makefile" "arcdocdb.asd") :test #'string=)
      (some (lambda (prefix) (uiop:string-prefix-p prefix path))
            '("src/" "tests/" "tools/" "data/" ".github/" "spikes/SPK-"))))

(defun ancestor-p (sha)
  (multiple-value-bind (stdout stderr code)
      (uiop:run-program (list "git" "merge-base" "--is-ancestor" sha "HEAD")
                        :output :string :error-output :string :ignore-error-status t)
    (unless (and (string= stdout "") (string= stderr "")) (error "Diagnostica Git inattesa"))
    (zerop code)))

(defparameter *ci-only-inputs*
  '("tools/ci/archive.lisp" "tools/ci/identity.lisp" "tools/ci/measure.lisp"
    "tools/ci/producer.lisp" "tools/ci/publish.lisp" "tools/ci/self-test.lisp"))

(defun verified-row (row actual supplemental)
  (let* ((path (getf row :path)) (same (equal actual (getf row :git-blob)))
         (copy (copy-list row)))
    (unless (or same (and (member path *ci-only-inputs* :test #'equal)
                          (equal actual supplemental)))
      (error "Input non verificato: ~A" path))
    (setf (getf copy :git-blob) actual)
    (append copy (list :verified-by (if same :full-check :ci-self-test)))))

(defun rejected-row-p (row actual supplemental)
  (handler-case (progn (verified-row row actual supplemental) nil) (error () t)))

(let* ((args (uiop:command-line-arguments)) (source (first args))
       (supplement (second args)) (destination (third args))
       (record (arcdocdb.evidence:read-evidence source))
       (ci-record (arcdocdb.evidence:read-evidence supplement))
       (rows (remove-if-not (lambda (row) (executable-input-p (getf row :path)))
                            (getf record :source-blobs-after)))
       (actual-paths (remove-if-not #'executable-input-p
                                   (remove "" (uiop:split-string
                                                (uiop:run-program '("git" "ls-files" "-z" "--cached" "--others" "--exclude-standard")
                                                                  :output :string)
                                                :separator '(#\Null)) :test #'string=)))
       (heads nil) (verified nil) (ci-table (make-hash-table :test 'equal)))
  (unless (and (eq :ok (getf record :status)) (eql 0 (getf record :exit-code))
               (eq :stable (getf record :source-consistency))) (error "Verifica completa non OK/STABLE"))
  (unless (and (eq :ok (getf ci-record :status)) (eql 0 (getf ci-record :exit-code))
               (eq :stable (getf ci-record :source-consistency))
               (member "tools/ci.lisp" (getf ci-record :command) :test #'equal)
               (member "self-test" (getf ci-record :command) :test #'equal))
    (error "Supplemento non e un self-test CI OK/STABLE"))
  (unless (and (rejected-row-p '(:path "src/runtime/types.lisp" :git-blob "old") "new" "new")
               (rejected-row-p '(:path "tools/ci/producer.lisp" :git-blob "old") "new" "different"))
    (error "Controlli negativi del confronto falliti"))
  (dolist (row (getf ci-record :source-blobs-after))
    (setf (gethash (getf row :path) ci-table) (getf row :git-blob)))
  (unless (equal (sort (copy-list actual-paths) #'string<)
                 (sort (mapcar (lambda (row) (getf row :path)) rows) #'string<))
    (error "Insieme degli ingressi eseguibili cambiato"))
  (let ((hashes (uiop:split-string (apply #'git-output "hash-object" "--" actual-paths)
                                  :separator '(#\Newline)))
        (table (make-hash-table :test 'equal)))
    (unless (= (length hashes) (length actual-paths)) (error "Hash incompleti"))
    (loop for path in actual-paths for hash in hashes do (setf (gethash path table) hash))
    (dolist (row rows)
      (push (verified-row row (gethash (getf row :path) table)
                          (gethash (getf row :path) ci-table)) verified)))
  (dolist (field (uiop:split-string (uiop:run-program '("git" "worktree" "list" "--porcelain" "-z") :output :string)
                                  :separator '(#\Null)))
    (when (uiop:string-prefix-p "worktree " field)
      (let* ((path (subseq field 9))
             (sha (git-output "-C" path "rev-parse" "HEAD"))
             (dirty (git-output "-C" path "status" "--porcelain")))
        (unless (string= dirty "") (error "Worktree non pulito: ~A~%~A" path dirty))
        (unless (ancestor-p sha) (error "Head non integrato: ~A ~A" path sha))
        (push (list :path path :head sha :ancestor t :clean t) heads))))
  (with-open-file (stream destination :direction :output :if-exists :error)
    (write (list :schema-version 1 :kind :integration-input-comparison :status :ok
                 :tested-record (namestring (truename source))
                 :ci-only-supplement (namestring (truename supplement))
                 :ci-allowed-paths *ci-only-inputs* :negative-controls 2
                 :integrated-head (git-output "rev-parse" "HEAD")
                 :input-count (length verified) :inputs (nreverse verified) :worktrees (nreverse heads)
                 :branch-state-before-publication (git-output "for-each-ref" "--format=%(refname) %(objectname)" "refs/heads" "refs/remotes")
                 :limits '(:identical-runtime-tests-and-other-inputs-from-full-check
                           :only-six-ci-tools-may-use-the-current-ci-self-test
                           :documentation-checked-separately :results-archives-added-after-full-verification
                           :snapshot-after-coordinated-task-closure :no-claim-of-product-qualification))
           :stream stream :pretty t)
    (terpri stream))
  (format t "~D ingressi identici e ~D worktree puliti e integrati.~%" (length rows) (length heads)))
