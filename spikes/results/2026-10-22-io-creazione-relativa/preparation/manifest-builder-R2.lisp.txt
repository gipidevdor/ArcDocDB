(require :asdf)
(require :sb-posix)
(load "tools/evidence-storage.lisp")
(let* ((root (truename "./"))
       (pub (merge-pathnames "spikes/out/creazione-relativa-publication/" root))
       (rows nil)
       (processes nil))
  (flet ((rows-data (name)
           (with-open-file (in (merge-pathnames name pub))
             (let ((*read-eval* nil) (eof (gensym)))
               (let ((data (read in nil eof)))
                 (assert (and (listp data) (eq eof (read in nil eof)))) data))))
         (add-file (path target)
           (push (list :source (namestring (merge-pathnames path root)) :path target) rows)))
    (dolist (pair '(("source-S1/collector-source-rows.lisp" . "selection/source-rows.lisp")
                    ("fixture-source-rows.lisp" . "selection/fixture-rows.lisp")
                    ("preparation-source-rows.lisp" . "selection/preparation-rows.lisp")))
      (setf rows (append (rows-data (car pair)) rows))
      (add-file (merge-pathnames (car pair) pub) (cdr pair)))
    (add-file (merge-pathnames "README.md.txt" pub) "README.md")
    (dolist (name '("manifest-builder.lisp.txt" "manifest-builder.stdout.txt"
                    "manifest-builder.stderr.txt" "manifest.lisp" "manifest-builder-R2.lisp.txt"
                    "fixture-context.lisp" "fixture-inventory-R1.nul"
                    "prepare-fixture-rows.lisp.txt" "prepare-fixture-rows.stdout.txt"
                    "prepare-fixture-rows.stderr.txt"))
      (add-file (merge-pathnames name pub) (concatenate 'string "preparation/" name)))
    (dolist (spec '(("trace-write" "spikes/out/creazione-relativa-trace-R0/")
                    ("r1-outer" "spikes/out/creazione-relativa-R1/")
                    ("r1-inner" "spikes/out/4001237419-command-89011-0/")
                    ("r2-integration" "spikes/out/creazione-relativa-R2/")))
      (let* ((label (first spec)) (dir (merge-pathnames (second spec) root))
             (report (arcdocdb.evidence:read-evidence (merge-pathnames "report.lisp" dir))))
        (assert (and (eq :ok (getf report :status)) (eql 0 (getf report :exit-code))
                     (eq :stable (getf report :source-consistency))))
        (push (list :label label :record-path (format nil "processes/~A/report.lisp" label)) processes)
        (dolist (file (uiop:directory-files dir))
          (add-file file (format nil "processes/~A/~A" label (file-namestring file))))))
    (setf rows (sort rows #'string< :key (lambda (row) (getf row :path)))
          processes (nreverse processes))
    (with-open-file (out (merge-pathnames "manifest-R2.lisp" pub) :direction :output :if-exists :error)
      (let ((*print-readably* t) (*print-pretty* t))
        (write (list :schema-version 1 :kind :command-archive-manifest :date "2026-10-22"
                     :files rows :processes processes) :stream out)
        (terpri out)))
    (format t "Manifest closed: ~D files, ~D processes.~%" (length rows) (length processes))))
