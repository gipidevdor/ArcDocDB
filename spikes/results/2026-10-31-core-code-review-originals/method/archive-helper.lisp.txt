(require :asdf)
(let ((files nil) (processes nil) (root (truename "./")))
  (labels ((add (source path)
             (push (list :source (namestring source) :path path) files))
           (add-record (source path)
             (add source path)
             (let ((*read-eval* nil))
               (with-open-file (in source)
                 (let ((data (read in)))
                   (when (eq (getf data :kind) :compressed-evidence)
                     (add (merge-pathnames (getf data :payload) source)
                          (namestring (merge-pathnames (getf data :payload) path)))))))))
    (dolist (label '("initial" "full-check" "process-sample" "qualified"))
      (let ((source (merge-pathnames
                     (format nil "spikes/out/core-review-20261031/~A/report.lisp" label) root))
            (path (format nil "processes/~A/report.lisp" label)))
        (add-record source path)
        (add-record (merge-pathnames "conservazione.lisp" source)
                    (format nil "processes/~A/conservazione.lisp" label))
        (push (list :label label :record-path path) processes)))
    (add (truename *load-truename*) "method/archive-helper.lisp.txt")
    (add (truename "spikes/core-code-review.lisp") "method/qualified-driver.lisp.txt")
    (add (truename "/tmp/arcdocdb-core-review-stacks.txt") "diagnostics/process-sample.txt")
    (with-open-file (out "spikes/out/core-review-20261031/archive-input.lisp"
                         :direction :output :if-exists :error)
      (write (list :schema-version 1 :kind :command-archive-manifest :date "2026-10-31"
                   :files (nreverse files) :processes (nreverse processes))
             :stream out :readably t :pretty t)
      (terpri out))))
