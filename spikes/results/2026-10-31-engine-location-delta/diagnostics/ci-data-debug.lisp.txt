(require :asdf)
(require :sb-posix)
(defpackage #:arcdocdb.ci (:use #:cl))
(in-package #:arcdocdb.ci)
(defparameter *ci-root* (truename "./"))
(defparameter *ci-entry* (merge-pathnames "tools/ci.lisp" *ci-root*))
(defparameter *ci-sbcl* (namestring sb-ext:*runtime-pathname*))
(load "tools/evidence-storage.lisp")
(load "tests/ci-catalog.lisp")
(dolist (module '("data" "identity" "support" "tests" "coverage" "measure" "archive" "producer" "github" "publish"
                  "self-test" "data-self-test" "test-self-test" "conditions-self-test" "measure-self-test" "workflow-self-test" "publish-self-test"))
  (load (merge-pathnames (format nil "tools/ci/~A.lisp" module) *ci-root*)))
(let ((original (symbol-function 'ci-working-clean-p)) (count 0))
  (setf (symbol-function 'ci-working-clean-p)
        (lambda ()
          (let ((values (multiple-value-list (funcall original))))
            (incf count)
            (format t "~&CLEAN-CALL ~D ~S~%" count values)
            (when (null (first values))
              (format t "STATUS ~A~%" (ci-git '("status" "--porcelain" "--untracked-files=all"))))
            (values-list values))))
  (let ((report (ci-data-self-test (merge-pathnames "spikes/out/m1-location-delta/data-debug/" *ci-root*))))
    (format t "~&~S~%" (list :status (getf report :status) :passed (getf report :passed) :failed (getf report :failed)))
    (dolist (case (getf report :cases)) (when (eq :failed (getf case :status)) (format t "~S~%" case)))
    (unless (eq :passed (getf report :status)) (sb-ext:exit :code 1))))
