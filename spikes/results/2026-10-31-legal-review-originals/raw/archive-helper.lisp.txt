(require :asdf)
(let* ((root (truename "./")) (files nil) (processes nil))
  (labels ((add (source path) (push (list :source (namestring source) :path path) files))
           (original (source path)
             (add source path)
             (let ((*read-eval* nil))
               (with-open-file (in source)
                 (let ((record (read in)))
                   (when (eq (getf record :kind) :compressed-evidence)
                     (add (merge-pathnames (getf record :payload) source)
                          (namestring (merge-pathnames (getf record :payload) path)))))))))
    (loop for (label relative) on (cdr (uiop:command-line-arguments)) by #'cddr
          do (let* ((source (merge-pathnames relative root))
                    (target (format nil "processes/~A/report.lisp" label)))
               (original source target)
               (original (merge-pathnames "conservazione.lisp" source)
                         (format nil "processes/~A/conservazione.lisp" label))
               (push (list :label label :record-path target) processes)))
    (dolist (pair '(("spikes/out/legal-review-20261031/upstream-audit.json" "raw/upstream-audit-initial.json")
                    ("spikes/out/legal-review-20261031/notice-follow-up.json" "raw/notice-follow-up.json")
                    ("spikes/out/legal-review-20261031/table-payload.json" "raw/table-payload.json")
                    ("spikes/out/legal-review-20261031/algorithm-reference.json" "raw/algorithm-reference.json")
                    ("spikes/out/legal-review-20261031/requests.tsv" "raw/upstream-requests.tsv")
                    ("/tmp/arcdocdb-license-audit.py" "raw/upstream-audit-helper.py.txt")
                    ("/tmp/arcdocdb-legal-archive.lisp" "raw/archive-helper.lisp.txt")))
      (add (merge-pathnames (first pair) root) (second pair)))
    (with-open-file (out (first (uiop:command-line-arguments)) :direction :output :if-exists :error)
      (write (list :schema-version 1 :kind :command-archive-manifest :date "2026-10-31"
                   :files (nreverse files) :processes (nreverse processes))
             :stream out :readably t :pretty t)
      (terpri out))))
