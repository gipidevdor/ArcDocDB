from pathlib import Path
import concurrent.futures, hashlib, json, tarfile, urllib.request

root = Path('/Users/gpicchiarelli/.codex/worktrees/legal-distribution/ArcDocDB')
output = root / 'spikes/out/legal-review-20261031'
cache = Path('/Users/gpicchiarelli/Documents/ArcDocDB/_build/docs/archives')
upstream = output / 'upstream'
upstream.mkdir(exist_ok=False)
notices = (root / 'tools/pages/third-party-notices.txt').read_text()
rows = [line.split('\t') for line in (output / 'requests.tsv').read_text().splitlines()]

def sha(data):
    return hashlib.sha256(data).hexdigest()

def audit(row):
    kind, name, url, expected, target, *extra = row
    result = dict(kind=kind, name=name, url=url, expected_sha256=expected)
    try:
        path = upstream / expected
        cached = cache / expected
        if cached.exists():
            data = cached.read_bytes()
            result['acquisition'] = 'existing-cache-rehashed'
        else:
            request = urllib.request.Request(url, headers={'User-Agent': 'ArcDocDB-license-review/1.0'})
            with urllib.request.urlopen(request, timeout=60) as response:
                data = response.read(33_554_433)
            result['acquisition'] = 'https-upstream'
        if len(data) > 33_554_432 or sha(data) != expected:
            raise ValueError('Archive exceeds budget or SHA256 mismatch')
        if not path.exists():
            path.write_bytes(data)
        if kind == 'build':
            with tarfile.open(path, 'r:gz') as archive:
                files = [m for m in archive.getmembers() if m.isfile() and m.name.startswith(target + '/')]
                candidates = [m for m in files if m.name.count('/') == 1 and
                              any(x in Path(m.name).name.lower() for x in ('license', 'licence', 'copying', 'copyright', 'notice'))]
                if not candidates:
                    candidates = [m for m in files if m.name.count('/') == 1 and
                                  (m.name.endswith('.asd') or Path(m.name).name.lower().startswith('readme'))]
                extracted = []
                destination = upstream / ('build-' + name)
                destination.mkdir(exist_ok=True)
                for member in candidates:
                    if member.size > 1_048_576:
                        raise ValueError('License candidate exceeds budget')
                    payload = archive.extractfile(member).read()
                    (destination / Path(member.name).name).write_bytes(payload)
                    extracted.append(dict(path=member.name, sha256=sha(payload), bytes=len(payload)))
                result['license_evidence'] = extracted
                result['status'] = 'notice-files-collected' if any('license' in x['path'].lower() or 'licence' in x['path'].lower() or 'copying' in x['path'].lower() for x in extracted) else 'review-embedded-notices'
        elif url.endswith('.map'):
            source = json.loads(data)
            matches = [text for key, text in zip(source['sources'], source['sourcesContent']) if key == target.split(' (')[0]]
            if len(matches) != 1:
                raise ValueError('Virtual notice source missing or ambiguous')
            lines = matches[0].splitlines(keepends=True)
            payload = None
            for i in range(1, min(len(lines), 100) + 1):
                candidate = ''.join(lines[:i]).encode()
                if sha(candidate) == extra[0]:
                    payload = candidate
                    break
            if payload is None:
                raise ValueError('Virtual leading notice hash not found')
            if payload.decode() not in notices:
                raise ValueError('Virtual notice not preserved verbatim')
            result.update(status='notice-verified-verbatim', notice_sha256=sha(payload))
        else:
            with tarfile.open(path, 'r:gz') as archive:
                member = archive.getmember('package/' + target)
                if not member.isfile() or member.size > 1_048_576:
                    raise ValueError('Invalid license member')
                payload = archive.extractfile(member).read()
            if sha(payload) != extra[0]:
                raise ValueError('Notice SHA256 mismatch')
            if payload.decode('utf-8-sig') not in notices:
                raise ValueError('License body not preserved verbatim')
            result.update(status='notice-verified-verbatim', notice_sha256=sha(payload))
    except Exception as error:
        result.update(status='failed', error=str(error))
    return result

with concurrent.futures.ThreadPoolExecutor(max_workers=4) as pool:
    results = list(pool.map(audit, rows))
(output / 'upstream-audit.json').write_text(json.dumps(dict(schema=1, date='2026-10-31',
    scope='Pinned archive and notice integrity; no legal opinion or license compatibility certification',
    records=results), indent=2) + '\n')
for result in results:
    if result['status'] != 'notice-verified-verbatim':
        print(result['kind'], result['name'], result['status'], result.get('error', ''), flush=True)
print('SUMMARY', json.dumps({status: sum(x['status'] == status for x in results) for status in sorted({x['status'] for x in results})}), flush=True)
raise SystemExit(1 if any(x['status'] == 'failed' for x in results) else 0)
