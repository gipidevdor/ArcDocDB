(require :asdf)
(let ((files nil) (processes nil) (root (truename "./"))
      (args (uiop:command-line-arguments)))
  (labels ((add (source path)
             (push (list :source (namestring source) :path path) files))
           (original (source path)
             (add source path)
             (let ((*read-eval* nil))
               (with-open-file (in source)
                 (let ((record (read in)))
                   (when (eq (getf record :kind) :compressed-evidence)
                     (add (merge-pathnames (getf record :payload) source)
                          (namestring (merge-pathnames (getf record :payload) path)))))))))
    (dolist (label (cdr args))
      (let* ((source (merge-pathnames (format nil "spikes/out/verify-m1-20261031/~A/report.lisp" label) root))
             (path (format nil "processes/~A/report.lisp" label)))
        (original source path)
        (original (merge-pathnames "conservazione.lisp" source)
                  (format nil "processes/~A/conservazione.lisp" label))
        (push (list :label label :record-path path) processes)))
    (let* ((source (merge-pathnames "spikes/out/4002473968-command-19525-0/report.lisp" root))
           (path "processes/ci-initial-child/report.lisp"))
      (original source path)
      (original (merge-pathnames "conservazione.lisp" source)
                "processes/ci-initial-child/conservazione.lisp")
      (push (list :label "ci-initial-child" :record-path path) processes))
    (dolist (pair '(("spikes/out/ci/selftest-1793485167-19521/tests.lisp" "diagnostics/ci-initial-tests.lisp")
                    ("spikes/out/ci/selftest-1793485167-19521/report.lisp" "diagnostics/ci-initial-summary.lisp")
                    ("spikes/out/ci/selftest-1793485293-20722/tests.lisp" "diagnostics/ci-follow-up-tests.lisp")
                    ("spikes/out/ci/selftest-1793485293-20722/report.lisp" "diagnostics/ci-follow-up-summary.lisp")))
      (add (merge-pathnames (first pair) root) (second pair)))
    (add (truename "spikes/verify-m1-check.lisp") "method/focused-driver.lisp.txt")
    (add (truename *load-truename*) "method/archive-helper.lisp.txt")
    (with-open-file (out (first args) :direction :output :if-exists :error)
      (write (list :schema-version 1 :kind :command-archive-manifest :date "2026-10-31"
                   :files (nreverse files) :processes (nreverse processes))
             :stream out :readably t :pretty t)
      (terpri out))))
