(require :asdf)
(load (merge-pathnames "tools/evidence-storage.lisp" (truename "./")))
(require :sb-posix)
(defparameter *session* (uiop:pathname-directory-pathname *load-truename*))
(defun save-data (data path)
  (with-open-file (stream path :direction :output :if-exists :error)
    (let ((*print-readably* t)) (write data :stream stream :pretty t) (terpri stream))))
(let* ((cleanup (arcdocdb.evidence:read-evidence (merge-pathnames "cleanup.lisp" *session*)))
       (plan (arcdocdb.evidence:read-evidence (merge-pathnames "plan.lisp" *session*)))
       (source (merge-pathnames (getf cleanup :archive) (truename "./")))
       (parent (merge-pathnames ".local/share/arcdocdb/evidence/" (user-homedir-pathname)))
       (directory nil) (archive nil) (inventory nil))
  (unless (and (eq :ok (getf cleanup :status))
               (string= (getf cleanup :archive-sha256) (arcdocdb.evidence:file-sha256 source)))
    (error "Local archive is not the verified output."))
  (ensure-directories-exist (merge-pathnames ".parent" parent))
  (setf directory (merge-pathnames "space-recovery-20261031-GYsdCA/" (truename parent)))
  (sb-posix:mkdir (namestring directory) #o700)
  (setf archive (merge-pathnames "retained-output.tar.zst" directory)
        inventory (merge-pathnames "inventory.sha256" directory))
  (with-open-file (stream inventory :direction :output :if-exists :error)
    (dolist (row (getf plan :files)) (format stream "~A  ~A~%" (getf row :sha256) (getf row :path))))
  ;; Same user data volume; rename never overwrites a file in this exclusive directory.
  (rename-file source archive)
  (unless (string= (getf cleanup :archive-sha256) (arcdocdb.evidence:file-sha256 archive))
    (error "Archive differs after relocation; retained in the recovery directory."))
  (with-open-file (stream (merge-pathnames "SHA256SUMS" directory) :direction :output :if-exists :error)
    (format stream "~A  retained-output.tar.zst~%~A  inventory.sha256~%"
            (getf cleanup :archive-sha256) (arcdocdb.evidence:file-sha256 inventory)))
  (with-open-file (stream (merge-pathnames "README.txt" directory) :direction :output :if-exists :error)
    (format stream "Verified local spike output recovery, 2026-10-31.~%~D files, all original bytes retained.~%~%" (length (getf plan :files)))
    (format stream "Verify archive here: shasum -a 256 --strict --status -c SHA256SUMS~%")
    (format stream "Restore into a NEW separate directory, never over the current checkout:~%")
    (format stream "mkdir /tmp/arcdocdb-spikes-restored-GYsdCA~%")
    (format stream "zstd -dc '~A' | tar -xf - -C /tmp/arcdocdb-spikes-restored-GYsdCA~%" archive)
    (format stream "cd /tmp/arcdocdb-spikes-restored-GYsdCA~%shasum -a 256 --strict --status -c '~A'~%" inventory))
  (save-data (list :schema-version 1 :kind :local-spike-backup-location :status :ok
                   :directory (namestring directory) :archive (namestring archive)
                   :archive-sha256 (getf cleanup :archive-sha256) :inventory (namestring inventory)
                   :files (length (getf plan :files)) :outside-disposable-output t :published-remotely nil)
             (merge-pathnames "backup-location.lisp" *session*))
  (format t "Recovery archive retained outside disposable spike output: ~A~%" directory))
