;;;; Explicit lossless selection; no CI payload or sealed inventory is rewritten.
(require :asdf)
(require :sb-posix)
(load (merge-pathnames "tools/evidence-storage.lisp" (truename "./")))
(defparameter *root* (truename "./"))
(defparameter *session* (uiop:pathname-directory-pathname *load-truename*))
(defun save-data (data path)
  (with-open-file (stream path :direction :output :if-exists :error)
    (let ((*print-readably* t) (*print-pretty* t)) (write data :stream stream) (terpri stream))))
(let ((empty (merge-pathnames "empty-compactor/" *session*)))
  (sb-posix:mkdir (namestring empty) #o700)
  (let ((uiop:*command-line-arguments* (list "--root" (namestring empty) "--jobs" "4")))
    (load (merge-pathnames "tools/compact-evidence.lisp" *root*))))
(let* ((selection (arcdocdb.evidence:read-evidence (merge-pathnames "candidates.lisp" *session*)))
       (before (getf (arcdocdb.evidence:read-evidence (merge-pathnames "published-before.lisp" *session*)) :files))
       (names (getf selection :files)) (paths nil) (baseline nil))
  (unless (and names (listp names) (every #'stringp names)) (error "Invalid explicit candidate list."))
  (dolist (name names)
    (let* ((path (merge-pathnames name *root*))
           (old (find name before :test #'equal :key (lambda (r) (getf r :path)))))
      (loop for directory = (uiop:pathname-directory-pathname path)
              then (uiop:pathname-parent-directory-pathname directory)
            while (uiop:subpathp directory (merge-pathnames "spikes/results/" *root*))
            do (when (probe-file (merge-pathnames "archive-index.lisp" directory))
                 (error "Candidate belongs to a sealed inventory: ~A" name)))
      (unless (and old (uiop:string-prefix-p "spikes/results/" name)
                   (not (uiop:string-prefix-p "spikes/results/ci/" name))
                   (equal "report.lisp" (file-namestring path))
                   (not (probe-file (concatenate 'string (namestring path) ".gz")))
                   (string= (getf old :sha256) (arcdocdb.evidence:file-sha256 path)))
        (error "Candidate changed or outside the approved scope: ~A" name))
      (push path paths) (push old baseline)))
  (setf paths (nreverse paths) baseline (nreverse baseline))
  ;; Verify every existing file outside the selection with one bounded hashing process.
  (let ((checks (merge-pathnames "protected-results.sha256" *session*)))
    (with-open-file (stream checks :direction :output :if-exists :error)
      (dolist (row before)
        (unless (member (getf row :path) names :test #'equal)
          (format stream "~A  ~A~%" (getf row :sha256) (getf row :path)))))
    (let* ((results (compact-many paths 4))
           (ok (every (lambda (row) (eq :ok (getf row :status))) results)))
      (save-data (list :schema-version 1 :kind :published-evidence-compaction
                       :status (if ok :ok :failed) :jobs 4 :files (length results)
                       :original-bytes (reduce #'+ results :key (lambda (r) (getf r :original-bytes 0)))
                       :stored-bytes (reduce #'+ results :key (lambda (r) (getf r :stored-bytes 0)))
                       :results results :source-id-change :expected-retention-representation-change
                       :limits '(:not-a-new-benchmark :checkout-space-only :no-git-history-rewrite))
                 (merge-pathnames "compaction.lisp" *session*))
      (unless ok (error "One or more selected files could not be compacted.")))
    (dolist (old baseline)
      (arcdocdb.evidence:call-with-evidence-bytes
       (merge-pathnames (getf old :path) *root*)
       (lambda (expanded)
         (unless (and (= (getf old :bytes) (arcdocdb.evidence:file-bytes expanded))
                      (string= (getf old :sha256) (arcdocdb.evidence:file-sha256 expanded)))
           (error "Reconstructed original differs: ~A" (getf old :path))))))
    (uiop:run-program (list "shasum" "-a" "256" "--strict" "--status" "-c" (namestring checks))
                      :output :interactive :error-output :interactive))
  (save-data (list :schema-version 1 :kind :published-evidence-equivalence :status :ok
                   :compacted-files (length paths) :all-original-byte-streams-preserved t
                   :protected-files-verified (- (length before) (length paths))
                   :ci-and-closed-archives-unchanged t :engine-source-files-modified nil)
             (merge-pathnames "equivalence.lisp" *session*))
  (format t "~D selected reports reconstruct exactly; all other existing published files have identical SHA-256.~%"
          (length paths)))
