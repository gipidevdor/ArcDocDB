(load (merge-pathnames "tools/evidence-storage.lisp" (truename "./")))
(defparameter *repository* (truename "./"))
(defparameter *session* (uiop:pathname-directory-pathname *load-truename*))
(defparameter *publication* (merge-pathnames "spikes/results/2026-10-31-space-recovery/" *repository*))
(defun save-data (data path)
  (ensure-directories-exist path)
  (with-open-file (stream path :direction :output :if-exists :error)
    (let ((*print-readably* t)) (write data :stream stream :pretty t) (terpri stream))))
(let ((empty (merge-pathnames "publication-empty-compactor/" *session*)))
  (sb-posix:mkdir (namestring empty) #o700)
  (let ((uiop:*command-line-arguments* (list "--root" (namestring empty) "--jobs" "4")))
    (load (merge-pathnames "tools/compact-evidence.lisp" *repository*))))
(defun copy-verified (source relative &key (compact t))
  (let* ((destination (merge-pathnames relative *publication*))
         (bytes (arcdocdb.evidence:file-bytes source)) (hash (arcdocdb.evidence:file-sha256 source)))
    (ensure-directories-exist destination)
    (with-open-file (in source :element-type '(unsigned-byte 8))
      (with-open-file (out destination :direction :output :if-exists :error :element-type '(unsigned-byte 8))
        (let ((buffer (make-array 65536 :element-type '(unsigned-byte 8))))
          (loop for count = (read-sequence buffer in) while (plusp count) do
            (write-sequence buffer out :end count)))))
    (unless (and (= bytes (arcdocdb.evidence:file-bytes destination))
                 (string= hash (arcdocdb.evidence:file-sha256 destination))
                 (string= hash (arcdocdb.evidence:file-sha256 source)))
      (error "Publication copy differs: ~A" source))
    (when (and compact (> bytes 4096) (not (equal (pathname-type source) "gz")))
      (compact-one destination))))
(defun all-files (directory)
  (append (uiop:directory-files directory) (loop for child in (uiop:subdirectories directory) append (all-files child))))

(let* ((cleanup (arcdocdb.evidence:read-evidence (merge-pathnames "cleanup.lisp" *session*)))
       (compaction (arcdocdb.evidence:read-evidence (merge-pathnames "compaction.lisp" *session*)))
       (equivalence (arcdocdb.evidence:read-evidence (merge-pathnames "equivalence.lisp" *session*)))
       (backup (arcdocdb.evidence:read-evidence (merge-pathnames "backup-location.lisp" *session*)))
       (commands nil)
       (reduction (+ (- (getf cleanup :original-logical-bytes) (getf cleanup :archive-bytes))
                     (- (getf compaction :original-bytes) (getf compaction :stored-bytes)))))
  (dolist (row (list cleanup compaction equivalence backup))
    (unless (eq :ok (getf row :status)) (error "A phase did not pass.")))
  (unless (string= (getf backup :archive-sha256)
                   (arcdocdb.evidence:file-sha256 (getf backup :archive)))
    (error "Persistent recovery archive differs."))
  (dolist (name '("before" "reclaim" "compact" "evidence" "reader" "backup"))
    (let ((row (arcdocdb.evidence:read-evidence
                (merge-pathnames (format nil "records/~A/report.lisp" name) *session*))))
      (unless (and (eq :ok (getf row :status)) (eql 0 (getf row :exit-code))
                   (eq :stable (getf row :source-consistency)))
        (error "Command did not finish successfully with stable sources: ~A" name))
      (push (list :name name :status (getf row :status) :exit-code (getf row :exit-code)
                  :source-consistency (getf row :source-consistency)) commands)))
  (sb-posix:mkdir (namestring *publication*) #o755)
  (dolist (name '("method.lisp" "assessment.lisp" "candidates.lisp" "plan.lisp" "published-before.lisp"
                  "cleanup.lisp" "compaction.lisp" "equivalence.lisp" "backup-location.lisp"))
    (copy-verified (merge-pathnames name *session*) name))
  (dolist (name '("reclaim.lisp" "compact-results.lisp" "backup.lisp" "publish.lisp"))
    (copy-verified (merge-pathnames name *session*) (format nil "raw/~A.txt" name) :compact nil))
  (dolist (name '("before" "reclaim" "compact" "evidence" "reader" "backup"))
    (dolist (side '("report.lisp" "report.lisp.gz" "conservazione.lisp"))
      (let ((source (merge-pathnames (format nil "records/~A/~A" name side) *session*)))
        (when (probe-file source)
          (copy-verified source (format nil "processes/~A/~A" name side))))))
  (let ((before (arcdocdb.evidence:read-evidence (merge-pathnames "records/before/report.lisp" *session*))))
    (save-data
     (list :schema-version 1 :kind :space-recovery-summary :status :ok :date "2026-10-31"
           :local-files (getf cleanup :files) :local-original-bytes (getf cleanup :original-logical-bytes)
           :archive-bytes (getf cleanup :archive-bytes) :archive (getf backup :archive)
           :archive-sha256 (getf backup :archive-sha256)
           :compacted-published-reports (getf compaction :files)
           :published-original-bytes (getf compaction :original-bytes)
           :published-stored-bytes (getf compaction :stored-bytes)
           :selected-data-reduction-bytes reduction :receipt-and-backup-metadata-excluded t
           :protected-published-files (getf equivalence :protected-files-verified)
           :commands (nreverse commands) :allocated-kib-before (getf before :stdout)
           :allocated-kib-after (uiop:run-program '("du" "-sk" "spikes") :output :string)
           :limits '(:local-archive-not-a-remote-asset :historical-failures-preserved
                     :concurrent-development-may-change-total-checkout-space
                     :source-id-changes-for-new-retention-representation :no-product-benchmark
                     :no-engine-source-or-api-change-by-this-operation :no-git-history-rewrite))
     (merge-pathnames "report.lisp" *publication*)))
  (with-open-file (stream (merge-pathnames "report.md" *publication*) :direction :output :if-exists :error)
    (format stream "# Spike output recovery — 2026-10-31~%~%")
    (format stream "**~:D bytes** removed from selected representations, before the small recovery receipts and backup inventory. Logical sizes are distinct from allocated filesystem blocks.~%~%" reduction)
    (format stream "- ~D completed local files: ~:D bytes retained as a ~:D-byte TAR/zstd archive.~%"
            (getf cleanup :files) (getf cleanup :original-logical-bytes) (getf cleanup :archive-bytes))
    (format stream "- ~D published reports: ~:D → ~:D bytes, preserving original paths through the existing gzip evidence reader.~%"
            (getf compaction :files) (getf compaction :original-bytes) (getf compaction :stored-bytes))
    (format stream "- Every reconstructed original compared against its preoperation SHA-256; ~D other preexisting published files checked unchanged.~%"
            (getf equivalence :protected-files-verified))
    (format stream "- Existing evidence audit and reader controls passed, including hashes and invalid-input handling. This is storage maintenance, not a new engine test or performance campaign.~%~%")
    (format stream "## Local recovery~%~%The archive is retained outside disposable output at:~%~%`~A`~%~%" (getf backup :archive))
    (format stream "SHA-256: `~A`. Adjacent `README.txt`, `SHA256SUMS` and `inventory.sha256` describe restoration into a new separate directory and verification of all ~D files. This local archive is not a GitHub asset.~%~%"
            (getf backup :archive-sha256) (getf cleanup :files))
    (format stream "## Preserved function and attribution~%~%")
    (format stream "Metrics, samples, parameters, source identities and failed outcomes are retained byte for byte. CI payloads and sealed inventories were preserved. Sources, tests and APIs are not modified by this operation. Other worktrees and active documentation output were excluded.~%~%")
    (format stream "The 32 selected report streams belong to the local index benchmark and local performance fix campaigns. Static selection and method are retained as data; old execution SHAs remain historical identities. The new representation changes future CI SOURCE-ID, without creating a new measurement.~%~%")
    (format stream "The static assessment identifies its reviewed driver hashes. Final drivers additionally enforce ancestor confinement, repeat the tracked-file guard before removal, and create the empty compactor bootstrap exclusively; the reviewer accepted those described deltas before closure.~%~%")
    (format stream "Original command records, before inventories, driver source and the equivalence record accompany this summary. The publication's own command record remains outside its output to keep the receipt finite. Total checkout size may also change because of concurrent development.~%"))
  (save-data
   (list :schema-version 1 :kind :space-recovery-receipt-inventory :status :ok
         :self-excluded t :files
         (loop for path in (all-files *publication*) collect
           (list :path (enough-namestring path *publication*) :bytes (arcdocdb.evidence:file-bytes path)
                 :sha256 (arcdocdb.evidence:file-sha256 path))))
   (merge-pathnames "inventory.lisp" *publication*))
  (format t "Recovery receipt published; selected representations reduced by ~D bytes.~%" reduction))
