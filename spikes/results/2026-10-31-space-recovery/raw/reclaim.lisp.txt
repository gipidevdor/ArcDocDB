;;;; Local maintenance only: archive and independently restore before unlinking.
(require :asdf)
(require :sb-posix)
(load (merge-pathnames "tools/evidence-storage.lisp" (truename "./")))
(defparameter *root* (truename "./"))
(defparameter *session* (uiop:pathname-directory-pathname *load-truename*))
(defparameter *minimum-age* 7200)
(defvar *owners* nil)
(defvar *prune-roots* nil)
(defvar *skipped* nil)

(defun save-data (data path)
  (ensure-directories-exist path)
  (with-open-file (out path :direction :output :if-exists :error)
    (let ((*print-readably* t) (*print-pretty* t)) (write data :stream out) (terpri out))))

(defun relative (path) (enough-namestring path *root*))
(defun absolute (name) (merge-pathnames name *root*))
(defun safe-stat (path)
  (unless (or (equal path *root*) (uiop:subpathp path *root*))
    (error "Input outside the physical workspace: ~A" path))
  (loop for parent = (uiop:pathname-directory-pathname path)
          then (uiop:pathname-parent-directory-pathname parent)
        until (equal parent *root*)
        do (let ((mode (sb-posix:stat-mode
                         (sb-posix:lstat (string-right-trim "/" (namestring parent))))))
             (unless (sb-posix:s-isdir mode) (error "Non-directory ancestor: ~A" parent))))
  (let* ((name (string-right-trim "/" (namestring path))) (stat (sb-posix:lstat name)))
    (when (sb-posix:s-islnk (sb-posix:stat-mode stat)) (error "Symbolic input: ~A" path))
    stat))
(defun regular-files (directory)
  (unless (sb-posix:s-isdir (sb-posix:stat-mode (safe-stat directory)))
    (error "Not a directory: ~A" directory))
  (append
   (loop for path in (uiop:directory-files directory)
         do (unless (sb-posix:s-isreg (sb-posix:stat-mode (safe-stat path)))
              (error "Nonregular file: ~A" path)) collect path)
   (loop for child in (uiop:subdirectories directory) append (regular-files child))))
(defun sha-rows (paths)
  "Batch hashing preserves output order and avoids a process per small file."
  (let ((remaining (sort (copy-list paths) #'string< :key #'namestring)) (rows nil))
    (loop while remaining do
      (let* ((count (min 64 (length remaining))) (batch (subseq remaining 0 count))
             (names (mapcar #'namestring batch)))
        (dolist (name names)
          (when (or (find #\Newline name) (find #\Return name) (find #\\ name))
            (error "Unsupported inventory filename: ~A" name)))
        (let* ((text (uiop:run-program (append '("shasum" "-a" "256" "--") names) :output :string))
               (lines (remove "" (uiop:split-string text :separator '(#\Newline)) :test #'equal)))
          (unless (= count (length lines)) (error "Incomplete SHA inventory."))
          (loop for path in batch for line in lines for stat = (safe-stat path) do
            (unless (and (>= (length line) 66) (equal "  " (subseq line 64 66))
                         (equal (namestring path) (subseq line 66)))
              (error "Unexpected SHA row: ~A" path))
            (push (list :path (relative path) :bytes (sb-posix:stat-size stat)
                        :device (sb-posix:stat-dev stat) :inode (sb-posix:stat-ino stat)
                        :mtime (sb-posix:stat-mtime stat) :sha256 (subseq line 0 64)) rows)))
        (setf remaining (nthcdr count remaining))))
    (nreverse rows)))
(defun pid-live-p (pid)
  (handler-case (progn (sb-posix:kill pid 0) t)
    (sb-posix:syscall-error (condition)
      (/= (sb-posix:syscall-errno condition) sb-posix:esrch))))
(defun old-files-p (files)
  (let ((now (- (get-universal-time) 2208988800)))
    (every (lambda (p) (>= (- now (sb-posix:stat-mtime (safe-stat p))) *minimum-age*)) files)))
(defun standard-owner (directory)
  (let* ((name (car (last (pathname-directory directory))))
         (parts (uiop:split-string name :separator '(#\-))))
    (when (and (= 4 (length parts))
               (every #'digit-char-p (first parts))
               (member (second parts) '("check" "command" "bench" "profile") :test #'equal)
               (every #'digit-char-p (third parts)) (every #'digit-char-p (fourth parts)))
      (parse-integer (third parts)))))
(defun completed-record-p (directory)
  (handler-case
      (let ((data (arcdocdb.evidence:read-evidence (merge-pathnames "report.lisp" directory))))
        (and (eql 1 (getf data :schema-version))
             (member (getf data :status) '(:ok :failed :source-changed :complete :completed :passed :unsupported))))
    (error () nil)))
(defun consider-directory (directory &optional owner)
  (handler-case
      (let ((files (regular-files directory)))
        (if (and files (old-files-p files) (or (not owner) (not (pid-live-p owner)))
                 (completed-record-p directory))
            (progn (when owner (pushnew owner *owners*)) (push directory *prune-roots*) files)
            (progn (push (list :path (relative directory) :reason :recent-active-or-incomplete) *skipped*) nil)))
    (error (condition)
      (push (list :path (relative directory) :reason :unsupported-tree
                  :detail (princ-to-string condition)) *skipped*) nil)))
(defun select-files ()
  (let ((selected nil) (out (absolute "spikes/out/")))
    (dolist (directory (uiop:subdirectories out))
      (let ((owner (standard-owner directory)))
        (when owner (setf selected (nconc selected (consider-directory directory owner))))))
    ;; Preserve custom scripts and active documentation output. Only stale compiler logs.
    (let ((directory (absolute "spikes/out/auto-completion/")))
      (when (probe-file directory)
        (dolist (path (uiop:directory-files directory))
          (when (and (equal (pathname-type path) "log") (old-files-p (list path))
                     (sb-posix:s-isreg (sb-posix:stat-mode (safe-stat path))))
            (push path selected)))))
    (let ((directory (absolute "spikes/out/ci/")))
      (when (probe-file directory)
        (dolist (child (uiop:subdirectories directory))
          (when (uiop:string-prefix-p "selftest-" (car (last (pathname-directory child))))
            (setf selected (nconc selected (consider-directory child)))))))
    ;; Only FASL caches: sources and old SPK records remain at their current paths.
    (dolist (spike (uiop:subdirectories (absolute "spikes/")))
      (when (uiop:string-prefix-p "SPK-" (car (last (pathname-directory spike))))
        (let ((directory (merge-pathnames "out/" spike)))
          (when (probe-file directory)
            (handler-case
                (dolist (path (regular-files directory))
                  (when (and (equal (pathname-type path) "fasl") (old-files-p (list path)))
                    (push path selected)))
              (error (condition)
                (push (list :path (relative directory) :reason :unsupported-cache-tree
                            :detail (princ-to-string condition)) *skipped*)))))))
    (remove-duplicates selected :test #'equal)))
(defun compare-rows (expected actual &key metadata)
  (unless (= (length expected) (length actual)) (error "Inventory cardinality changed."))
  (loop for old in expected for new in actual do
    (dolist (key (if metadata '(:path :bytes :sha256 :device :inode :mtime) '(:path :bytes :sha256)))
      (unless (equal (getf old key) (getf new key))
        (error "Input differs at ~A: ~A" (getf old :path) key)))))
(defun walk-empty (directory)
  ;; Never remove a root passed by the caller. New files prevent empty-dir deletion.
  (dolist (child (uiop:subdirectories directory))
    (when (sb-posix:s-isdir (sb-posix:stat-mode (safe-stat child)))
      (walk-empty child)
      (unless (or (uiop:directory-files child) (uiop:subdirectories child))
        (sb-posix:rmdir (string-right-trim "/" (namestring child)))))))
(defun reject-tracked (paths)
  (let ((tracked (uiop:split-string (uiop:run-program '("git" "ls-files" "-z") :output :string)
                                  :separator '(#\Null))))
    (dolist (path paths)
      (when (member (relative path) tracked :test #'equal) (error "Tracked candidate: ~A" path)))))

(let* ((paths (select-files)) (rows (sha-rows paths))
       (published (sha-rows (regular-files (absolute "spikes/results/"))))
       (plan (list :schema-version 1 :kind :local-output-recovery-plan :status :planned
                   :created-at (get-universal-time) :minimum-age-seconds *minimum-age*
                   :files rows :owners *owners* :skipped *skipped*
                   :preserves '(:all-published-results :custom-lisp-drivers :active-docs-output
                                :recent-files :live-pids :other-worktrees :git-history))))
  (unless paths (error "No completed output selected."))
  (reject-tracked paths)
  (save-data plan (merge-pathnames "plan.lisp" *session*))
  (save-data (list :kind :published-files-before :files published)
             (merge-pathnames "published-before.lisp" *session*))
  (let* ((names (merge-pathnames "members.txt" *session*))
         (tar (merge-pathnames "originals.tar" *session*))
         (archive (merge-pathnames "retained-output.tar.zst" *session*))
         (roundtrip (merge-pathnames "roundtrip.tar" *session*))
         (restored (merge-pathnames "restored/" *session*)))
    (with-open-file (stream names :direction :output :if-exists :error)
      (dolist (row rows) (write-line (getf row :path) stream)))
    (sb-posix:setenv "COPYFILE_DISABLE" "1" 1)
    (uiop:run-program (list "/usr/bin/tar" "-cf" (namestring tar) "-C" (namestring *root*)
                            "-T" (namestring names)) :output :interactive :error-output :interactive)
    (uiop:run-program (list "zstd" "-T4" "-9" "--no-progress" (namestring tar) "-o" (namestring archive))
                      :output :interactive :error-output :interactive)
    (uiop:run-program (list "zstd" "-d" "--no-progress" (namestring archive) "-o" (namestring roundtrip))
                      :output :interactive :error-output :interactive)
    (unless (string= (arcdocdb.evidence:file-sha256 tar) (arcdocdb.evidence:file-sha256 roundtrip))
      (error "TAR stream roundtrip differs."))
    (let* ((listing (uiop:run-program (list "/usr/bin/tar" "-tf" (namestring roundtrip)) :output :string))
           (members (remove "" (uiop:split-string listing :separator '(#\Newline)) :test #'equal)))
      (unless (equal (mapcar (lambda (row) (getf row :path)) rows) members)
        (error "Archive membership differs from the explicit allowlist.")))
    (sb-posix:mkdir (namestring restored) #o700)
    (uiop:run-program (list "/usr/bin/tar" "-xf" (namestring roundtrip) "-C" (namestring restored))
                      :output :interactive :error-output :interactive)
    (let* ((*root* (truename restored)) (reconstructed (sha-rows (regular-files *root*))))
      (compare-rows rows reconstructed))
    ;; All source and archive validation completes before the first unlink.
    (compare-rows rows (sha-rows paths) :metadata t)
    (reject-tracked paths)
    (when (some #'pid-live-p *owners*) (error "A recorded owner PID now exists; preserve originals."))
    (dolist (path paths) (delete-file path))
    (uiop:delete-directory-tree restored :validate t)
    (delete-file tar) (delete-file roundtrip)
    ;; Only empty, ignored top-level run directories and private cache directories are pruned.
    (dolist (directory *prune-roots*)
      (walk-empty directory)
      (unless (or (uiop:directory-files directory) (uiop:subdirectories directory))
        (sb-posix:rmdir (string-right-trim "/" (namestring directory)))))
    (compare-rows published (sha-rows (regular-files (absolute "spikes/results/"))))
    (save-data
     (list :schema-version 1 :kind :local-output-recovery :status :ok
           :files (length rows) :original-logical-bytes (reduce #'+ rows :key (lambda (r) (getf r :bytes)))
           :archive (relative archive) :archive-bytes (arcdocdb.evidence:file-bytes archive)
           :archive-sha256 (arcdocdb.evidence:file-sha256 archive)
           :verification '(:exact-member-list :tar-stream-roundtrip :independent-extraction
                           :all-original-file-sha256 :source-recheck-before-unlink :published-files-unchanged)
           :limits '(:local-archive :external-load-uncontrolled :no-engine-performance-claim
                     :directory-metadata-not-archived :archives-require-explicit-restoration))
     (merge-pathnames "cleanup.lisp" *session*))
    (format t "Recovered ~D old files into ~D archive bytes; published results unchanged.~%"
            (length rows) (arcdocdb.evidence:file-bytes archive))))
