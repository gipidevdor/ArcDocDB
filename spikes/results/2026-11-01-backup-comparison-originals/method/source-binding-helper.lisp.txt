(require :asdf)
(load "tools/evidence-storage.lisp")
(defun comparison-proof-blob (path)
  (string-trim '(#\Newline #\Return)
               (uiop:run-program (list "git" "hash-object" path) :output :string)))
(defun comparison-proof-record (label)
  (let ((data (arcdocdb.evidence:read-evidence
               (format nil "spikes/out/backup-compare-20261101/~A/report.lisp" label))))
    (unless (eq :stable (getf data :source-consistency)) (error "Unstable record: ~A" label)) data))
(defun comparison-proof-source (data path)
  (getf (find path (getf data :source-blobs-before) :test #'string=
                   :key (lambda (row) (getf row :path))) :git-blob))
(let* ((driver "spikes/backup-compare-bench.lisp") (driver-blob (comparison-proof-blob driver))
       (timings nil) (variants nil) (qualified (comparison-proof-record "qualification"))
       (src (remove-if-not (lambda (row) (uiop:string-prefix-p "src/" (getf row :path)))
                           (getf qualified :source-blobs-before))) (bindings nil))
  (dolist (label '("baseline" "optimized" "repeat" "equalp" "equalp-repeat" "final-benchmark"))
    (let ((data (comparison-proof-record label)))
      (unless (and (eq :ok (getf data :status)) (eql 0 (getf data :exit-code))
                   (equal driver-blob (comparison-proof-source data driver)))
        (error "Timing driver mismatch: ~A" label))
      (push (list :label label :timing-driver-blob driver-blob) timings)))
  (dolist (pair '(("baseline-copy.lisp.txt" "baseline" "src/backup/copy.lisp")
                  ("driver-before-syntax-fix.lisp.txt" "driver-self-test" "spikes/backup-compare-bench.lisp")
                  ("copy-before-syntax-fix.lisp.txt" "product-self-test" "src/backup/copy.lisp")
                  ("mismatch-copy.lisp.txt" "optimized" "src/backup/copy.lisp")))
    (let* ((actual (comparison-proof-blob (concatenate 'string "spikes/out/backup-compare-20261101/" (first pair))))
           (expected (comparison-proof-source (comparison-proof-record (second pair)) (third pair))))
      (unless (equal actual expected) (error "Preserved source variant mismatch: ~A" pair))
      (push (list :variant (first pair) :record (second pair) :source-path (third pair)
                  :snapshot-blob expected :retained-blob actual :comparison :equal) variants)))
  (dolist (label '("final-benchmark" "qualification" "docs" "docs-runtime" "ci-selftest"))
    (let* ((data (comparison-proof-record label))
           (actual (remove-if-not (lambda (row) (uiop:string-prefix-p "src/" (getf row :path)))
                                   (getf data :source-blobs-before))))
      (unless (and (eq :ok (getf data :status)) (eql 0 (getf data :exit-code)) (equal src actual))
        (error "Final production source mismatch: ~A" label))
      (push (list :label label :source-files (length src) :comparison :equal) bindings)))
  (with-open-file (out "spikes/out/backup-compare-20261101/source-identity.lisp"
                       :direction :output :if-exists :error)
    (write (list :schema-version 1 :kind :comparison-source-identity :status :ok
                 :timing-driver (nreverse timings) :preserved-variants (nreverse variants)
                 :final-production-bindings (nreverse bindings) :production-inventory src
                 :prior-ci-artifact '(:run-id 41311168018 :job-id 124861769262 :artifact-id 13085950153
                   :sha256 "2ae98c372f39dc90db41105ea9bb5f5fa8a805e422b30e5d48ce52052b6acc74")
                 :limits '(:source-binding :not-runtime-or-performance-qualification))
           :stream out :readably t :pretty t)
    (terpri out)))
(format t "Timing driver, four retained source variants and final production inventories match.~%")
