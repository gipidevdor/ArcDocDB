(let ((files nil) (processes nil) (root (truename "./"))
      (args (uiop:command-line-arguments)))
  (labels ((add (source path)
             (push (list :source (namestring source) :path path) files))
           (original (source path)
             (add source path)
             (let ((*read-eval* nil))
               (with-open-file (in source)
                 (let ((record (read in)))
                   (when (eq (getf record :kind) :compressed-evidence)
                     (add (merge-pathnames (getf record :payload) source)
                          (namestring (merge-pathnames (getf record :payload) path)))))))))
    (dolist (label (cdr args))
      (let* ((source (merge-pathnames (format nil "spikes/out/backup-m1-20261101/~A/report.lisp" label) root))
             (path (format nil "processes/~A/report.lisp" label)))
        (original source path)
        (original (merge-pathnames "conservazione.lisp" source)
                  (format nil "processes/~A/conservazione.lisp" label))
        (push (list :label label :record-path path) processes)))
    (let* ((source (merge-pathnames "spikes/out/4002527808-command-21079-0/report.lisp" root))
           (path "processes/full-check-ci-child/report.lisp"))
      (original source path)
      (original (merge-pathnames "conservazione.lisp" source)
                "processes/full-check-ci-child/conservazione.lisp")
      (push (list :label "full-check-ci-child" :record-path path) processes))
    (dolist (pair '(("spikes/out/ci/selftest-1793539008-21075/tests.lisp" "diagnostics/ci-tests.lisp")
                    ("spikes/out/ci/selftest-1793539008-21075/report.lisp" "diagnostics/ci-summary.lisp")))
      (add (merge-pathnames (first pair) root) (second pair)))
    (dolist (pair '(("spikes/backup-m1-check.lisp" "method/focused-driver.lisp.txt")
                    ("spikes/backup-m1-load.lisp" "method/build-driver.lisp.txt")

                    ("/tmp/arcdocdb-backup-prefix-proof.lisp" "method/baseline-proof-helper.lisp.txt")))
      (add (truename (first pair)) (second pair)))
    (add (truename *load-truename*) "method/archive-helper.lisp.txt")

      (write (list :schema-version 1 :kind :command-archive-manifest :date "2026-11-01"
